"""Enumerate stale Codex shell snapshots that record the Claude messaging-token variable; print names only.

Never prints a value or a line of any file. Output: every regular file directly under ~/.codex/shell_snapshots with
its size, mtime (UTC) and whether it contains a declaration of the variable, then the count and a sha256 over the sorted
matching names (the set assertion for the deletion step), and any process whose open files include a snapshot."""
import hashlib
import os
import pathlib
import time

VAR = "CLAUDE_CODE_" + "MESSAGING_" + "TOKEN"  # built from parts so this file's text is not a secret-name line
ROOT = pathlib.Path.home() / ".codex" / "shell_snapshots"

matches, others = [], []
for entry in sorted(ROOT.iterdir()) if ROOT.is_dir() else []:
    st = entry.lstat()
    if not entry.is_file() or entry.is_symlink():
        others.append(f"{entry.name} (not a regular file)")
        continue
    data = entry.read_bytes()
    hit = any(line.lstrip().startswith((f"declare -x {VAR}=".encode(), f"export {VAR}=".encode()))
              for line in data.splitlines())
    stamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(st.st_mtime))
    (matches if hit else others).append(f"{entry.name} {st.st_size} {stamp}")
print(f"directory: ~/.codex/shell_snapshots exists={ROOT.is_dir()}")
print(f"files declaring the variable: {len(matches)}")
for line in matches:
    print("  M", line)
print(f"other entries: {len(others)}")
for line in others:
    print("  O", line)
names = sorted(line.split(" ", 1)[0] for line in matches)
print("set sha256:", hashlib.sha256("\n".join(names).encode()).hexdigest())

open_by = []
for pid in filter(str.isdigit, os.listdir("/proc")):
    fd_dir = f"/proc/{pid}/fd"
    try:
        for fd in os.listdir(fd_dir):
            try:
                target = os.readlink(f"{fd_dir}/{fd}")
            except OSError:
                continue
            if target.startswith(str(ROOT) + "/"):
                open_by.append((pid, os.path.basename(target)))
    except OSError:
        continue
print(f"open snapshot handles: {len(open_by)}", open_by[:10])
