#!/bin/bash
# Control for the smoke script: an UNRELATED process answers {"status":"ok"} on the port the smoke is told to use.
# SMOKE_SCRIPT selects the smoke script, DUMMY_PORT the port of the dummy (the old script hard-codes 29999).
# Expected for the fixed script: the smoke does not pass (health_ok=no, listener_owned_by_child=no, exit 1) and it does not kill the unrelated process.
set -u
SMOKE=${SMOKE_SCRIPT:?set SMOKE_SCRIPT to the smoke script to test (scripts/smoke.sh.txt)}
PORT=${DUMMY_PORT:-$(python3 -c 'import socket; s = socket.socket(); s.bind(("127.0.0.1", 0)); print(s.getsockname()[1]); s.close()')}
python3 - "$PORT" <<'PYEOF' &
import http.server, sys
class H(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        body = b'{"status":"ok"}'
        self.send_response(200); self.send_header("Content-Type", "application/json"); self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def log_message(self, *a):
        pass
http.server.HTTPServer(("127.0.0.1", int(sys.argv[1])), H).serve_forever()
PYEOF
DUMMY=$!
sleep 1
echo "unrelated dummy health server: pid $DUMMY on port $PORT, alive=$(kill -0 "$DUMMY" 2>/dev/null && echo yes || echo no)"
SMOKE_PORT=$PORT bash "$SMOKE" "${1:?prefix name}"
echo "smoke exit=$?"
echo "unrelated dummy still alive after the smoke: $(kill -0 "$DUMMY" 2>/dev/null && echo yes || echo no)"
kill "$DUMMY" 2>/dev/null
wait "$DUMMY" 2>/dev/null
echo "dummy stopped by this control (its own pid only)"
