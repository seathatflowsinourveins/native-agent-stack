#!/usr/bin/env python3
"""A fixture client for the pre-switch guard's controls: it stands in for the 20129 framework gateway's connections to 20128. It is run inside a transient systemd scope (`systemd-run --user --scope --unit=<name>`), which gives its
process a cgroup unit of a known name for the guard's --tolerate-unit.
  --mode empty    (default) what the framework gateway's poll is on the live host: a connection with NO byte on it, held for --hold seconds, then closed and replaced by a new one (a guard sampling every couple of seconds
                  sees a connection that is new during its window, with empty queues and no bytes received or sent)
  --mode request  a REAL request on the connection, as inference forwarded through the same unit would be: one long-lived connection on which a request is sent in slow pieces (a header piece every 0.5 s) for --hold seconds, then replaced
usage: poller.py --port N [--duration 60] [--mode empty|request] [--hold 2.5]
"""
import argparse
import socket
import time

ap = argparse.ArgumentParser()
ap.add_argument("--port", type=int, required=True)
ap.add_argument("--duration", type=float, default=60.0)
ap.add_argument("--mode", choices=["empty", "request"], default="empty")
ap.add_argument("--hold", type=float, default=2.5)
args = ap.parse_args()
end = time.time() + args.duration
while time.time() < end:
    try:
        s = socket.create_connection(("127.0.0.1", args.port), timeout=5)
        if args.mode == "request":
            s.sendall(b"POST /v1/responses HTTP/1.1\r\nHost: x\r\n")
            until = min(end, time.time() + args.hold)
            while time.time() < until:
                time.sleep(0.5)
                s.sendall(b"X-Slow: 1\r\n")
        else:
            time.sleep(args.hold)
        s.close()
    except OSError:
        pass
    time.sleep(0.02)   # a gap this short between connections keeps a sampling guard from lining up with it (a 0.2 s gap and a 2 s sampling interval phase-lock: every sample can fall into the gap)
