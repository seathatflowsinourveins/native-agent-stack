#!/usr/bin/env python3
"""Offline controls of the window census gate (omniroute-window-census-20261006.py, version 1; new, not frozen). Nothing here touches a live gateway: every scenario builds a scratch SQLite store from the gateway's schema, synthetic arm outputs in the formats of the frozen scripts
(raw 2.2 and native 2.1, with or without the wrapper's lines) and a guard 5 --idle-only receipt, and runs the census as a subprocess (exit status and receipt are what the protocol consumes).
usage: census-controls.py --work DIR --schema FILE --census FILE --raw-script FILE --native-script FILE
Run it as `uv run --with scipy==1.18.1 --with numpy==2.5.3 python census-controls.py ...` so that the matching cross-check against scipy's maximum_bipartite_matching runs (without scipy that line says NOT RUN).
Each scenario prints `CONTROL-RESULT <id> expected_exit=N observed_exit=M OK|UNEXPECTED` and the census's PROBLEM and CENSUS-RESULT lines (the reasons it gave). The scenarios, by the command center's required list:
  exact match (P1, P2), unique 2x2 (P3), replacement case (R1 distinguishable, R2 identical with a foreign call row, R3 identical with the foreign call row also dropped: the RESIDUAL, expected PASS), one row short (I1), one foreign call row (I2), the gateway's own provider-test rows (I3),
  two candidates (I4), complete 2x2 (I5), an orphan (I6), an unreadable store (U1-U4), and the guard, input, clock, tiling and format controls; X1-X3 check predicate parity with the frozen scripts (by AST, never by importing them: they run their main at import) and the matching functions
  against a brute force and scipy.
Added after the full-scope reads r10 and r11 of #757: Y11-Y13b (the reviewers' reproductions and the exact boundary check), D1-D8 and K3 (the canary gate's contract and the frozen request counts), G11-G12b (the guard receipt), W1-W3b (the declared interval in real time), P5, R3b, S1 (one read snapshot) and J1-J20 (the gateway's journal as the independent record of every inbound request: every world carries a journal fixture served by a `journalctl` shim first on the census's PATH; J12a-J20 and J13j reproduce the second independent review of the journal check: the counted range from 4 s before the window, every journal form that cannot be read, the pairing by path and its documented chain width, the leftovers of a maximum matching, an overflowing duration, the exact command in the receipt, journalctl's stderr and the escaping of labels).
Exit status 0 when every scenario behaved as expected."""

def utc(t):
    return t.strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def stamp(t):
    return t.strftime("%Y-%m-%dT%H:%M:%SZ")


def at(s):
    return dt.datetime.strptime(s, "%Y-%m-%dT%H:%M:%S.%fZ").replace(tzinfo=dt.timezone.utc)


T0 = at("2026-10-06T12:00:00.000Z")   # a fixed past date: the census waits for the declared interval to end on the host clock, so fixtures must lie in the past (the real-time controls W1-W3 shift their own)


def plus(seconds):
    return T0 + dt.timedelta(seconds=seconds)


def corr_id(tag):
    h = hashlib.sha256(tag.encode()).hexdigest()
    return f"{h[:8]}-{h[8:12]}-4{h[13:16]}-8{h[17:20]}-{h[20:32]}"


def resp_id(tag):
    return "resp_" + hashlib.sha256(("r" + tag).encode()).hexdigest()[:50]


# ---------------------------------------------------------------- the world: a spec that is materialized as a store, arm outputs and a guard receipt
def call(ts, tin, tout, duration, path="/v1/responses", ttft=500, status=200, model="gpt-6-astra", provider="codex", conn="conn-a", key="key-1"):
    return {"ts": ts, "tin": tin, "tout": tout, "duration": duration, "path": path, "ttft": ttft, "status": status, "model": model, "provider": provider, "conn": conn, "key": key}


def usage_for(c, latency=None, ttft=500.2, ts=None, tin=None, tout=None):
    """The usage row the gateway writes with a call row: same model, counts, provider, account and endpoint, latency 1 ms longer, time to first token with a fraction."""
    return {"ts": ts or c["ts"] - dt.timedelta(milliseconds=4), "tin": c["tin"] if tin is None else tin, "tout": c["tout"] if tout is None else tout, "latency": (c["duration"] + 1) if latency is None else latency, "ttft": ttft,
            "endpoint": c["path"], "model": c["model"], "provider": c["provider"], "conn": c["conn"], "key": c["key"], "status": "200"}


def default_spec(name):
    spec = {"name": name, "guard": {"start": plus(-35), "end": plus(0), "clean": True, "version": "5", "mode": "idle-only", "failed": "[]", "result": True}, "arms": [], "foreign_calls": [], "foreign_usage": [], "odd_calls": []}
    reqs = []
    for i, (mode, off, tin, tout, dur) in enumerate([("responses:eof", 20.0, 22, 21, 1000), ("chat:eof", 60.0, 23, 20, 1200), ("responses:early:0", 100.0, 22, 30, 900)], 1):
        tag = f"{name}-raw-{i}"
        path = "/v1/chat/completions" if mode.startswith("chat") else "/v1/responses"
        c = call(plus(off + 0.1), tin, tout, dur, path)
        reqs.append({"kind": "raw", "mode": mode, "corr": corr_id(tag), "rid": None if mode.startswith("chat") else resp_id(tag), "calls": [c], "usage": [usage_for(c)]})
    spec["arms"].append({"file": "arm-1-raw.txt", "kind": "raw", "version": "2.2", "sha": RAW_SHA, "wrapper_start": plus(5.0), "script_start": plus(5.3), "script_end": plus(175.2), "wrapper_end": plus(176.0),
                         "journal": "0", "summary": True, "run_status": 0, "requests": reqs, "db_override": None})
    nreqs = []
    c1 = call(plus(195.2), 15036, 5, 7000)
    nreqs.append({"kind": "native", "shape": "astra-max", "tag": f"acceptance-20261006T200300-astra-max-{name[:6]}-1", "min_rows": 1, "judged": True, "calls": [c1], "usage": [usage_for(c1)]})
    c2, c3 = call(plus(215.0), 15100, 40, 3000), call(plus(216.5), 15150, 37, 2500)
    nreqs.append({"kind": "native", "shape": "two-request", "tag": f"acceptance-20261006T200300-two-request-{name[:6]}-2", "min_rows": 2, "judged": True, "calls": [c2, c3], "usage": [usage_for(c2), usage_for(c3)]})
    spec["arms"].append({"file": "arm-2-native.txt", "kind": "native", "version": "2.1", "sha": NATIVE_SHA, "wrapper_start": plus(180.0), "script_start": plus(180.4), "script_end": plus(330.2), "wrapper_end": plus(331.0),
                         "journal": "0", "summary": True, "run_status": 0, "requests": nreqs, "db_override": None})
    return spec


def write_guard(path, g):
    L = [f"GUARD version={g['version']} mode={g['mode']} script_sha256={GUARD_SHA} start={utc(g['start'])} port={g.get('port', 20128)} unit={g.get('unit', 'omniroute.service')} clean_for={g.get('clean_for', 30.0)} interval=5.0 tolerate_units={g.get('tolerated', "['omniroute-fw.service']")}",
         *([] if g.get("no_identity") else ["   IDENTITY label=idle-only time=x unit=omniroute.service MainPID=1 ExecMainStartTimestamp=Tue_2026-10-06_09:51:19_EDT health=ok"]),
         "CHECK nothing in flight for 30 s (7 samples): " + ("ok (0 idle and 0 tolerated connection(s))" if g["clean"] else "FAIL ['an active connection']")]
    if g["result"]:
        if g["version"] == "4":
            L.append(f"GUARD-RESULT clean={g['clean']} failed_checks={g['failed']} end={utc(g['end'])}")
        else:
            L.append(f"IDLE-RESULT clean={g['clean']} failed_checks={g['failed']} end={utc(g['end'])}")
    path.write_text("\n".join(L) + "\n")


def write_arm(path, arm, db):
    dbp = arm["db_override"] or str(db)
    db_field = "" if arm.get("no_db") else f" db={dbp}"
    L = []
    wrapper = arm["wrapper_start"] is not None
    if wrapper:
        if not arm.get("no_header"):
            L.append(f"# trial arm ({arm['kind']}); started {stamp(arm['wrapper_start'])}; wrapper kind={arm.get('wrapper_kind', 'canary')}")
        if not arm.get("no_identity"):
            L.append(f"IDENTITY label=before-x time=x unit=omniroute.service MainPID={arm.get('pid', 1)} ExecMainStartTimestamp=Tue_2026-10-06_09:51:19_EDT health=ok")
    if arm["kind"] == "raw":
        L.append(f"FROZEN version={arm['version']} script_sha256={arm['sha']} argv=['--expect', 'all-logged'] cwd=/tmp start={utc(arm['script_start'])} expect=all-logged wait1=12.0 wait2=60.0 port=20128{db_field} effort_close_after=10.0 baseline=None baseline_sha256=None")
    else:
        L.append(f"FROZEN version={arm['version']} script_sha256={arm['sha']} argv=['--expect', 'all-logged'] cwd=/tmp start={utc(arm['script_start'])} expect=all-logged wait1=12.0 wait2=60.0 quiet_timeout=120.0 exec_timeout=300.0{db_field} run_id=20261006T200300 template_sha256={'7' * 64}")
    for r in arm["requests"]:
        n, m = len(r["calls"]), len(r["usage"])
        if r["kind"] == "raw":
            L.append(f"RESULT mode={r['mode']} http=200 terminal=True closed_early={str(r['mode'].startswith('responses:early')).title()} stream_s=3.4 corr={r['corr'] or 'None'} response_id={r['rid'] or 'None'} read1_call_logs={n} read1_usage={m} window_call_logs={n} window_usage={m} read1_usage_lookalikes=0 verdict=PASS")
        elif r["judged"]:
            L.append(f"RESULT shape={r['shape']} session_id={r['tag']} quiet=True recent_call_logs=0 other_workers=0 min_rows={r['min_rows']} exec_exit=0 elapsed_s=7.5 client_usage={{'input_tokens': 1, 'cached_input_tokens': 0, 'output_tokens': 1}} command_executions=1 events=5 read1_call_logs={n} read1_usage={m} read1_usage_unestablished=0 rows_tokens_in_out=(1, 1) rows_complete=True verdict=PASS")
        else:
            L.append(f"RESULT shape={r['shape']} session_id={r['tag']} quiet=True recent_call_logs=0 other_workers=0 min_rows={r['min_rows']} exec_exit=1 timed_out=False elapsed_s=2.0 client_usage=None command_executions=0 events=0 rows_with_session_id={n} verdict=INCONCLUSIVE reasons=['the exec did not complete']")
    if arm["summary"]:
        L.append(f"SUMMARY {{\"PASS\": {len(arm['requests'])}}} end={utc(arm['script_end'])} exit=0")
    if wrapper:
        L.append("exit=0")
        if not arm.get("no_identity"):
            L.append(f"IDENTITY label=after-x time=x unit=omniroute.service MainPID={arm.get('after_pid', arm.get('pid', 1))} ExecMainStartTimestamp=Tue_2026-10-06_09:51:19_EDT health=ok")
        if arm["journal"] != "missing":
            L.append(f"journal_failed_to_save_call_log={arm['journal']} (journalctl --user -u omniroute.service --since <the unit's ExecMainStartTimestamp> -o cat --no-pager | grep -c 'Failed to save call log')")
        L.append(f"RUN-STATUS {Path(path).name} status={arm['run_status']} ({'VOID' if arm['run_status'] == 4 else 'ok'})")
        L.append(f"# finished {stamp(arm['wrapper_end'])}")
    elif arm["journal"] != "missing":
        L.append(f"journal_failed_to_save_call_log={arm['journal']} (direct run)")
    path.write_text("\n".join(L) + "\n")


def materialize(spec, work):
    work = Path(work)
    db = work / f"{spec['name']}.sqlite"
    if db.exists():
        db.unlink()
    con = sqlite3.connect(db)
    con.executescript(Path(ARGS.schema).read_text())
    n = [0]

    def put_call(c, corr=None, rid=None, session=None):
        n[0] += 1
        con.execute("insert into call_logs (id, timestamp, path, status, model, requested_model, provider, connection_id, api_key_id, duration, ttft_ms, tokens_in, tokens_out, correlation_id, response_id, session_tag) values (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
                    (f"call-{spec['name']}-{n[0]}", c["ts"] if isinstance(c["ts"], (str, bytes)) else utc(c["ts"]), c["path"], c["status"], c["model"], ("codex/" + c["model"] if c["model"] else None), c["provider"], c["conn"], c["key"], c["duration"], c["ttft"], c["tin"], c["tout"], corr, rid, session))

    def put_usage(u):
        con.execute("insert into usage_history (provider, model, connection_id, api_key_id, tokens_input, tokens_output, status, latency_ms, ttft_ms, timestamp, endpoint) values (?,?,?,?,?,?,?,?,?,?,?)",
                    (u["provider"], u["model"], u["conn"], u["key"], u["tin"], u["tout"], u["status"], u["latency"], u["ttft"], utc(u["ts"]), u["endpoint"]))

    for arm in spec["arms"]:
        for r in arm["requests"]:
            for c in r["calls"]:
                put_call(c, corr=r.get("corr"), rid=r.get("rid"), session=r.get("tag"))
            for u in r["usage"]:
                put_usage(u)
    for c in spec["foreign_calls"]:
        put_call(c, corr=corr_id("foreign-" + str(c["ts"])), rid=None)
    for c in spec["odd_calls"]:
        put_call(c)
    for u in spec["foreign_usage"]:
        put_usage(u)
    con.commit()
    con.close()
    # ROUND-15: the gateway journal fixture: one inbound-request line per own and per foreign inference call row (the gateway logs it when the request ARRIVES, about `duration` before its row is written), plus noise
    journal = []

    def post_line(ts, label="/v1/responses | cx/gpt-6-astra | 14 msgs | effort=max"):
        journal.append(json.dumps({"__REALTIME_TIMESTAMP": str(int(ts.timestamp() * 1e6)), "MESSAGE": json.dumps({"level": 30, "time": utc(ts), "service": "omniroute", "module": "sse", "tag": "HTTP", "method": "POST", "path": label, "msg": "\U0001F4E5 POST " + label})}))

    for a_index, arm in enumerate(spec["arms"]):
        for r_index, r in enumerate(arm["requests"]):
            for c_index, c in enumerate(r["calls"]):
                if isinstance(c["ts"], (str, bytes)) or (a_index, r_index, c_index) in spec.get("journal_drop", ()):
                    continue
                try:
                    arrival = c["ts"] - dt.timedelta(milliseconds=float(c["duration"]))
                except (TypeError, ValueError, OverflowError):
                    arrival = arm["script_start"]   # a duration that is not a number or not representable (controls about exactly that): the arrival is taken at the arm's start
                post_line(max(arrival, arm["script_start"]), f"{c['path']} | cx/{c['model']} | 14 msgs")
    for c in spec["foreign_calls"]:
        if not isinstance(c["ts"], (str, bytes)) and not str(c["path"]).startswith("/api/"):   # the gateway's own provider-test calls are not inbound requests
            post_line(c["ts"] - dt.timedelta(milliseconds=c["duration"] if isinstance(c["duration"], (int, float)) else 0), f"{c['path']} | cx/{c['model']} | 14 msgs")
    for item in spec.get("journal_extra", ()):   # ROUND-17: a datetime, or (datetime, path)
        if isinstance(item, tuple):
            post_line(item[0], f"{item[1]} | cx/gpt-6-astra | 14 msgs")
        else:
            post_line(item)
    base = plus(-5)
    journal += [json.dumps({"__REALTIME_TIMESTAMP": str(int(base.timestamp() * 1e6)), "MESSAGE": "[ProxyEgress] codex status=success"}),
                json.dumps({"__REALTIME_TIMESTAMP": str(int(base.timestamp() * 1e6)), "MESSAGE": json.dumps({"level": 30, "time": utc(base), "tag": "AUTH", "msg": "Using codex account: ***..."})}),
                json.dumps({"__REALTIME_TIMESTAMP": str(int(base.timestamp() * 1e6)), "MESSAGE": json.dumps({"level": 30, "time": utc(base), "tag": "HTTP", "method": "GET", "msg": "\U0001F4E5 GET /v1/models"})}),
                json.dumps({"__REALTIME_TIMESTAMP": str(int(base.timestamp() * 1e6)), "MESSAGE": [27, 91, 51, 50, 109, 91, 85, 83, 65, 71, 69, 93, 27, 91, 48, 109]})]
    journal += list(spec.get("journal_noise", ()))
    (Path(work) / f"{spec['name']}-journal.jsonl").write_text("\n".join(journal) + "\n")
    if spec.get("journal_raw"):   # raw bytes after the text lines (a journal line that is not valid UTF-8)
        with open(Path(work) / f"{spec['name']}-journal.jsonl", "ab") as handle:
            handle.write(spec["journal_raw"])
    (Path(work) / f"{spec['name']}-journal.rc").write_text(str(spec.get("journal_rc", 0)))
    (Path(work) / f"{spec['name']}-journal.err").write_text(spec.get("journal_stderr", ""))
    argv_log = Path(work) / f"{spec['name']}-journal.argv"
    if argv_log.exists():
        argv_log.unlink()
    files = []
    for arm in spec["arms"]:
        p = work / f"{spec['name']}-{arm['file']}"
        write_arm(p, arm, db)
        files.append(p)
    guard = work / f"{spec['name']}-guard.txt"
    write_guard(guard, spec["guard"])
    return db, files, guard


def journal_env(guard):
    """The environment of every census run (ROUND-15): a `journalctl` shim first on PATH that serves the world's journal fixture (derived from the guard receipt's name) and its exit status."""
    shim = Path(ARGS.work) / "journal-shim"
    shim.mkdir(exist_ok=True)
    (shim / "journalctl").write_text('#!/bin/sh\n[ -n "$JOURNAL_ARGV_LOG" ] && printf \'%s\\n\' "$*" >> "$JOURNAL_ARGV_LOG"\ncase "$1" in --version) echo "systemd 255 (255.4-1ubuntu8.17)"; exit 0;; esac\n[ -n "$JOURNAL_STDERR_FILE" ] && [ -s "$JOURNAL_STDERR_FILE" ] && cat "$JOURNAL_STDERR_FILE" >&2\n[ -n "$JOURNAL_FIXTURE" ] && [ -f "$JOURNAL_FIXTURE" ] && cat "$JOURNAL_FIXTURE"\nexit "${JOURNAL_RC:-0}"\n')
    (shim / "journalctl").chmod(0o755)
    fixture = Path(str(guard).replace("-guard.txt", "-journal.jsonl"))
    rc_file = Path(str(guard).replace("-guard.txt", "-journal.rc"))
    return dict(os.environ, PATH=f"{shim}:{os.environ['PATH']}", JOURNAL_FIXTURE=str(fixture), JOURNAL_RC=rc_file.read_text().strip() if rc_file.exists() else "0",
                JOURNAL_STDERR_FILE=str(Path(str(guard).replace("-guard.txt", "-journal.err"))), JOURNAL_ARGV_LOG=str(Path(str(guard).replace("-guard.txt", "-journal.argv"))))


def run_census(db, files, guard, diagnostic=False, no_guard_arg=False, extra=None):
    cmd = [sys.executable, ARGS.census]
    if diagnostic:
        cmd.append("--diagnostic")
    elif not no_guard_arg:
        cmd += ["--guard-receipt", str(guard)]
    extra = list(extra or [])
    if "@NO_REQUIRE_ARMS@" in extra:
        extra.remove("@NO_REQUIRE_ARMS@")   # run with the census's own default
    elif "--require-arms" not in extra:
        extra += ["--require-arms", str(len(files))]   # the fixtures have two arms
    cmd += ["--db", str(db)] + [str(guard) if x == "@GUARD@" else x for x in extra] + [str(f) for f in files]
    p = subprocess.run(cmd, capture_output=True, text=True, env=journal_env(guard))
    return p.returncode, p.stdout, p.stderr


def check(sid, desc, rc, out, err, want, must=(), mustnot=()):
    text = out + err
    missing = [m for m in must if m not in text]
    present = [m for m in mustnot if m in text]
    ok = rc == want and not missing and not present
    print(f"CONTROL-RESULT {sid} expected_exit={want} observed_exit={rc} {'OK' if ok else 'UNEXPECTED'} :: {desc}")
    home = str(Path.home())
    for line in out.splitlines():
        if line.startswith("PROBLEM ") or line.startswith("CENSUS-RESULT"):
            print("    " + line[:260].replace(home, "~"))
    for line in err.splitlines()[:3]:
        print("    stderr: " + line[:200].replace(home, "~"))
    if not ok:
        FAILED.append(sid)
        print(f"    expected text missing: {missing}; unexpected text present: {present}")
    print(flush=True)
    return ok


def scenario(sid, desc, mutate=None, want=2, must=(), mustnot=(), diagnostic=False, no_guard_arg=False, db_arg=None, extra=None, post=None):
    spec = default_spec(sid.lower())
    if mutate:
        mutate(spec)
    if db_arg is not None:
        for arm in spec["arms"]:
            arm["db_override"] = str(db_arg)   # the scripts read the store under test: the unreadable store is the only fault of these controls
    db, files, guard = materialize(spec, ARGS.work)
    if post:
        post(db, files, guard)
    rc, out, err = run_census(db_arg or db, files, guard, diagnostic=diagnostic, no_guard_arg=no_guard_arg, extra=extra)
    return check(sid, desc, rc, out, err, want, must, mustnot)


# ---------------------------------------------------------------- mutations
def drop_usage(spec, arm, req):
    spec["arms"][arm]["requests"][req]["usage"] = []


def add_foreign(spec, ts=120.0, tin=40, tout=7, dur=800, path="/v1/responses", with_usage=True, model="gpt-6-astra"):
    c = call(plus(ts), tin, tout, dur, path, model=model)
    spec["foreign_calls"].append(c)
    if with_usage:
        spec["foreign_usage"].append(usage_for(c))

    try:

    except ImportError:
        scipy_line = "NOT AVAILABLE (the scipy cross-check X3 will say NOT RUN)"
    print(f"# Offline controls of the window census gate, version 1 (new, not frozen); python {sys.version.split()[0]}; {scipy_line}")
    print(f"# census: sha256 {hashlib.sha256(census_bytes).hexdigest()}")
    print(f"# this controls script: sha256 {hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}")

    # ---- PASS controls
    def p1_post(db, files, guard):
        p1_post.before = hashlib.sha256(db.read_bytes()).hexdigest()

    # ---- X3: the matching functions against a brute force (all permutations) and, when scipy is available, scipy's maximum_bipartite_matching plus its edge-removal uniqueness test
    try:
        import numpy as np
        from scipy.sparse import csr_matrix
        from scipy.sparse.csgraph import maximum_bipartite_matching
        have_scipy = True
    except ImportError:
        have_scipy = False

    def brute(nc, nu, edge_set):
        if nc != nu:
            return "unequal"
        count = 0
        for perm in itertools.permutations(range(nu)):
            if all((i, perm[i]) in edge_set for i in range(nc)):
                count += 1
                if count > 1:
                    break
        return {0: "none", 1: "unique"}.get(count, "multiple")

    def scipy_status(nc, nu, edge_set):
        if nc != nu:
            return "unequal"

        def matrix(edges):
            return csr_matrix((np.ones(len(edges), dtype=np.int8), ([i for i, _ in edges], [j for _, j in edges])), shape=(nc, nu))

        def size(edges):
            return int((maximum_bipartite_matching(matrix(edges), perm_type="column") >= 0).sum()) if edges else 0
        if size(sorted(edge_set)) < nc:
            return "none"
        m = maximum_bipartite_matching(matrix(sorted(edge_set)), perm_type="column")
        for i, j in [(i, int(j)) for i, j in enumerate(m)]:
            if size(sorted(edge_set - {(i, j)})) == nc:
                return "multiple"
        return "unique"
