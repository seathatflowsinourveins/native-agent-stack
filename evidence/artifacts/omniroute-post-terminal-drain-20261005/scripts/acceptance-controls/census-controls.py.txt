def utc(t):
    return t.strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def stamp(t):
    return t.strftime("%Y-%m-%dT%H:%M:%SZ")


def at(s):
    return dt.datetime.strptime(s, "%Y-%m-%dT%H:%M:%S.%fZ").replace(tzinfo=dt.timezone.utc)


T0 = at("2026-10-06T12:00:00.000Z")   # a fixed past date: the census waits for the declared interval to end on the host clock, so fixtures must lie in the past (the real-time controls W1-W3 shift their own)


def plus(seconds):
    return T0 + dt.timedelta(seconds=seconds)


def corr_id(tag):
    h = hashlib.sha256(tag.encode()).hexdigest()
    return f"{h[:8]}-{h[8:12]}-4{h[13:16]}-8{h[17:20]}-{h[20:32]}"


def resp_id(tag):
    return "resp_" + hashlib.sha256(("r" + tag).encode()).hexdigest()[:50]

def augment_journal(fixture, world, root):
    """ROUND-25: the fields journald adds to every entry, from the fake /proc's boot clock: `_BOOT_ID` and `__MONOTONIC_TIMESTAMP` (microseconds on the monotonic clock of the entry's stamp AS THE FIXTURE WRITES IT: a world that encodes a step by moving realtime stamps, as J44 and the property fault `clock_step` do, moves the monotonic stamp with it, and
    census r26 refuses such an entry as dated before its process's birth; only `clock_step` models a physical step). A world's `clock_step` {"age": a, "s": S} (or a list of them) then moves the realtime stamp and the gateway's own `time` of every entry stamped before the moment `a` seconds after the earliest listener's start S seconds earlier (the wall clock stepped
    forward by S at that moment: those lines were stamped before the step, the monotonic stamps are not moved). Entries that are not JSON objects with a realtime stamp, and the raw bytes of `journal_raw`, are left as they are."""
    frame = json.loads((root / ".frame.json").read_text())
    boot_wall = frame["now"] - frame["uptime"]
    starts = {str(pid): s.timestamp() for pid, s in world_starts(world).items()}
    steps = world.get("clock_step") or []
    steps = [steps] if isinstance(steps, dict) else list(steps)
    step_walls = [(min(starts.values()) + st["age"], st["s"]) for st in steps]   # the moment of each step (in the world's true time: the earliest listener's start plus its age) and its size
    out = []
    for raw in Path(fixture).read_bytes().split(b"\n"):
        try:
            entry = json.loads(raw)
            rt = int(entry["__REALTIME_TIMESTAMP"])
        except (ValueError, KeyError, TypeError):
            out.append(raw)
            continue
        if not isinstance(entry, dict):
            out.append(raw)
            continue
        wall = rt / 1e6
        entry.setdefault("_BOOT_ID", FIXTURE_BOOT_HEX)
        entry.setdefault("__MONOTONIC_TIMESTAMP", str(int((wall - boot_wall) * 1e6)))
        shift = sum(s_ for at_, s_ in step_walls if wall < at_)   # the seconds by which the wall clock stepped forward AFTER this entry was stamped: it was stamped that much earlier

    path.write_bytes(b"\n".join(out))

# ---------------------------------------------------------------- the world: a spec that is materialized as a store, arm outputs and a guard receipt
def call(ts, tin, tout, duration, path="/v1/responses", ttft=500, status=200, model="gpt-6-astra", provider="codex", conn="conn-a", key="key-1"):
    return {"ts": ts, "tin": tin, "tout": tout, "duration": duration, "path": path, "ttft": ttft, "status": status, "model": model, "provider": provider, "conn": conn, "key": key}


def usage_for(c, latency=None, ttft=500.2, ts=None, tin=None, tout=None):
    """The usage row the gateway writes with a call row: same model, counts, provider, account and endpoint, latency 1 ms longer, time to first token with a fraction."""
    return {"ts": ts or c["ts"] - dt.timedelta(milliseconds=4), "tin": c["tin"] if tin is None else tin, "tout": c["tout"] if tout is None else tout, "latency": (c["duration"] + 1) if latency is None else latency, "ttft": ttft,
            "endpoint": c["path"], "model": c["model"], "provider": c["provider"], "conn": c["conn"], "key": c["key"], "status": "200"}

def materialize(spec, work):
    work = Path(work)
    db = work / f"{spec['name']}.sqlite"
    if db.exists():
        db.unlink()
    con = sqlite3.connect(db)
    con.executescript(Path(ARGS.schema).read_text())
    n = [0]

    def put_call(c, corr=None, rid=None, session=None):
        n[0] += 1
        con.execute("insert into call_logs (id, timestamp, path, status, model, requested_model, provider, connection_id, api_key_id, duration, ttft_ms, tokens_in, tokens_out, correlation_id, response_id, session_tag) values (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
                    (f"call-{spec['name']}-{n[0]}", c["ts"] if isinstance(c["ts"], (str, bytes)) else utc(c["ts"]), c["path"], c["status"], c["model"], ("codex/" + c["model"] if c["model"] else None), c["provider"], c["conn"], c["key"], c["duration"], c["ttft"], c["tin"], c["tout"], corr, rid, session))

    def put_usage(u):
        con.execute("insert into usage_history (provider, model, connection_id, api_key_id, tokens_input, tokens_output, status, latency_ms, ttft_ms, timestamp, endpoint) values (?,?,?,?,?,?,?,?,?,?,?)",
                    (u["provider"], u["model"], u["conn"], u["key"], u["tin"], u["tout"], u["status"], u["latency"], u["ttft"], utc(u["ts"]), u["endpoint"]))

    for arm in spec["arms"]:
        for r in arm["requests"]:
            for c in r["calls"]:
                put_call(c, corr=(r.get("corr") if r["kind"] == "raw" else corr_id(f"{spec['name']}-native-{n[0] + 1}")), rid=r.get("rid"), session=r.get("tag"))   # ROUND-25: every native row has its own id, as in the live store
            for u in r["usage"]:
                put_usage(u)
    for c in spec["foreign_calls"]:
        put_call(c, corr=(None if c.get("corr_none") else (c.get("corr") or corr_id(f"foreign-{n[0] + 1}-{c['ts']}"))), rid=None)   # ROUND-25: one id per row unless the world gives the rows of one request the same (or none: `corr_none`)
    for c in spec["odd_calls"]:
        put_call(c)
    for w in spec.get("ws_turns", ()):   # ROUND-20: a Responses-over-WebSocket turn as the gateway's proxy writes it (src/app/api/internal/codex-responses-ws/history.ts): method WEBSOCKET, request_type responses_websocket, stamped at the turn's START, its length as the duration; no journal line

def journal_env(guard):
    """The environment of every census run (ROUND-15): a `journalctl` shim first on PATH that serves the world's journal fixture (derived from the guard receipt's name) and its exit status."""
    shim = Path(ARGS.work) / "journal-shim"
    shim.mkdir(exist_ok=True)
    (shim / "journalctl").write_text('#!/bin/sh\n[ -n "$JOURNAL_ARGV_LOG" ] && printf \'%s\\n\' "$*" >> "$JOURNAL_ARGV_LOG"\ncase "$1" in --version) echo "systemd 255 (255.4-1ubuntu8.17)"; exit 0;; esac\n[ -n "$JOURNAL_STDERR_FILE" ] && [ -s "$JOURNAL_STDERR_FILE" ] && cat "$JOURNAL_STDERR_FILE" >&2\n[ -n "$JOURNAL_HONOUR" ] && [ -n "$JOURNAL_FIXTURE" ] && [ -f "$JOURNAL_FIXTURE" ] && exec python3 "$JOURNAL_FILTER" "$@"\n[ -n "$JOURNAL_FIXTURE" ] && [ -f "$JOURNAL_FIXTURE" ] && cat "$JOURNAL_FIXTURE"\nexit "${JOURNAL_RC:-0}"\n')
    (shim / "journal-filter.py").write_text(FILTER_SOURCE)   # ROUND-25: used by the worlds with `honour_bounds`
    (shim / "journalctl").chmod(0o755)
    fixture = Path(str(guard).replace("-guard.txt", "-journal.jsonl"))
    rc_file = Path(str(guard).replace("-guard.txt", "-journal.rc"))
    world_file = Path(str(guard).replace("-guard.txt", "-world.json"))
    try:
        world = json.loads(world_file.read_text()) if world_file.exists() else {}
    except (ValueError, OSError):   # a harness that renamed or truncated the world's files (the corpus of review 1 does): no fixture, the census then fails closed
        world = {}
    name = Path(guard).name.replace("-guard.txt", "")
    (shim / "ss").write_text('#!/bin/sh\n[ -n "$SS_ARGV_LOG" ] && printf \'%s\\n\' "$*" >> "$SS_ARGV_LOG"\n[ -n "$SS_FIXTURE" ] && [ -f "$SS_FIXTURE" ] && cat "$SS_FIXTURE"\nexit "${SS_RC:-0}"\n')
    (shim / "systemctl").write_text('#!/bin/sh\ncase "$1 $2" in "--user show") [ -f "$SYSTEMCTL_SHOW" ] && cat "$SYSTEMCTL_SHOW"; exit "${SYSTEMCTL_RC:-0}";; esac\nexit 1\n')
    (shim / "systemd-analyze").write_text('#!/bin/sh\n[ -f "$ANALYZE_OUT" ] && cat "$ANALYZE_OUT"\nexit "${ANALYZE_RC:-0}"\n')
    for tool in ("ss", "systemctl", "systemd-analyze"):
        (shim / tool).chmod(0o755)
    listeners = world.get("listeners", [1326])
    ss_file = Path(ARGS.work) / f"{name}-ss.out"
    ss_file.write_text(world["ss_text"] if world.get("ss_text") is not None else "".join(f'LISTEN 0      511    127.0.0.1:20128 0.0.0.0:* users:(("omniroute (v16.",pid={pid},fd=21))\n' for pid in listeners))
    root = build_proc_root(name, world) if world else Path("/nonexistent-proc-root")
    props = {"StandardOutput": "journal", "StandardError": "inherit", "LogRateLimitIntervalUSec": "0", "LogRateLimitBurst": "0"}
    if world:   # ROUND-25: the service manager's monotonic start stamp of the unit's main process (2 s before the earliest listener's boot-clock start) and its pid, and the journal fixture with the fields journald adds
        frame = json.loads((root / ".frame.json").read_text())
        unit_mono = min(s.timestamp() for s in world_starts(world).values()) + world.get("unit_stamp_offset", -2.0) - (frame["now"] - frame["uptime"])
        props.update({"MainPID": "341", "ExecMainStartTimestampMonotonic": str(int(unit_mono * 1e6))})
        if fixture.exists():
            fixture = augment_journal(fixture, world, root)
    props.update(world.get("unit_props", {}))
    show_file = Path(ARGS.work) / f"{name}-systemctl.out"
    show_file.write_text("".join(f"{k}={v}\n" for k, v in props.items()))
    analyze_file = Path(ARGS.work) / f"{name}-analyze.out"
    analyze_file.write_text("# /etc/systemd/journald.conf\n[Journal]\nForwardToSyslog=yes\n" + world.get("journald_conf", ""))
    return dict(os.environ, PATH=f"{shim}:{os.environ['PATH']}", JOURNAL_FIXTURE=str(fixture), JOURNAL_HONOUR=("1" if world.get("honour_bounds") else ""), JOURNAL_FILTER=str(shim / "journal-filter.py"), JOURNAL_RC=rc_file.read_text().strip() if rc_file.exists() else "0",

def build_proc_root(name, world):
    """ROUND-19: a fake /proc for one world, written at run time (the census reads the host clock): /proc/uptime, the stat (field 22: starttime in clock ticks since boot), cgroup and environ of each listener, and a file descriptor onto the store for each holder. The listeners started `listener_start_offset` seconds
    after the guard receipt began (negative: before), or at their own `start_offsets`."""
    root = Path(ARGS.work) / f"{name}-proc"
    shutil.rmtree(root, ignore_errors=True)
    root.mkdir(parents=True)
    now = dt.datetime.now(dt.timezone.utc)
    starts = world_starts(world)
    uptime = max([(now - s).total_seconds() for s in starts.values()] + [0.0]) + 1000.0
    (root / "uptime").write_text(f"{uptime:.2f} 0.00\n")
    (root / "sys" / "kernel" / "random").mkdir(parents=True)   # ROUND-25: the boot id (sys/kernel/random/boot_id) and the frame (the host clock and the uptime at run time) that the journal entries' monotonic stamps are made from
    if "boot_id" not in world.get("proc_broken", []):
        (root / "sys" / "kernel" / "random" / "boot_id").write_text(world.get("boot_id", FIXTURE_BOOT_ID) + "\n")
    (root / ".frame.json").write_text(json.dumps({"now": now.timestamp(), "uptime": uptime}))
    for pid, started in starts.items():
        d = root / str(pid)
        d.mkdir(exist_ok=True)
        ticks = int((uptime - (now - started).total_seconds()) * os.sysconf("SC_CLK_TCK"))
        if "stat" not in world.get("proc_broken", []):
            (d / "stat").write_text(f"{pid} (omniroute (v16.) S 341 {pid} {pid} 0 -1 4194560 0 0 0 0 0 0 0 0 20 0 20 0 {ticks} 0 0\n")
        (d / "cgroup").write_text(f"0::{world.get('cgroups', {}).get(str(pid), CGROUP)}\n")
        env = "PORT=20128\0NODE_ENV=production\0" + (f"STREAM_DISCONNECT_GRACE_PERIOD_MS={world['grace_env']}\0" if world.get("grace_env") is not None else "")
        (d / "environ").write_bytes(env.encode())
    for pid in world.get("holders", [1326]):
        d = root / str(pid) / "fd"
        d.mkdir(parents=True, exist_ok=True)
        (d / "3").symlink_to(world["db"])
    return root


def run_census(db, files, guard, diagnostic=False, no_guard_arg=False, extra=None, real_host=False, shims_only=False):
    cmd = [sys.executable, ARGS.census]
    if diagnostic:
        cmd.append("--diagnostic")
    elif not no_guard_arg:
        cmd += ["--guard-receipt", str(guard)]
    extra = list(extra or [])
    if "@NO_REQUIRE_ARMS@" in extra:
        extra.remove("@NO_REQUIRE_ARMS@")   # run with the census's own default
    elif "--require-arms" not in extra:
        extra += ["--require-arms", str(len(files))]   # the fixtures have two arms
    cmd += ["--db", str(db)] + [str(guard) if x == "@GUARD@" else x for x in extra] + [str(f) for f in files]
    env = {k: v for k, v in os.environ.items() if k != "CENSUS_PROC_ROOT"} if real_host else ({k: v for k, v in journal_env(guard).items() if k != "CENSUS_PROC_ROOT"} if shims_only else journal_env(guard))   # ROUND-23: shims_only keeps the shims and the fixture journal but not the fixture proc root (the host's own /proc)
       # ROUND-21: real_host runs the census on the host's own /proc, journalctl, ss and systemctl (no fixture root, no shims)
    p = subprocess.run(cmd, capture_output=True, text=True, env=env)
    return p.returncode, p.stdout, p.stderr


def check(sid, desc, rc, out, err, want, must=(), mustnot=()):
    text = out + err
    missing = [m for m in must if m not in text]
    present = [m for m in mustnot if m in text]
    ok = rc == want and not missing and not present
    print(f"CONTROL-RESULT {sid} expected_exit={want} observed_exit={rc} {'OK' if ok else 'UNEXPECTED'} :: {desc}")
    home = str(Path.home())
    for line in out.splitlines():
        if line.startswith("PROBLEM ") or line.startswith("CENSUS-RESULT"):
            print("    " + line[:260].replace(home, "~"))
    for line in err.splitlines()[:3]:
        print("    stderr: " + line[:200].replace(home, "~"))
    if not ok:
        FAILED.append(sid)
        print(f"    expected text missing: {missing}; unexpected text present: {present}")
    print(flush=True)
    return ok


def scenario(sid, desc, mutate=None, want=2, must=(), mustnot=(), diagnostic=False, no_guard_arg=False, db_arg=None, extra=None, post=None, real_host=False, shims_only=False):
    spec = default_spec(sid.lower())
    if mutate:
        mutate(spec)
    if db_arg is not None:
        for arm in spec["arms"]:
            arm["db_override"] = str(db_arg)   # the scripts read the store under test: the unreadable store is the only fault of these controls
    db, files, guard = materialize(spec, ARGS.work)
    if post:
        post(db, files, guard)
    rc, out, err = run_census(db_arg or db, files, guard, diagnostic=diagnostic, no_guard_arg=no_guard_arg, extra=extra, real_host=real_host, shims_only=shims_only)
    return check(sid, desc, rc, out, err, want, must, mustnot)


# ---------------------------------------------------------------- the canary's four arms (module level since round 21: the property test builds them too)
def four_arms(spec, last_kind="native", wrapper_kind="canary"):
    """The canary's four arms with the frozen request counts (17 raw requests, 6 race requests, 4 native one-request turns, 8 native tool-call turns of two call rows each): every request has its own row(s) and distinguishable numbers (ROUND-14)."""
    n = spec["name"]
    counter = itertools.count(1)

    def raw_request(arm_no, i, mode, off):
        k = next(counter)
        tag = f"{n}-a{arm_no}-{i}"
        path_ = "/v1/chat/completions" if mode.startswith("chat") else "/v1/responses"
        c = call(plus(off + 0.1), 100 + k, 20 + k % 7, 900 + 13 * k, path_)
        return {"kind": "raw", "mode": mode, "corr": corr_id(tag), "rid": None if mode.startswith("chat") else resp_id(tag), "calls": [c], "usage": [usage_for(c)]}

    def native_request(shape, i, off, rows):
        k = next(counter)
        calls = [call(plus(off + 1.5 * j), 15000 + 10 * k + j, 30 + k % 9 + j, 2500 + 17 * k + 11 * j) for j in range(rows)]
        return {"kind": "native", "shape": shape, "tag": f"acceptance-20261006T120300-{shape}-{n[:6]}-{i}", "min_rows": rows, "judged": True, "calls": calls, "usage": [usage_for(c) for c in calls]}

    def make_arm(file, kind, version, sha, begin, finish, reqs):
        return {"file": file, "kind": kind, "version": version, "sha": sha, "wrapper_start": plus(begin), "script_start": plus(begin + 0.3), "script_end": plus(finish), "wrapper_end": plus(finish + 0.8),
                "journal": "0", "summary": True, "run_status": 0, "requests": reqs, "db_override": None, "wrapper_kind": wrapper_kind}

    modes = ["effort:cx/gpt-6-astra:max"] * 3 + ["effort:cx/gpt-6.1-sol-max"] * 3 + ["effort:cx/gpt-6.1-sol:max"] * 3 + ["responses:early:0"] * 2 + ["chat:early:0"] * 2 + ["responses:eof", "chat:eof", "responses:early:2", "responses:early:0.5"]
    raw = [raw_request(1, i, modes[i], 10 + 8 * i) for i in range(17)]
    race = [raw_request(2, i, "responses:early:0", 160 + 10 * i) for i in range(6)]
    nat1 = [native_request("astra-max", i, 230 + 12 * i, 1) for i in range(4)]
    if last_kind == "raw":   # a wrong shape: the fourth arm is a raw arm
        nat2 = [raw_request(4, i, "responses:eof", 300 + 12 * i) for i in range(8)]
    else:
        nat2 = [native_request("two-request", i, 300 + 12 * i, 2) for i in range(8)]
    spec["arms"] = [make_arm("arm-1-raw.txt", "raw", "2.2", RAW_SHA, 5.0, 150.2, raw), make_arm("arm-2-race.txt", "raw", "2.2", RAW_SHA, 155.0, 220.2, race),
                    make_arm("arm-3-native.txt", "native", "2.1", NATIVE_SHA, 225.0, 290.2, nat1),
                    make_arm("arm-4-last.txt", last_kind, "2.1" if last_kind == "native" else "2.2", NATIVE_SHA if last_kind == "native" else RAW_SHA, 295.0, 400.2, nat2)]

    # ---- X6 (ROUND-21, review r13 P1): the incarnation's matching of inbound request lines to call rows (incarnation_unmatched) against a brute force and, when scipy is available, scipy's maximum_bipartite_matching on random instances.
    # The rule: a line (time, path) may take any call row of its path whose estimated start is not more than tau before the line, whatever the line's lead over that clock; each row serves one line; the leftover lines are the count that no maximum matching can serve.
    try:

        from scipy.sparse import csr_matrix
        from scipy.sparse.csgraph import maximum_bipartite_matching
        have_scipy6 = True

    def brute6(edges, n_lines, n_rows):
        """the size of a maximum matching by exhaustive search over the rows each line can take (lines in order, a bitmask of the rows used)"""
        best = [0]

        def go(i, used, served):
            if served + (n_lines - i) <= best[0]:
                return
            if i == n_lines:
                best[0] = max(best[0], served)
                return
            go(i + 1, used, served)   # this line left without a row
            for j in edges[i]:
                if not used & (1 << j):
                    go(i + 1, used | (1 << j), served + 1)
        go(0, 0, 0)
        return best[0]

    def scipy6(edges, n_lines, n_rows):
        pairs = sorted((i, j) for i in range(n_lines) for j in edges[i])
        if not pairs:
            return 0
        matrix = csr_matrix((np.ones(len(pairs), dtype=np.int8), ([i for i, _ in pairs], [j for _, j in pairs])), shape=(n_lines, n_rows))
        return int((maximum_bipartite_matching(matrix, perm_type="column") >= 0).sum())

    # ---- X3: the matching functions against a brute force (all permutations) and, when scipy is available, scipy's maximum_bipartite_matching plus its edge-removal uniqueness test
    try:
        import numpy as np
        from scipy.sparse import csr_matrix
        from scipy.sparse.csgraph import maximum_bipartite_matching
        have_scipy = True
    except ImportError:
        have_scipy = False

    def brute(nc, nu, edge_set):
        if nc != nu:
            return "unequal"
        count = 0
        for perm in itertools.permutations(range(nu)):
            if all((i, perm[i]) in edge_set for i in range(nc)):
                count += 1
                if count > 1:
                    break
        return {0: "none", 1: "unique"}.get(count, "multiple")

    def scipy_status(nc, nu, edge_set):
        if nc != nu:
            return "unequal"

        def matrix(edges):
            return csr_matrix((np.ones(len(edges), dtype=np.int8), ([i for i, _ in edges], [j for _, j in edges])), shape=(nc, nu))

        def size(edges):
            return int((maximum_bipartite_matching(matrix(edges), perm_type="column") >= 0).sum()) if edges else 0
        if size(sorted(edge_set)) < nc:
            return "none"
        m = maximum_bipartite_matching(matrix(sorted(edge_set)), perm_type="column")
        for i, j in [(i, int(j)) for i, j in enumerate(m)]:
            if size(sorted(edge_set - {(i, j)})) == nc:
                return "multiple"
        return "unique"

    rng = random.Random(757)

        elif roll < 0.55 and nc == nu and nc >= 3:   # a Hall violation inside one component: two calls whose only neighbour is usage row 0, a third call connecting everything
