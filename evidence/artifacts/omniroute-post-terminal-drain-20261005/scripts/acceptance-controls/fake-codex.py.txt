#!/usr/bin/env python3
"""A fake `codex` for the offline controls of the native acceptance script: a TEST FIXTURE that writes the rows a gateway would write, directly into a scratch database, and prints the JSONL events of
`codex exec --json` that a real client prints, for shapes the real client cannot produce against the stub (a tool-call turn with two requests) or on purpose (a failing exec, a hang). It reads the same command line
the real client gets: `-m <model>`, and the per-turn session id from `-c model_providers.omniroute.http_headers={"X-OmniRoute-Session-Id"="<id>"}`.

FAKE_CODEX_MODE:
  two-ok            two requests (15045/50 and 15106/24 tokens), both rows (call_logs + usage_history) written with the turn's session id, one command execution, turn usage = the sum, exit 0
  two-final-lost    the same turn and the same client usage, but only the first request's rows are written (the final request's rows are lost)
  two-notool        a turn with NO command execution and one request (the model skipped the tool call): one row, client usage = that row
  two-nousage       two-ok's rows and events but no turn.completed event (the client reported no usage)
  exit1             prints an error to stderr and exits 1, writes nothing
  hang              sleeps 120 s (the script's --exec-timeout must end it)
FAKE_CODEX_DB is the scratch database (the stub's, with the live store's table definitions).
"""
import datetime as dt
import json
import os
import re
import sqlite3
import sys
import time

argv = sys.argv[1:]
mode = os.environ.get("FAKE_CODEX_MODE", "two-ok")
db = os.environ["FAKE_CODEX_DB"]
model = "gpt-6-astra"
tag = None
for i, a in enumerate(argv):
    if a == "-m" and i + 1 < len(argv):
        model = argv[i + 1].split("/")[-1]
    m = re.search(r'"X-OmniRoute-Session-Id"="([^"]+)"', a)
    if m:
        tag = m.group(1)


def now():
    return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def rows(n, tin, tout):
    """The call_logs row of one request and its usage_history row (FAKE_CODEX_USAGE: unset = the request's own row, `foreign` = a foreign request's look-alike (another account and latency), `none` = no usage row)."""
    con = sqlite3.connect(db, timeout=30)
    ts = now()
    dur = 4000 + 211 * n
    con.execute(
        "insert into call_logs (id, timestamp, method, path, status, model, requested_model, provider, connection_id, duration, ttft_ms, tokens_in, tokens_out, detail_state, session_tag, reasoning_effort_requested, reasoning_effort_upstream, reasoning_source) "
        "values (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
        (f"fake-{n}-{time.time_ns()}", ts, "POST", "/v1/responses", 200, model, "codex/" + model, "codex", "ctl-conn", dur, 120, tin, tout, "ready", tag, "max", "max", "encrypted"),
    )
    variant = os.environ.get("FAKE_CODEX_USAGE", "")
    if variant != "none":
        own = variant != "foreign"
        con.execute("insert into usage_history (provider, model, connection_id, tokens_input, tokens_output, status, success, timestamp, endpoint, latency_ms, ttft_ms) values (?,?,?,?,?,?,?,?,?,?,?)",
                    ("codex", model, "ctl-conn" if own else "ctl-other", tin, tout, "200", 1, ts, "/v1/responses", dur + 1 if own else dur + 6000, 120.2 if own else 77.7))
    con.commit()
    con.close()


def event(obj):
    print(json.dumps(obj), flush=True)


if mode == "exit1":
    print("fake codex: simulated failure", file=sys.stderr)
    sys.exit(1)
if mode == "hang":
    time.sleep(120)
    sys.exit(0)
event({"type": "thread.started", "thread_id": "fake-thread"})
event({"type": "turn.started"})
time.sleep(0.5)
rows(1, 15045, 50)
if mode == "two-notool":
    event({"type": "item.completed", "item": {"id": "item_1", "type": "agent_message", "text": "ok"}})
    event({"type": "turn.completed", "usage": {"input_tokens": 15045, "cached_input_tokens": 0, "output_tokens": 50}})
    sys.exit(0)
event({"type": "item.started", "item": {"id": "item_1", "type": "command_execution", "command": "echo hi", "status": "in_progress"}})
event({"type": "item.completed", "item": {"id": "item_1", "type": "command_execution", "command": "echo hi", "aggregated_output": "hi\n", "exit_code": 0, "status": "completed"}})
if mode in ("two-ok", "two-nousage"):
    time.sleep(2.5)
    rows(2, 15106, 24)
event({"type": "item.completed", "item": {"id": "item_2", "type": "agent_message", "text": "ok"}})
if mode != "two-nousage":
    event({"type": "turn.completed", "usage": {"input_tokens": 15045 + 15106, "cached_input_tokens": 0, "output_tokens": 50 + 24}})
sys.exit(0)
