def load(path):
    """The guard module's functions without running its main(): everything above the trailing `try: main()` block."""
    text = Path(path).read_text(encoding="utf-8")
    head = text.split("\ntry:\n    main()")[0]
    ns = {"__file__": str(path), "__name__": "guard_under_test"}
    exec(compile(head, str(path), "exec"), ns)
    return ns

def info(tx=0, rx=0):

def rec(side, cip, cport, sip="127.0.0.1", sport=20128, recvq=0, sendq=0, rx=0, tx=0, owner=None, tcp_info=True, info_text=None):
    """A record of the connection (cip:cport -> sip:sport): the SERVER end (local = server) or the CLIENT end (local = client), as ss prints it. rx/tx are the END's own bytes_received and bytes_sent."""
    local, peer = (f"{sip}:{sport}", f"{cip}:{cport}") if side == "server" else (f"{cip}:{cport}", f"{sip}:{sport}")
    users = f' users:(("{owner[0]}",pid={owner[1]},fd={owner[2]}))' if owner else ""
    text = info_text if info_text is not None else (info(tx, rx) if tcp_info else "")
    return f"{recvq:<6} {sendq:<6} {local} {peer}{users} {text}".rstrip()


def to_two_line(lines):
    """The same records in guard 4's format (`ss -Htnpi`: an address line, then the info line indented by a tab)."""
    out = []
    for line in lines:
        m = re.match(r"^(\d+\s+\d+\s+\S+\s+\S+)(\s+users:\(\(.*?\)\))?\s*(.*)$", line)
        out += [m.group(1) + (m.group(2) or ""), "\t " + m.group(3)]
    return out


LISTEN_OK = "LISTEN 0      511    127.0.0.1:20128 0.0.0.0:*"


def fake_sh(listen_lines, record_lines, two_line=False, ss_rc=0, listen_rc=0):
    def sh(*a):
        if a[0] == "ss" and a[1] == "-Htln":
            return subprocess.CompletedProcess(a, listen_rc, "\n".join(listen_lines) + ("\n" if listen_lines else ""), "ss: fixture failure" if listen_rc else "")
        if a[0] == "ss":
            lines = to_two_line(record_lines) if two_line else record_lines
            return subprocess.CompletedProcess(a, ss_rc, "\n".join(lines) + ("\n" if lines else ""), "ss: fixture failure" if ss_rc else "")
        raise AssertionError(f"unexpected command {a}")
    return sh


def run_guard(ns, records, listen=(LISTEN_OK,), two_line=False, **kw):
    ns["sh"] = fake_sh(list(listen), records, two_line=two_line, **kw)
    try:
        c = ns["connections"](20128)
        return "OK", {k: (v["counters"], v["owners"]) for k, v in c.items()}, getattr(c, "unrelated", None)
    except ns["ObservationError"] as e:
        return "ObservationError", str(e)[:150], None
    except Exception as e:  # a guard that crashes is not an observation either
        return "CRASH", f"{type(e).__name__}: {str(e)[:100]}", None

def make_prefix(tools, which):
    """One install prefix under tools/ shaped as an installed one: bin/omniroute a relative symlink to lib/node_modules/omniroute/bin/omniroute.mjs (mode 0755, as npm links it), shim/lsof (0755) with the stand-in's
    content and the matching pinned hash, dist/BUILD_SHA, and for the target the drain marker in a server chunk."""
    name, build, marker = PREFIXES[which]
    base = tools / name
    entry = base / "lib/node_modules/omniroute/bin/omniroute.mjs"
    entry.parent.mkdir(parents=True)
    entry.write_text("#!/usr/bin/env node\n")
    entry.chmod(0o755)
    (base / "bin").mkdir()
    (base / "bin/omniroute").symlink_to("../lib/node_modules/omniroute/bin/omniroute.mjs")
    (base / "lib/node_modules/omniroute/dist").mkdir()
    (base / "lib/node_modules/omniroute/dist/BUILD_SHA").write_text(build + "\n")
    (base / "shim").mkdir()
    (base / "shim/lsof").write_text(SHIM_TEXT)
    (base / "shim/lsof").chmod(0o755)
    if marker:
        chunks = base / "lib/node_modules/omniroute/dist/.build/next/server/chunks"
        chunks.mkdir(parents=True)
        (chunks / "a.js").write_text("x;abort(),n.clearPendingRequest?.(),Promise.allSettled([y]);")
    return base


def prefix_verdict(ns, tools, which):
    """('intact', []) or ('FAIL', problems) of check_prefix(), or ('CRASH', the exception): a guard that crashes on a prefix exits 3 as a whole (ERROR ...), with no CHECK line."""
    name, build, marker = PREFIXES[which]
    try:
        problems = ns["check_prefix"](str(tools), name, build, SHIM_SHA, marker)
        return ("FAIL" if problems else "intact"), problems
    except Exception as e:
        return "CRASH", f"{type(e).__name__}: {str(e)[:100]}"

def lose_mode(rel, mode):
    return lambda base: os.chmod(os.path.realpath(base / rel), mode)


def to_directory(rel):
    def change(base):

    return change

def unit_segment():
    for line in open("/proc/self/cgroup", encoding="utf-8"):
        if line.startswith("0::"):
            return line.strip()[3:].rsplit("/", 1)[-1]
    return ""

def in_netns(cmd, env=None, timeout=120):
    """Run cmd inside a fresh private network namespace with the stub on 127.0.0.1:20128; (CompletedProcess, stub hits, netns id)."""
    p = subprocess.run(["unshare", "-rn", "--", sys.executable, str(launcher)] + cmd, capture_output=True, text=True, env=env, timeout=timeout)
    m = re.search(r"NETNS-STUB netns=(\S+) hits=(\d+) paths=(.*)", p.stderr)
    return p, (int(m.group(2)) if m else None), (m.group(1) if m else None)

# E0b: the guard's only network call, by AST: urlopen inside pending_total, and no other network primitive (so the one call is the one the namespace and the stub account for)
import ast
tree = ast.parse(Path(args.guard5).read_text(encoding="utf-8"))
network_calls, network_imports = [], []
for fn in [n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef)]:
    for node in ast.walk(fn):
        if isinstance(node, ast.Call):
            name = node.func.attr if isinstance(node.func, ast.Attribute) else getattr(node.func, "id", "")
            owner = getattr(node.func.value, "id", "") if isinstance(node.func, ast.Attribute) else ""
            if name in ("urlopen", "urlretrieve", "create_connection", "getaddrinfo", "gethostbyname", "HTTPConnection", "HTTPSConnection") or (name == "connect" and owner != "sqlite3"):   # sqlite3.connect opens a local file
                network_calls.append((fn.name, name))
for node in ast.walk(tree):
    if isinstance(node, ast.Import):
        network_imports += [a.name for a in node.names if a.name.split(".")[0] in ("socket", "requests", "httpx", "aiohttp", "urllib3", "http")]
    elif isinstance(node, ast.ImportFrom) and node.module and node.module.split(".")[0] in ("socket", "requests", "httpx", "aiohttp", "urllib3", "http"):
        network_imports.append(node.module)
ok0b = network_calls == [("pending_total", "urlopen")] and not network_imports
print(f"### E0b: the guard's network primitives, by AST: calls {sorted(set(network_calls))}, imports {network_imports or 'none besides urllib.request'}   [{'as expected' if ok0b else 'UNEXPECTED'}]")

def guard_e(case, title, records, expect_exit, expect_text, extra=()):
    rf = work / f"records-{case}.txt"

    env = dict(os.environ, PATH=f"{shim_dir}:{os.environ['PATH']}", SS_LISTEN=str(listen_file), SS_RECORDS=str(rf))
    cmd = [sys.executable, args.guard5, "--idle-only", "--identity-script", str(ident), "--db", str(db), "--worker-pattern", "nonexistent-process-pattern-for-the-controls", "--clean-for", "2", "--interval", "1", "--tolerate-unit", MY_UNIT, *extra]
    p, hits, netns = in_netns(cmd, env=env)
    out = p.stdout
    last = [l for l in out.splitlines() if l.startswith(("IDLE-RESULT", "GUARD-RESULT"))][-1:]
    fails = [l.strip()[:200] for l in out.splitlines() if "the socket observation failed" in l or "ACTIVE connection" in l][:1]
    ok = p.returncode == expect_exit and (expect_text in out or not expect_text) and hits is not None and hits >= 1 and netns != HOST_NETNS
    print(f"### {case}: {title}\n    exit={p.returncode} {last[0] if last else p.stderr.strip()[:160]}" + (f"\n    {fails[0]}" if fails else "") + f"\n    the guard's HTTP read reached the private stub {hits} time(s), in namespace {netns} (the host's is {HOST_NETNS})   [{'as expected' if ok else 'UNEXPECTED'}]")
    if not ok:
        UNEXPECTED.append(case)
