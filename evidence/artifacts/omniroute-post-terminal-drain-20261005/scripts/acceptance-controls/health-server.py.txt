#!/usr/bin/env python3
"""A fixture for the wrapper controls: a tiny HTTP server that stands in for the gateway's health endpoint for the IDENTITY read-back only (the acceptance scripts keep talking to the stub gateway).
GET /api/health answers 200 {"status":"ok"} for the first --fail-after requests and 503 {"status":"down"} from then on (the identity read-back makes exactly one health request per read, so the read
that first sees the failure is chosen by the count: the reads are, in order, the start of the wrapper, then the before and the after read of every run); GET /api/usage/history answers an empty pending counter.
usage: health-server.py --port-file FILE --fail-after N
"""
import argparse
import json
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

ap = argparse.ArgumentParser()
ap.add_argument("--port-file", required=True)
ap.add_argument("--fail-after", type=int, required=True)
args = ap.parse_args()
COUNT = [0]


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def log_message(self, *a):
        pass

    def do_GET(self):
        if self.path.startswith("/api/health"):
            COUNT[0] += 1
            ok = COUNT[0] <= args.fail_after
            body, status = json.dumps({"status": "ok" if ok else "down"}).encode(), 200 if ok else 503
        elif self.path.startswith("/api/usage/history"):
            body, status = json.dumps({"pending": {"byModel": {}, "byAccount": {}}}).encode(), 200
        else:
            body, status = b"", 404
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Connection", "close")
        self.end_headers()
        self.wfile.write(body)
        self.close_connection = True


server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
with open(args.port_file, "w") as f:
    f.write(str(server.server_address[1]))
server.serve_forever()
