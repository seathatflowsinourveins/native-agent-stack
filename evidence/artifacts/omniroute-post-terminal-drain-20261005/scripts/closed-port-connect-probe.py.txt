#!/usr/bin/env python3
"""Value-free probe of how a connect to a CLOSED loopback port behaves on this host (nothing listens on any probed port, nothing is sent): for ports in the range upstream's pack-boot harness uses (23000-26999) that nothing in this network
namespace holds, the connect timeout is observed; then the same ports are bound once (a bind without listen, then closed) and probed again. On an ordinary Linux kernel both give an immediate ECONNREFUSED.
usage: closed-port-connect-probe-20261006.py [--count 4] [--timeout 3]
"""
import argparse
import random
import socket
import subprocess
import time

ap = argparse.ArgumentParser()
ap.add_argument("--count", type=int, default=4)
ap.add_argument("--timeout", type=float, default=3.0)
a = ap.parse_args()


def free_of_sockets(port):
    return not subprocess.run(["ss", "-Htan", f"( sport = :{port} or dport = :{port} )"], capture_output=True, text=True).stdout.strip()


def probe(port):
    s = socket.socket()
    s.settimeout(a.timeout)
    t = time.time()
    try:
        s.connect(("127.0.0.1", port))
        r = "connected"
    except Exception as e:
        r = type(e).__name__
    s.close()
    return r, round((time.time() - t) * 1000)


ports = []
while len(ports) < a.count:
    p = random.randrange(23000, 27000)
    if p not in ports and free_of_sockets(p):
        ports.append(p)
print(f"# closed-port connect probe on this host, timeout {a.timeout} s; ports are picked at random from 23000-26999 with no socket on them (checked with ss) at the time of the pick and not bound by this script before the first probe; the earlier binding history of the ports is not known")
print("## 1. not bound by this script (no socket on the port per ss; nothing listens, nothing is sent)")
for p in ports:
    r1, ms1 = probe(p)
    r2, ms2 = probe(p)
    print(f"port {p}: {r1} after {ms1} ms (second probe: {r2} after {ms2} ms)")
print("## 2. the same ports after one bind (no listen) and close")
for p in ports:
    b = socket.socket()
    b.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    b.bind(("127.0.0.1", p))
    b.close()
    r, ms = probe(p)
    print(f"port {p}: {r} after {ms} ms")
