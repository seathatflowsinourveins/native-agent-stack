#!/bin/bash
# Handler-level check, in a scratch tree built from public refs (nothing is deployed), of the identity mechanism the native acceptance relies on: a caller-supplied X-OmniRoute-Session-Id header must end up as
# call_logs.session_tag of that request's row, through the real handleChat -> resolveConversationId -> chatCore -> call-log writer path with a stubbed upstream fetch and a temporary database
# (handler-e2e-session-tag.test.ts, sha256 pinned below; upstream's own tests/unit/call-logs-session-tag.test.ts covers only the storage layer). The client reads each response to the end, so the rows come from the normal completion path on every build.
# Four scenarios: a Responses request and a Chat Completions request with the header, two sequential requests with different tags, a request without the header.
# Steps (each logs its exit code to $OUT/progress.txt; a failed setup step, a tree-id mismatch or a violated expectation (counts, failing test names or the test runner's exit status) stops the script with exit 3):
#   composition C (v3.8.51 tag + affinity patch + PR 15167; tree id checked), npm ci
#   U  the test on the unpatched composition: 4 of 4 must pass, exit 0 (the mechanism is independent of the drain patch, and it is what the live gateway does today: checks/native-usage-calibration.txt)
#   N  the same test with E2E_SESSION_HEADER naming another header: the three tests that assert a tag must FAIL and the no-header test must pass, and node --test must exit 1 (the oracle discriminates; the control disables nothing else)
#   P  the patch under test applied (its tree id must equal EXPECT_TREE, or the script stops before this stage), then the same test: 4 of 4 must pass, exit 0
# Each stage keeps the runner's summary counters, its exit status ($OUT/test-<label>.exit), the failing tests' names, the full log and the scenarios' `E2E_SCENARIO {...}` lines in $OUT.
# The script owns only the directory it creates (mktemp under $WORK_ROOT, default ~/.cache) and deletes nothing. A missing or changed input file, or a missing mirror, stops it with exit 2 before anything is created.
# usage: DRAIN_DIR=<directory with drain-after-terminal.patch and handler-e2e-session-tag.test.ts> EV=<evidence folder of omniroute-wire-effort-20261005> EXPECT_TREE=<tree id of the patched tree> omniroute-handler-e2e-session-tag-20261006.sh
# Version 2 (2026-10-06): every stage's expectation includes the test runner's own exit status (full-scope read r12, P2-2); version 1 recorded that status but checked only the printed counters, so a run that printed the expected
# counters and then crashed, timed out (`timeout` exits 124) or ended some other way was accepted.
set -u
set -o pipefail
M=${MIRROR:-$HOME/.cache/omniroute-upstream-mirror-20261005.git}
D=${DRAIN_DIR:?set DRAIN_DIR to the directory holding drain-after-terminal.patch and handler-e2e-session-tag.test.ts}
TAG=c1e30b7676975feb298b49eff6ff58923c04b89e
PR=0585aba5589d5a1f49243a13a8db249558e7c9e3
COMPOSITION_TREE=f1336dfd6c8ebd81586dd6c71e7ef668ef8c949f
E2E_SHA256=f40472876b743faaa537ca80e3b58e8c7e3d35a2e0717b134f9fa54180e9b737
EV=${EV:?set EV to the evidence folder of the gateway-composition record}
EXPECT_TREE=${EXPECT_TREE:?set EXPECT_TREE to the tree id of the composition with the patch applied}
PFX="handler e2e session tag: "
T1="${PFX}a Responses request with X-OmniRoute-Session-Id stores that value as its row's session tag"
T2="${PFX}a Chat Completions request with X-OmniRoute-Session-Id stores that value as its row's session tag"
T3="${PFX}two requests with different tags each get their own tag (no cross-contamination)"
T4="${PFX}a request without the header does not carry a tag that no request sent"
export PATH="$HOME/.local/share/codex-ecosystem/tools/node-24.21.0/bin:/usr/local/bin:/usr/bin:/bin"
for f in "$D/drain-after-terminal.patch" "$D/handler-e2e-session-tag.test.ts" "$EV/patches/045aa81f3.patch"; do
  [ -f "$f" ] || { echo "missing input file: $f" >&2; exit 2; }
done
[ -d "$M" ] || { echo "missing OmniRoute mirror: $M" >&2; exit 2; }
[ "$(sha256sum "$D/handler-e2e-session-tag.test.ts" | cut -d' ' -f1)" = "$E2E_SHA256" ] || { echo "handler-e2e-session-tag.test.ts is not the pinned file (sha256 $E2E_SHA256)" >&2; exit 2; }
WORK=$(mktemp -d "${WORK_ROOT:-$HOME/.cache}/omniroute-handler-e2e-session-tag-XXXXXX") || exit 2
T=$WORK/tree
OUT=$WORK/out
mkdir -p "$OUT"
log() { echo "$(date -u +%H:%M:%SZ) $*" | tee -a "$OUT/progress.txt"; }
G() { git -c user.name=Scout -c user.email=scout@local -c core.hooksPath=/dev/null -C "$T" "$@"; }
step() { local label=$1; shift; "$@"; local rc=$?; log "$label exit=$rc"; [ $rc -eq 0 ] || exit 3; }
apply_pr_diff() { git --git-dir="$M" diff $PR^ $PR | G apply --index; }
counter() { grep -aE "^ℹ $2 " "$OUT/test-$1.log" | awk '{print $3}'; }
failing() { awk '/^✖ failing tests:/{f=1;next} f&&/^✖ /{sub(/ \([0-9.]+ms\)$/,""); sub(/^✖ /,""); print}' "$OUT/test-$1.log" | sort -u; }
expect_counts() { # label tests pass fail exit-status (the test runner's own status, which run() keeps in $OUT/test-<label>.exit)
  local t p f x; t=$(counter "$1" tests); p=$(counter "$1" pass); f=$(counter "$1" fail); x=$(cat "$OUT/test-$1.exit" 2>/dev/null)
  { [ "$t" = "$2" ] && [ "$p" = "$3" ] && [ "$f" = "$4" ]; } || { log "$1 EXPECTATION VIOLATED: tests=$t pass=$p fail=$f, expected tests=$2 pass=$3 fail=$4: stopping"; exit 3; }
  # version 2: a run can print the expected counters and still crash, time out or end some other way, so the stage's exit status is part of its expectation (a call that names none fails)
  if [ -n "${5:-}" ] && [ "$x" = "$5" ]; then log "$1 expectation met: tests=$t pass=$p fail=$f"; else log "$1 EXIT STATUS VIOLATED: exit=${x:-none}, expected exit=${5:-none}: stopping"; exit 3; fi
}
expect_failing() {
  local got; got=$(failing "$1")
  if [ "$got" = "$2" ]; then log "$1 failing tests are as expected: $(printf '%s' "$got" | grep -c . | tr -d ' ') name(s)"; else log "$1 FAILING SET VIOLATED: got [$got]: stopping"; exit 3; fi
}
step clone git clone -q --shared --no-checkout "$M" "$T"
step "fetch pr15167" G fetch -q "$M" "+refs/pull/15167/head:refs/pull/15167/head"
step "checkout tag" G checkout -q --detach $TAG
( cd "$T" && nice -n 19 npm ci --no-audit --no-fund > "$OUT/npm-ci.log" 2>&1 ); rc=$?; log "npm ci exit=$rc"; [ $rc -eq 0 ] || exit 3
step "apply affinity patch" G apply --index "$EV/patches/045aa81f3.patch"
step "apply pr15167 diff" apply_pr_diff
step "commit composition" G commit -q -m "composition"
ACTUAL_TREE=$(G rev-parse 'HEAD^{tree}')
if [ "$ACTUAL_TREE" = "$COMPOSITION_TREE" ]; then log "composition tree=$ACTUAL_TREE equals the expected tree"; else log "composition tree=$ACTUAL_TREE DIFFERS from the expected tree $COMPOSITION_TREE: stopping"; exit 3; fi
FLAGS="--import tsx/esm --import ./open-sse/utils/setupPolyfill.ts --import ./tests/_setup/isolateDataDir.ts"
run() { # label [VAR=value ...]; runs $E2E
  local label=$1; shift
  ( cd "$T" && env DISABLE_SQLITE_AUTO_BACKUP=true "$@" nice -n 19 timeout 600 node $FLAGS --test --test-force-exit $E2E > "$OUT/test-$label.log" 2>&1 ); local rc=$?
  echo "$rc" > "$OUT/test-$label.exit"
  log "$label exit=$rc tests=$(counter "$label" tests) suites=$(counter "$label" suites) pass=$(counter "$label" pass) fail=$(counter "$label" fail) cancelled=$(counter "$label" cancelled) skipped=$(counter "$label" skipped) todo=$(counter "$label" todo)"
  grep -a "E2E_SCENARIO" "$OUT/test-$label.log" > "$OUT/scenarios-$label.txt"
}
cd "$T" || exit 2
E2E=tests/integration/handler-e2e-session-tag.test.ts
step "copy the e2e test file" cp "$D/handler-e2e-session-tag.test.ts" $E2E
log "e2e test file sha256 $(sha256sum $E2E | cut -d' ' -f1) git blob $(G hash-object $E2E)"
run U-session-tag-unpatched
expect_counts U-session-tag-unpatched 4 4 0 0
expect_failing U-session-tag-unpatched ""
run N-session-tag-wrong-header E2E_SESSION_HEADER=X-Not-The-Session-Header
expect_counts N-session-tag-wrong-header 4 1 3 1
expect_failing N-session-tag-wrong-header "$(printf '%s\n%s\n%s' "$T1" "$T2" "$T3" | sort -u)"
step "apply patch under test" G apply --index "$D/drain-after-terminal.patch"
step "commit patch under test" G commit -q -m "drain patch under test"
PATCHED_TREE=$(G rev-parse 'HEAD^{tree}')
if [ "$PATCHED_TREE" = "$EXPECT_TREE" ]; then log "patched tree=$PATCHED_TREE equals the expected tree"; else log "patched tree=$PATCHED_TREE DIFFERS from the expected tree $EXPECT_TREE: stopping before the patched stage"; exit 3; fi
run P-session-tag-patched
expect_counts P-session-tag-patched 4 4 0 0
expect_failing P-session-tag-patched ""
log "DONE; the scratch tree and the logs stay in $WORK (remove it with rm -rf after reading)"
