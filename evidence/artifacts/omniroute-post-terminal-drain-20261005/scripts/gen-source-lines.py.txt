def show(path):
    return subprocess.run(["git", "--git-dir=" + M, "show", f"{R}:{path}"], capture_output=True, text=True, check=True).stdout.split("\n")

def sec(path, ranges, title):
    lines = show(path)
    out = [f"## {path} {title}"]
    prev = None
    for a, b in ranges:
        if prev is not None and a > prev + 1:
            out.append("...")
        for n in range(a, b + 1):
            out.append(f"{n}: {lines[n-1]}".rstrip() if lines[n-1].strip() else f"{n}:")
        prev = b
    return "\n".join(out)
