#!/bin/bash
# Control for the ownership check of the smoke script (owned_listener): the smoke script is sourced (SMOKE_SOURCE_ONLY=1) so that owned_listener and members are defined, then
# a dummy health server (its own process group, under set -m) listens on a free port and the check is asked twice: with the dummy's own group (must say yes) and with an unrelated group (must say no).
set -u
set -m
SMOKE=${SMOKE_SCRIPT:?set SMOKE_SCRIPT to the smoke script (scripts/smoke.sh.txt)}
SMOKE_SOURCE_ONLY=1 . "$SMOKE"
PORT=$(python3 -c 'import socket; s = socket.socket(); s.bind(("127.0.0.1", 0)); print(s.getsockname()[1]); s.close()')
python3 - "$PORT" <<'PYEOF' &
import http.server, sys
class H(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        body = b'{"status":"ok"}'
        self.send_response(200); self.send_header("Content-Type", "application/json"); self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def log_message(self, *a):
        pass
http.server.HTTPServer(("127.0.0.1", int(sys.argv[1])), H).serve_forever()
PYEOF
DUMMY=$!
sleep 1
DUMMY_PGID=$(ps -o pgid= -p "$DUMMY" | tr -d ' ')
sleep 300 &
OTHER=$!
OTHER_PGID=$(ps -o pgid= -p "$OTHER" | tr -d ' ')
echo "listener: pid $DUMMY in process group $DUMMY_PGID on port $PORT; an unrelated process: pid $OTHER in process group $OTHER_PGID"
PGID=$DUMMY_PGID
echo "owned_listener for the listener's own process group $PGID: $(owned_listener)   (expected: yes)"
PGID=$OTHER_PGID
echo "owned_listener for the unrelated process group $PGID: $(owned_listener)   (expected: no)"
kill -- "-$DUMMY_PGID" "-$OTHER_PGID" 2>/dev/null
wait 2>/dev/null
echo "both stopped (their own process groups only)"
