#!/bin/bash
# Handler-level end-to-end check of the post-terminal client disconnect, in a scratch tree built from public refs (nothing is deployed).
# The test (handler-e2e-terminal-close.test.ts) calls the real handleChat -> chatCore -> stream pipeline -> call-log and usage writers with a stubbed upstream fetch, in the style of
# upstream's tests/integration/chat-pipeline.test.ts: a Responses client reads until response.completed and closes while the upstream body is still open for a short tail.
# Steps (each logs its exit code to $OUT/progress.txt; a failed setup step or a tree-id mismatch stops the script with exit 3):
#   composition C (v3.8.51 tag + affinity patch + PR 15167; tree id checked), npm ci
#   U  the e2e test on the unpatched composition: the scenarios that the bug affects must FAIL (3 of 4: a Codex-originated client, and clients whose request signal aborts first)
#   P  the patch under test applied (its tree id must equal EXPECT_TREE, or the script stops before this stage), then the same test: 4 of 4 must pass
# The script owns only the directory it creates (mktemp under $WORK_ROOT, default ~/.cache) and deletes nothing. A missing input file or mirror stops it with exit 2 before anything is created.
# usage: DRAIN_DIR=<directory with drain-after-terminal.patch and handler-e2e-terminal-close.test.ts> EV=<evidence folder of omniroute-wire-effort-20261005> EXPECT_TREE=<tree id of the patched tree> omniroute-handler-e2e-20261005.sh
set -u
set -o pipefail
M=${MIRROR:-$HOME/.cache/omniroute-upstream-mirror-20261005.git}
D=${DRAIN_DIR:?set DRAIN_DIR to the directory holding drain-after-terminal.patch and handler-e2e-terminal-close.test.ts}
TAG=c1e30b7676975feb298b49eff6ff58923c04b89e
PR=0585aba5589d5a1f49243a13a8db249558e7c9e3
COMPOSITION_TREE=f1336dfd6c8ebd81586dd6c71e7ef668ef8c949f
EV=${EV:?set EV to the evidence folder of the gateway-composition record}
EXPECT_TREE=${EXPECT_TREE:?set EXPECT_TREE to the tree id of the composition with the patch applied}
export PATH="$HOME/.local/share/codex-ecosystem/tools/node-24.21.0/bin:/usr/local/bin:/usr/bin:/bin"
for f in "$D/drain-after-terminal.patch" "$D/handler-e2e-terminal-close.test.ts" "$EV/patches/045aa81f3.patch"; do
  [ -f "$f" ] || { echo "missing input file: $f" >&2; exit 2; }
done
[ -d "$M" ] || { echo "missing OmniRoute mirror: $M" >&2; exit 2; }
WORK=$(mktemp -d "${WORK_ROOT:-$HOME/.cache}/omniroute-handler-e2e-XXXXXX") || exit 2
T=$WORK/tree
OUT=$WORK/out
mkdir -p "$OUT"
log() { echo "$(date -u +%H:%M:%SZ) $*" | tee -a "$OUT/progress.txt"; }
G() { git -c user.name=Scout -c user.email=scout@local -c core.hooksPath=/dev/null -C "$T" "$@"; }
step() { local label=$1; shift; "$@"; local rc=$?; log "$label exit=$rc"; [ $rc -eq 0 ] || exit 3; }
apply_pr_diff() { git --git-dir="$M" diff $PR^ $PR | G apply --index; }
step clone git clone -q --shared --no-checkout "$M" "$T"
step "fetch pr15167" G fetch -q "$M" "+refs/pull/15167/head:refs/pull/15167/head"
step "checkout tag" G checkout -q --detach $TAG
( cd "$T" && nice -n 19 npm ci --no-audit --no-fund > "$OUT/npm-ci.log" 2>&1 ); rc=$?; log "npm ci exit=$rc"; [ $rc -eq 0 ] || exit 3
step "apply affinity patch" G apply --index "$EV/patches/045aa81f3.patch"
step "apply pr15167 diff" apply_pr_diff
step "commit composition" G commit -q -m "composition"
ACTUAL_TREE=$(G rev-parse 'HEAD^{tree}')
if [ "$ACTUAL_TREE" = "$COMPOSITION_TREE" ]; then log "composition tree=$ACTUAL_TREE equals the expected tree"; else log "composition tree=$ACTUAL_TREE DIFFERS from the expected tree $COMPOSITION_TREE: stopping"; exit 3; fi
FLAGS="--import tsx/esm --import ./open-sse/utils/setupPolyfill.ts --import ./tests/_setup/isolateDataDir.ts"
run() {
  local label=$1; shift
  ( cd "$T" && DISABLE_SQLITE_AUTO_BACKUP=true nice -n 19 timeout 600 node $FLAGS --test --test-force-exit "$@" > "$OUT/test-$label.log" 2>&1 ); local rc=$?
  log "$label exit=$rc tests=$(grep -aE '^ℹ tests ' "$OUT/test-$label.log" | awk '{print $3}') pass=$(grep -aE '^ℹ pass ' "$OUT/test-$label.log" | awk '{print $3}') fail=$(grep -aE '^ℹ fail ' "$OUT/test-$label.log" | awk '{print $3}')"
}
cd "$T" || exit 2
E2E=tests/integration/handler-e2e-terminal-close.test.ts
step "copy the e2e test file" cp "$D/handler-e2e-terminal-close.test.ts" $E2E
run U-e2e-unpatched $E2E
step "apply patch under test" G apply --index "$D/drain-after-terminal.patch"
step "commit patch under test" G commit -q -m "drain patch under test"
PATCHED_TREE=$(G rev-parse 'HEAD^{tree}')
if [ "$PATCHED_TREE" = "$EXPECT_TREE" ]; then log "patched tree=$PATCHED_TREE equals the expected tree"; else log "patched tree=$PATCHED_TREE DIFFERS from the expected tree $EXPECT_TREE: stopping before the patched stage"; exit 3; fi
run P-e2e-patched $E2E
log "DONE; the scratch tree and the logs stay in $WORK (remove it with rm -rf after reading)"
