#!/usr/bin/env python3
"""Read-only look (mode=ro, metadata columns only) at which call_logs rows of the Responses path carry the effort columns, and what else is different about the rows that do not.

For the rows of the given models since a given time: counts of rows with and without reasoning_effort_upstream, split by reasoning_source and reasoning_encrypted, and by whether tokens_reasoning is positive; then the individual rows
of one chosen window (timestamp, model, status, effort columns, reasoning_source, reasoning_encrypted, tokens_reasoning, tokens_out, duration).
usage: probe-effort-null-rows-20261006.py --since 2026-10-06T00:00:00Z [--models gpt-6.1-sol,gpt-6.1-sol-max,gpt-6-astra] [--rows-from 2026-10-06T02:08:00Z --rows-to 2026-10-06T02:10:00Z]
"""
import argparse
import os
import sqlite3

ap = argparse.ArgumentParser()
ap.add_argument("--since", required=True)
ap.add_argument("--models", default="gpt-6.1-sol,gpt-6.1-sol-max,gpt-6-astra")
ap.add_argument("--rows-from")
ap.add_argument("--rows-to")
ap.add_argument("--db", default=os.path.expanduser("~/.local/share/omniroute/storage.sqlite"))
args = ap.parse_args()
con = sqlite3.connect(f"file:{args.db}?mode=ro", uri=True)
models = args.models.split(",")
q = ",".join("?" for _ in models)
base = f"from call_logs where path = '/v1/responses' and status = 200 and timestamp >= ? and model in ({q})"
print("model | rows | effort_upstream NOT NULL | NULL | NULL with tokens_reasoning>0 | NULL with reasoning_source set")
for m in models:
    r = con.execute(f"select count(*), sum(reasoning_effort_upstream is not null), sum(reasoning_effort_upstream is null), "
                    f"sum(reasoning_effort_upstream is null and coalesce(tokens_reasoning,0) > 0), sum(reasoning_effort_upstream is null and reasoning_source is not null) {base} and model = ?",
                    (args.since, *models, m)).fetchone()
    print(m, "|", *r)
print()
print("reasoning_source x reasoning_encrypted x effort_upstream-is-null (all selected models)")
for row in con.execute(f"select reasoning_source, reasoning_encrypted, reasoning_effort_upstream is null, count(*) {base} group by 1, 2, 3 order by 4 desc", (args.since, *models)).fetchall():
    print(row)
if args.rows_from and args.rows_to:
    print()
    print("rows", args.rows_from, "to", args.rows_to, "(timestamp, model, effort_requested, effort_upstream, reasoning_source, reasoning_encrypted, tokens_reasoning, tokens_out, duration_ms)")
    for row in con.execute("select timestamp, model, reasoning_effort_requested, reasoning_effort_upstream, reasoning_source, reasoning_encrypted, tokens_reasoning, tokens_out, duration from call_logs "
                           "where path = '/v1/responses' and timestamp >= ? and timestamp <= ? order by timestamp", (args.rows_from, args.rows_to)).fetchall():
        print(row)
