#!/usr/bin/env python3
"""Offline controls for the live-acceptance scripts: version 1 (superseded) and version 2 (frozen) run against the stub gateway and a scratch database, scenario by scenario, with the raw output kept.

Nothing here touches a live gateway: the stub (stub-gateway.py) is a test fixture with the live store's table definitions and its own database; the native shapes run the REAL `codex exec` against the stub
(it closes right after the terminal event, as in the field) except where a tool-call turn or a failing exec is needed, which use fake-codex.py. A scenario is a control of a script's DECISION LOGIC on synthetic
rows (evidence class local_integration), never evidence about OmniRoute.

Each run prints `CONTROL-RESULT <scenario> <script> expected_exit=N observed_exit=M OK|UNEXPECTED`. For a version-1 run that is expected to give the WRONG answer (the defect a review reproduced) the line says
`defect-demonstration`. The driver's own exit status is 0 when every observed exit status is the expected one.

usage: run-controls.py --work DIR --schema FILE --repo REPO [--only ID[,ID...]]
scenarios: U (usage errors U1-U5), I (identity I1-I3), G (guard G1-G16), W (wrapper W1-W12), R1..R9, R10 (R10a-d) (raw), N1, N1v1 (slow), N2a, N2b, N3, N4a, N4b, N4c, N4d, N5a, N5b, N6, N6o, N7, N10 (N10a-b) (native), RC1, RC2 (receipt)
"""

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


class Fixture:
    def __init__(self, work, schema, repo):
        self.work = Path(work)
        self.work.mkdir(parents=True, exist_ok=True)
        os.chmod(self.work, 0o700)
        self.home = self.work / "home"
        self.db = self.home / ".local/share/omniroute/storage.sqlite"
        self.db.parent.mkdir(parents=True, exist_ok=True)
        self.modefile = self.work / "mode.json"
        self.portfile = self.work / "port"
        self.reqlog = Path(str(self.db) + ".requests.jsonl")
        self.set_mode({"mode": "normal"})
        self.stub = subprocess.Popen([sys.executable, str(HERE / "stub-gateway.py"), "--db", str(self.db), "--schema", schema, "--mode-file", str(self.modefile), "--port-file", str(self.portfile)],
                                     stdout=open(self.work / "stub.log", "w"), stderr=subprocess.STDOUT)
        for _ in range(100):
            if self.portfile.exists() and self.portfile.read_text().strip():
                break
            time.sleep(0.1)
        self.port = int(self.portfile.read_text())
        self.repo = self.work / "repo"
        template = self.repo / "adoption/templates/codex.omniroute.config.toml"
        template.parent.mkdir(parents=True, exist_ok=True)
        template.write_text((Path(repo) / "adoption/templates/codex.omniroute.config.toml").read_text().replace("http://127.0.0.1:20128/v1", f"http://127.0.0.1:{self.port}/v1"))
        self.bin = self.work / "bin"
        self.bin.mkdir(exist_ok=True)
        fake = self.bin / "codex"
        if not fake.exists():
            fake.symlink_to(HERE / "fake-codex.py")
        os.chmod(HERE / "fake-codex.py", 0o755)

    def set_mode(self, cfg):
        self.modefile.write_text(json.dumps(cfg))

    def reset(self, cfg):
        time.sleep(1.5)  # let a previous scenario's delayed stub writes (duplicate rows, status flips) land before the tables are emptied
        con = sqlite3.connect(self.db, timeout=30)
        con.execute("delete from call_logs")
        con.execute("delete from usage_history")
        con.commit()
        con.close()
        self.reqlog.write_text("")
        self.set_mode(cfg)

    def insert_unrelated(self, delay):
        def run():
            time.sleep(delay)
            con = sqlite3.connect(self.db, timeout=30)
            ts = dt_now()
            con.execute("insert into call_logs (id, timestamp, method, path, status, model, requested_model, provider, tokens_in, tokens_out, detail_state, session_tag) values (?,?,?,?,?,?,?,?,?,?,?,?)",
                        ("unrelated-1", ts, "POST", "/v1/responses", 200, "gpt-6-astra", "codex/gpt-6-astra", "codex", 15045, 50, "ready", "someone-else"))
            con.execute("insert into usage_history (provider, model, tokens_input, tokens_output, status, success, timestamp) values (?,?,?,?,?,?,?)", ("codex", "gpt-6-astra", 15045, 50, "200", 1, ts))
            con.commit()
            con.close()
        t = threading.Thread(target=run, daemon=True)
        t.start()
        return t

    def close(self):
        self.stub.terminate()
        try:
            self.stub.wait(timeout=10)
        except Exception:
            self.stub.kill()


def dt_now():
    import datetime as dt
    return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def mask(text, fx):
    return str(text).replace(str(fx.work), "$WORK").replace(HOME + "/", "$HOME/")


def run(fx, scenario, label, argv, expected, env_extra=None, defect=False, timeout=900, note=None):
    env = dict(os.environ)
    env.update(env_extra or {})
    shown = " ".join(mask(a, fx) for a in argv)
    print(f"--- {scenario} / {label}: {'bash' if str(argv[0]).endswith('.sh') else 'python3'} {shown}" + (f"   [env: {', '.join(k + '=' + mask(v, fx) for k, v in (env_extra or {}).items())}]" if env_extra else ""), flush=True)
    if note:
        print(f"    note: {note}")
    t0 = time.time()
    try:
        exe = ["bash"] if str(argv[0]).endswith(".sh") else [sys.executable]
        proc = subprocess.run(exe + [str(a) for a in argv], env=env, capture_output=True, text=True, timeout=timeout)
        rc, out = proc.returncode, proc.stdout + proc.stderr
    except subprocess.TimeoutExpired as e:
        rc, out = 124, f"(driver timeout after {timeout} s)\n{(e.stdout or b'').decode() if isinstance(e.stdout, bytes) else (e.stdout or '')}"
    for line in mask(out, fx).rstrip("\n").splitlines():
        print("    " + line)
    ok = rc == expected
    kind = "defect-demonstration " if defect else ""
    print(f"CONTROL-RESULT {scenario} {label} {kind}expected_exit={expected} observed_exit={rc} elapsed_s={time.time() - t0:.0f} {'OK' if ok else 'UNEXPECTED'}\n", flush=True)
    if not ok:
        UNEXPECTED.append(f"{scenario} {label}")
    return rc, out


def raw(fx, scenario, title, cfg, modes, v1_exit, v2_exit, v1_note=None, v2_extra=None, v1_extra=None, v2_only=False):
    print(f"### {scenario}: {title}\n    stub mode: {json.dumps(cfg)}", flush=True)
    for label, script, expected, extra, note in (("v1", V1_RAW, v1_exit, v1_extra, v1_note), ("v2", V2_RAW, v2_exit, v2_extra, None)):
        if label == "v1" and v2_only:
            continue
        fx.reset(cfg)
        if label == "v1":
            argv = [script, "--port", fx.port, "--expect", "all-logged", *RAW_WAITS, *(extra or []), *modes]
            env = {"HOME": str(fx.home)}
        else:
            argv = [script, "--port", fx.port, "--db", fx.db, "--expect", "all-logged", *RAW_WAITS, *(extra or []), *modes]
            env = None
        defect = label == "v1" and v1_exit in (0,) and scenario in ("R3", "R4", "N2a", "N2b", "N5b")
        run(fx, scenario, label, argv, expected, env_extra=env, defect=defect, note=note)


def native(fx, scenario, title, cfg, shapes, v1_exit, v2_exit, mode_env=None, side=None, v1_note=None, v2_extra=None, v1_extra=None, v2_only=False, v1_timeout=900):
    print(f"### {scenario}: {title}\n    stub mode: {json.dumps(cfg)}" + (f"   fake codex mode: {mode_env}" if mode_env else ""), flush=True)
    for label, script, expected in (("v1", V1_NAT, v1_exit), ("v2", V2_NAT, v2_exit)):
        if label == "v1" and v2_only:
            continue
        fx.reset(cfg)
        scratch = fx.work / f"codex-home-{scenario}-{label}"
        if label == "v1":
            argv = [script, "--expect", "all-logged", "--wait1", "5", "--wait2", "10", "--quiet-timeout", "1", "--repo", fx.repo, *(v1_extra or []), *shapes]
            env = {"HOME": str(fx.home)}
        else:
            argv = [script, *NAT_V2, "--repo", fx.repo, "--db", fx.db, "--scratch", scratch, *(v2_extra or []), *shapes]
            env = {}
        if mode_env:
            env.update({"FAKE_CODEX_MODE": mode_env, "FAKE_CODEX_DB": str(fx.db)})
            if label == "v1":
                env["PATH"] = f"{fx.bin}:{os.environ['PATH']}"
            else:
                argv += ["--codex", str(fx.bin / "codex")]
        thread = side(fx) if side else None
        defect = label == "v1" and expected == 0 and scenario in ("N2a", "N2b", "N5b")
        run(fx, scenario, label, argv, expected, env_extra=env, defect=defect, note=v1_note if label == "v1" else None, timeout=v1_timeout if label == "v1" else 900)
        if thread:
            thread.join()
        if scenario == "N1" and label == "v2":
            ids = sorted({json.loads(l)["x-omniroute-session-id"] for l in fx.reqlog.read_text().splitlines() if l.strip()})
            print(f"    the stub saw {len(ids)} distinct X-OmniRoute-Session-Id values from the three codex exec turns: {ids}")
            if len(ids) != 3:
                UNEXPECTED.append("N1 distinct session ids")


def receipt(work):
    work = Path(work)
    work.mkdir(parents=True, exist_ok=True)
    clean = work / "show-log-clean.conf"
    clean.write_text("[Service]\nEnvironment=OMNIROUTE_SHOW_LOG=1\n")
    extra = work / "show-log-with-extra.conf"
    extra.write_text("[Service]\nEnvironment=OMNIROUTE_SHOW_LOG=1\nEnvironment=SYNTHETIC_CONTROL_SECRET=control-value-0123456789\n")
    for sid, title, path, must_not in (("RC1", "a drop-in with only the show-log line", clean, []), ("RC2", "a drop-in with an extra synthetic assignment that looks like a credential", extra, ["SYNTHETIC_CONTROL_SECRET", "control-value-0123456789"])):
        print(f"### {sid}: {title}")
        v1_line = [l for l in path.read_text().splitlines() if l.strip() and not l.lstrip().startswith("#")]
        print(f"--- {sid} / v1 (the print statement of version 1 applied to the same file): drop-in file: effective lines: {v1_line}")
        proc = subprocess.run([sys.executable, str(V2_RCP), "--describe-dropin", str(path)], capture_output=True, text=True)
        out = (proc.stdout + proc.stderr).replace(HOME + "/", "$HOME/").replace(str(work), "$WORK")
        print(f"--- {sid} / v2: python3 {V2_RCP.name} --describe-dropin {path.name}\n    {out.strip()}")
        leaked = [m for m in must_not if m in proc.stdout + proc.stderr]
        ok = proc.returncode == 0 and not leaked and "declared by its own line: True" in out
        print(f"CONTROL-RESULT {sid} v2 expected_no_leak observed_leaked={leaked} {'OK' if ok else 'UNEXPECTED'}\n")
        if not ok:
            UNEXPECTED.append(sid)

def make_tools(work):
    """A scratch tools directory with a stand-in rollback prefix and a stand-in target prefix (the target carries the drain marker); returns (tools dir, sha256 of the stand-in shim)."""
    import hashlib as _h
    tools = Path(work) / "tools"
    shim_text = "shim v2 stand-in\n"
    for name, build in (("omniroute-ctl-rollback", "cf6748d04"), ("omniroute-ctl-target", "2a41147b7")):
        base = tools / name
        (base / "bin").mkdir(parents=True, exist_ok=True)
        (base / "bin/omniroute").write_text("#!/bin/sh\n")
        os.chmod(base / "bin/omniroute", 0o755)
        (base / "lib/node_modules/omniroute/dist").mkdir(parents=True, exist_ok=True)
        (base / "lib/node_modules/omniroute/dist/BUILD_SHA").write_text(build + "\n")
        (base / "shim").mkdir(exist_ok=True)
        (base / "shim/lsof").write_text(shim_text)
    chunks = tools / "omniroute-ctl-target/lib/node_modules/omniroute/dist/.build/next/server/chunks"
    chunks.mkdir(parents=True, exist_ok=True)
    (chunks / "a.js").write_text("x;abort(),n.clearPendingRequest?.(),Promise.allSettled([y]);")
    return tools, _h.sha256(shim_text.encode()).hexdigest()


def dummy(tools, name="omniroute-ctl-rollback", extra=()):

    return subprocess.Popen([sys.executable, "-c", "import time; time.sleep(3600)", f"{tools}/{name}/bin/omniroute", "serve", "--port", "20128", *extra])   # long enough for a driver that runs twelve wrapper runs (a 600 s dummy ended in the middle of W12 in a first attempt)


def effort_controls(fx):
    """The effort probes as GROUPS of attempts (version 2.1) against the stub that can write NULL effort columns, as the real gateway does for a response without encrypted reasoning; and the flaw of version 2.0 (a single NULL attempt fails a healthy canary)."""
    modes3 = ["effort:cx/gpt-6-astra:max"] * 3
    waits = [*RAW_WAITS, "--effort-close-after", "3"]

    def baseline(label, cfg, modes=None):
        fx.reset(cfg)
        _, out = run(fx, "R8-baseline", label, [V2_RAW, "--port", fx.port, "--db", fx.db, "--expect", "observe", *waits, *(modes or modes3)], 0)
        path = fx.work / f"effort-baseline-{label.split()[0]}.txt"
        path.write_text(out)
        return path

    def canary(sid, title, cfg, base, expected, modes=None, script=V2_RAW, extra=()):
        print(f"### {sid}: {title}\n    stub mode: {json.dumps(cfg)}", flush=True)
        fx.reset(cfg)
        used = RAW_WAITS if script == V20_RAW else waits   # version 2.0 has no --effort-close-after (it reads an effort probe to the end of the stream, which the stub ends 3 s after the terminal event)
        run(fx, sid, "v2.2" if script == V2_RAW else "v2.0", [script, "--port", fx.port, "--db", fx.db, "--expect", "all-logged", "--baseline", base, *extra, *used, *(modes or modes3)], expected, defect=(script == V20_RAW and expected == 1))

    print("### R8 baselines: version 2.1 in observe mode against the healthy stub (three attempts, the stub writes max/max), against a stub whose responses all lack the effort columns, and one with a different upstream effort", flush=True)
    healthy = baseline("healthy three attempts", {"mode": "normal"})
    all_null = baseline("all-null three attempts", {"mode": "normal", "null_effort": [True]})
    other = baseline("xhigh three attempts", {"mode": "normal", "effort_upstream_override": "xhigh"})
    mixed = fx.work / "effort-baseline-mixed.txt"
    mixed.write_text(healthy.read_text() + other.read_text())
    canary("R8a", "healthy canary, three attempts, all observed and equal to the baseline: PASS", {"mode": "normal"}, healthy, 0)
    canary("R8b", "the upstream effort differs from the baseline's on every attempt (xhigh against max): FAIL", {"mode": "normal", "effort_upstream_override": "xhigh"}, healthy, 1)
    canary("R8c", "no attempt carries the effort columns (all NULL): the rows pass, the group is INCONCLUSIVE, never a FAIL and never a PASS", {"mode": "normal", "null_effort": [True]}, healthy, 2)
    canary("R8d", "one attempt of three without the effort columns, two observed and equal: PASS (the NULL attempt is unobservable, not wrong)", {"mode": "normal", "null_effort": [True, False, False]}, healthy, 0)
    canary("R8e", "two attempts NULL and the one observed attempt differs (xhigh): FAIL", {"mode": "normal", "null_effort": [True, True, False], "effort_upstream_override": "xhigh"}, healthy, 1)
    canary("R8f", "the baseline file holds no OBSERVED effort line for the mode (every baseline attempt had NULL columns): exit 3 before any request", {"mode": "normal"}, all_null, 3)
    canary("R8g", "the baseline's observed lines for the mode disagree (max and xhigh): exit 3 before any request", {"mode": "normal"}, mixed, 3)
    canary("R8h", "VERSION 2.0 (the first version of this script, kept in superseded/) with ONE attempt of which the gateway stored no effort columns: it FAILS a healthy canary, the flaw this round fixes", {"mode": "normal", "null_effort": [True]}, healthy, 1,
           modes=["effort:cx/gpt-6-astra:max"], script=V20_RAW, extra=())
    canary("R8i", "VERSION 2.1 with the same ONE NULL attempt: the group is INCONCLUSIVE (exit 2), not a FAIL", {"mode": "normal", "null_effort": [True]}, healthy, 2, modes=["effort:cx/gpt-6-astra:max"])
    print("### R8j: the suffix model (no client effort, the model id carries it): requested NULL and upstream max is an OBSERVED attempt, and it is compared as such")
    sm = ["effort:cx/gpt-6.1-sol-max"] * 3
    sbase = baseline("suffix-model three attempts", {"mode": "normal"}, modes=sm)
    canary("R8j", "suffix model, healthy: PASS", {"mode": "normal"}, sbase, 0, modes=sm)

    cwd_note = "run from /tmp, where the relative path does not exist"
    print("### U1: --baseline names a file that does not exist (a relative path run from /tmp, as the frozen version-1 command was): version 1 crashes with exit 1, the code of a FAIL; version 2 stops with exit 3 before sending any request")
    for label, script, expected, port_arg in (("v1", V1_RAW, 1, []), ("v2", V2_RAW, 3, ["--db", fx.db])):
        fx.reset({"mode": "normal"})
        argv = [script, "--port", fx.port, *port_arg, "--expect", "all-logged", "--baseline", "checks/live-acceptance-baseline-raw.txt", *RAW_WAITS, "effort:cx/gpt-6-astra:max"]
        run(fx, "U1", label, argv, expected, env_extra={"HOME": str(fx.home)} if label == "v1" else None, defect=(label == "v1"), note=cwd_note)
        print(f"    requests the stub received: {len([l for l in fx.reqlog.read_text().splitlines() if l.strip()])}")
    print("### U2: an unknown mode: version 1 exits 1 (SystemExit with a message), version 2 exits 3")
    for label, script, expected, extra in (("v1", V1_RAW, 1, []), ("v2", V2_RAW, 3, ["--db", fx.db])):
        fx.reset({"mode": "normal"})
        run(fx, "U2", label, [script, "--port", fx.port, *extra, "--expect", "observe", *RAW_WAITS, "responses:nonsense"], expected, env_extra={"HOME": str(fx.home)} if label == "v1" else None, defect=(label == "v1"))
    print("### U3: a bad option value (--wait1 abc): version 1 exits 2 (argparse), the code of INCONCLUSIVE; version 2 exits 3")
    for label, script, expected, extra in (("v1", V1_RAW, 2, []), ("v2", V2_RAW, 3, ["--db", fx.db])):
        fx.reset({"mode": "normal"})
        run(fx, "U3", label, [script, "--port", fx.port, *extra, "--expect", "observe", "--wait1", "abc", "--wait2", "10", "responses:eof"], expected, env_extra={"HOME": str(fx.home)} if label == "v1" else None, defect=(label == "v1"))
    print("### U4: no gateway answers on the port: version 2 stops with exit 3 at its health preflight, before any request")
    fx.reset({"mode": "normal"})
    run(fx, "U4", "v2", [V2_RAW, "--port", "1", "--db", fx.db, "--expect", "observe", *RAW_WAITS, "responses:eof"], 3)
    print("### U5: native: an unknown shape (version 1 raises a KeyError and exits 1, version 2 exits 3), and a missing profile template (version 2 exits 3)")
    fx.reset({"mode": "normal"})
    run(fx, "U5", "v1", [V1_NAT, "--expect", "all-logged", "--wait1", "5", "--wait2", "10", "--quiet-timeout", "1", "--repo", fx.repo, "nonsense"], 1, env_extra={"HOME": str(fx.home)}, defect=True)
    run(fx, "U5", "v2 unknown shape", [V2_NAT, *NAT_V2, "--repo", fx.repo, "--db", fx.db, "--scratch", fx.work / "codex-home-U5", "nonsense"], 3)
    run(fx, "U5", "v2 missing template", [V2_NAT, *NAT_V2, "--repo", fx.work / "no-such-repo", "--db", fx.db, "--scratch", fx.work / "codex-home-U5b", "astra-max"], 3)


def identity_controls(fx):
    tools, _ = make_tools(fx.work)
    proc = dummy(tools)
    time.sleep(0.5)
    try:
        print("### I1: the identity read-back of a dummy process that carries the rollback prefix in its command line, against the stub's health endpoint: it prints the line and exits 0 when the expectations hold")
        base = [IDENT, "--pid", proc.pid, "--port", fx.port, "--tools", tools]
        run(fx, "I1", "expectations hold", [*base, "--label", "control", "--expect-prefix", "omniroute-ctl-rollback", "--expect-build", "cf6748d04"], 0)
        print("### I2: a build id that is not the running one: exit 5")
        run(fx, "I2", "wrong build", [*base, "--label", "control", "--expect-prefix", "omniroute-ctl-rollback", "--expect-build", "2a41147b7"], 5)
        print("### I3: no healthy gateway on the port: exit 5")
        run(fx, "I3", "no health", [IDENT, "--pid", proc.pid, "--port", "1", "--tools", tools, "--label", "control"], 5)
    finally:
        proc.kill()

    if want("G13"):
        print("### G13: a connection of the TOLERATED unit that carries a request (a slow request in pieces, as inference forwarded through the 20129 gateway would): version 3 exempted every connection of the unit and exited 0, version 4 judges it ACTIVE (exit 4)", flush=True)
        scope = f"ctl-poller-req-{os.getpid()}"
        fx.reset({"mode": "normal"})
        poller = subprocess.Popen(["systemd-run", "--user", "--scope", "--quiet", f"--unit={scope}", sys.executable, str(HERE / "poller.py"), "--port", str(fx.port), "--duration", "120", "--mode", "request", "--hold", "40"])
        time.sleep(1.5)
        for script, label, expected in ((GUARD_V3, "v3", 0), (GUARD, "v4", 4)):
            run(fx, "G13", label, [script, *common[1:], "--tolerate-unit", f"{scope}.scope"], expected, defect=(script == GUARD_V3))
        poller.terminate()
        try:
            poller.wait(timeout=5)
        except Exception:
            poller.kill()
    sub = lambda name: fx.work / f"guard-shim-{name}"
    for name, body in (("ss", "#!/bin/sh\necho 'ss: stub failure' >&2\nexit 2\n"), ("pgrep", "#!/bin/sh\necho 'pgrep: stub failure' >&2\nexit 2\n"), ("pgrep-nomatch", "#!/bin/sh\nexit 1\n")):
        d = sub(name)
        d.mkdir(exist_ok=True)
        f = d / ("pgrep" if name.startswith("pgrep") else "ss")
        f.write_text(body)
        f.chmod(0o755)
    if want("G14"):
        print("### G14: `ss` fails (exit 2, empty stdout): version 3 read the empty output as an idle host and exited 0, version 4 FAILS the check (exit 4)", flush=True)
        fx.reset({"mode": "normal"})
        env = {"PATH": f"{sub('ss')}:{os.environ['PATH']}"}
        for script, label, expected in ((GUARD_V3, "v3", 0), (GUARD, "v4", 4)):
            run(fx, "G14", label, [script, *common[1:]], expected, env_extra=env, defect=(script == GUARD_V3))
    if want("G15"):
        print("### G15: `pgrep` fails (exit 2, empty stdout): version 3 read it as no worker present and exited 0, version 4 FAILS the check (exit 4)", flush=True)
        fx.reset({"mode": "normal"})
        env = {"PATH": f"{sub('pgrep')}:{os.environ['PATH']}"}
        for script, label, expected in ((GUARD_V3, "v3", 0), (GUARD, "v4", 4)):
            run(fx, "G15", label, [script, *common[1:]], expected, env_extra=env, defect=(script == GUARD_V3))
    if want("G16"):
        print("### G16: `pgrep` exits 1 (no process matched, an idle host) and prints nothing: that is NOT a failure, version 4 exits 0", flush=True)
        fx.reset({"mode": "normal"})
        run(fx, "G16", "v4", [GUARD, *common[1:]], 0, env_extra={"PATH": f"{sub('pgrep-nomatch')}:{os.environ['PATH']}"})
