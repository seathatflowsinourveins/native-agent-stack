#!/bin/bash
# Validation of the post-terminal client-disconnect drain patch in a scratch tree built from public refs (nothing is deployed). Version 5 (2026-10-06): the blob id parsed from the patch's index line must be a hex id (a vacuous prefix match is no longer possible).
# Patch under test: $DRAIN_DIR/drain-after-terminal.patch (one commit: the streamHandler.ts change plus tests/unit/stream-terminal-seen-client-disconnect.test.ts).
# The unit test file is EXTRACTED from that patch (git apply --include) for the negative controls, and its git blob id is checked against the patch's own index line: no standalone copy exists to drift.
# Reference implementation: drainCompletedToolHandoff in open-sse/utils/streamHandler.ts at upstream 0585aba5589d5a1f49243a13a8db249558e7c9e3 (L665-688), reused from cancel() for the terminal-seen case.
# Steps (each logs its exit code to $OUT/progress.txt; a failed setup step, a tree-id mismatch or a violated expectation stops the script with exit 3):
#   composition C (v3.8.51 tag + affinity patch + PR 15167; its tree id is checked), npm ci, then
#   A   baseline: the stream-related test files (name globs plus every unit test that mentions streamHandler), unchanged, on the unpatched tree
#   N   the new test file (extracted from the patch) on the unpatched tree (negative control: exactly three of its six tests fail, by name)
#   N2  the new test file with only the streamHandler.ts hunk of the first version of the patch, superseded after review (exactly two fail, by name: the tool-handoff drain that never starts, and the pending request that stays)
#   P   the patch under test applied: its tree id must equal EXPECT_TREE, or the script stops here, before any patched test stage; then the new test file alone (six of six pass)
#   A2  the same stream-related files plus the new test file on the patched tree; B2 the call-log, usage-history and chat-pipeline tests
# Frozen expectations (constants below): test counts and failing test names per stage. The one failure of the 239 upstream files is the same unrelated test before and after.
# For every stage the runner's summary counters (tests, suites, pass, fail, cancelled, skipped, todo), the failing tests' names and the full log are kept in $OUT; the file lists are in $OUT/test-file-list.txt.
# The script owns only the directory it creates (mktemp under $WORK_ROOT, default ~/.cache) and deletes nothing but its own temporary copy of the extracted test file, so it cannot touch another session's directories.
# A missing input file or mirror stops it with exit 2 before anything is created.
# usage: DRAIN_DIR=<directory with drain-after-terminal.patch and superseded-first-version.patch> EV=<evidence folder of omniroute-wire-effort-20261005> EXPECT_TREE=<tree id of the patched tree> omniroute-drain-prototype-20261005.sh
set -u
set -o pipefail
M=${MIRROR:-$HOME/.cache/omniroute-upstream-mirror-20261005.git}
D=${DRAIN_DIR:?set DRAIN_DIR to the directory holding drain-after-terminal.patch and superseded-first-version.patch}
TAG=c1e30b7676975feb298b49eff6ff58923c04b89e
PR=0585aba5589d5a1f49243a13a8db249558e7c9e3
COMPOSITION_TREE=f1336dfd6c8ebd81586dd6c71e7ef668ef8c949f
EV=${EV:?set EV to the evidence folder of the gateway-composition record}
EXPECT_TREE=${EXPECT_TREE:?set EXPECT_TREE to the tree id of the composition with the patch applied}
UNRELATED_FAILURE="repository provider asset manifest covers the audited 141-file snapshot"
T1="a client that closes right after response.completed still lets the stream finish: completion and usage bookkeeping run"
T3="the drain is bounded: an upstream that never ends is cancelled when the grace period expires, and the pending request is cleared"
T6="a completed tool handoff whose terminal event was already delivered has a bounded drain, and the pending request is cleared"
export PATH="$HOME/.local/share/codex-ecosystem/tools/node-24.21.0/bin:/usr/local/bin:/usr/bin:/bin"
for f in "$D/drain-after-terminal.patch" "$D/superseded-first-version.patch" "$EV/patches/045aa81f3.patch"; do
  [ -f "$f" ] || { echo "missing input file: $f" >&2; exit 2; }
done
[ -d "$M" ] || { echo "missing OmniRoute mirror: $M" >&2; exit 2; }
WORK=$(mktemp -d "${WORK_ROOT:-$HOME/.cache}/omniroute-drain-validate-XXXXXX") || exit 2
T=$WORK/tree
OUT=$WORK/out
mkdir -p "$OUT"
log() { echo "$(date -u +%H:%M:%SZ) $*" | tee -a "$OUT/progress.txt"; }
G() { git -c user.name=Scout -c user.email=scout@local -c core.hooksPath=/dev/null -C "$T" "$@"; }
step() { local label=$1; shift; "$@"; local rc=$?; log "$label exit=$rc"; [ $rc -eq 0 ] || exit 3; }
apply_pr_diff() { git --git-dir="$M" diff $PR^ $PR | G apply --index; }
counter() { grep -aE "^ℹ $2 " "$OUT/test-$1.log" | awk '{print $3}'; }
failing() { awk '/^✖ failing tests:/{f=1;next} f&&/^✖ /{sub(/ \([0-9.]+ms\)$/,""); sub(/^✖ /,""); print}' "$OUT/test-$1.log" | sort -u; }
expect_counts() { # label tests pass fail
  local t p f; t=$(counter "$1" tests); p=$(counter "$1" pass); f=$(counter "$1" fail)
  if [ "$t" = "$2" ] && [ "$p" = "$3" ] && [ "$f" = "$4" ]; then log "$1 expectation met: tests=$t pass=$p fail=$f"; else log "$1 EXPECTATION VIOLATED: tests=$t pass=$p fail=$f, expected tests=$2 pass=$3 fail=$4: stopping"; exit 3; fi
}
expect_failing() { # label, the expected failing test names, one per line (may be empty)
  local got; got=$(failing "$1")
  if [ "$got" = "$2" ]; then log "$1 failing tests are as expected: $(printf '%s' "$got" | grep -c . | tr -d ' ') name(s)"; else log "$1 FAILING SET VIOLATED: got [$got]: stopping"; exit 3; fi
}
step clone git clone -q --shared --no-checkout "$M" "$T"
step "fetch pr15167" G fetch -q "$M" "+refs/pull/15167/head:refs/pull/15167/head"
step "checkout tag" G checkout -q --detach $TAG
( cd "$T" && nice -n 19 npm ci --no-audit --no-fund > "$OUT/npm-ci.log" 2>&1 ); rc=$?; log "npm ci exit=$rc"; [ $rc -eq 0 ] || exit 3
step "apply affinity patch" G apply --index "$EV/patches/045aa81f3.patch"
step "apply pr15167 diff" apply_pr_diff
step "commit composition" G commit -q -m "composition"
ACTUAL_TREE=$(G rev-parse 'HEAD^{tree}')
if [ "$ACTUAL_TREE" = "$COMPOSITION_TREE" ]; then log "composition tree=$ACTUAL_TREE equals the expected tree"; else log "composition tree=$ACTUAL_TREE DIFFERS from the expected tree $COMPOSITION_TREE: stopping"; exit 3; fi
FLAGS="--import tsx/esm --import ./open-sse/utils/setupPolyfill.ts --import ./tests/_setup/isolateDataDir.ts"
run() {
  local label=$1; shift
  ( cd "$T" && DISABLE_SQLITE_AUTO_BACKUP=true nice -n 19 node $FLAGS --test --test-force-exit --test-concurrency=4 "$@" > "$OUT/test-$label.log" 2>&1 ); local rc=$?
  log "$label exit=$rc tests=$(counter "$label" tests) suites=$(counter "$label" suites) pass=$(counter "$label" pass) fail=$(counter "$label" fail) cancelled=$(counter "$label" cancelled) skipped=$(counter "$label" skipped) todo=$(counter "$label" todo)"
}
cd "$T" || exit 2
NEWTEST=tests/unit/stream-terminal-seen-client-disconnect.test.ts
STREAM_TESTS=$( { ls tests/unit/*stream*.test.ts tests/unit/*sse*.test.ts tests/unit/codex-tool-handoff-disconnect-499.test.ts tests/unit/empty-stream-no-content-8649.test.ts 2>/dev/null; grep -l streamHandler tests/unit/*.test.ts; } | sort -u | tr '\n' ' ')
CALLUSE_TESTS="tests/unit/call-log-artifact-worker.test.ts tests/unit/call-log-save-drain.test.ts tests/unit/call-log-startup.test.ts tests/unit/call-log-in-memory-usage.test.ts tests/unit/call-log-cap.test.ts tests/unit/call-log-oom-unbounded-5618.test.ts tests/unit/call-log-size-limit-error.test.ts tests/unit/call-log-artifact-bodies-first.test.ts tests/unit/call-log-detailed-tokens.test.ts tests/unit/call-log-rotate-corrupt.test.ts tests/unit/usage-history-db.test.ts tests/unit/usage/usageHistoryDedup.test.ts tests/unit/usage-history-reset.test.ts tests/unit/usagehistory-helpers-split.test.ts tests/integration/chat-pipeline.test.ts"
{ echo "# stream-related upstream test files ($(echo $STREAM_TESTS | wc -w)): the name globs *stream*, *sse*, the Codex tool-handoff and empty-stream files, and every unit test file that mentions streamHandler"; for f in $STREAM_TESTS; do echo "$f"; done; echo; echo "# call-log, usage-history and chat-pipeline files ($(echo $CALLUSE_TESTS | wc -w))"; for f in $CALLUSE_TESTS; do echo "$f"; done; } > "$OUT/test-file-list.txt"
log "stream-related test files: $(echo $STREAM_TESTS | wc -w)"
run A-baseline $STREAM_TESTS
expect_counts A-baseline 1746 1745 1
expect_failing A-baseline "$UNRELATED_FAILURE"
PATCH_BLOB=$(awk '/^diff --git a\/tests\/unit\/stream-terminal-seen-client-disconnect.test.ts/{f=1} f&&/^index /{print $2; exit}' "$D/drain-after-terminal.patch" | sed 's/.*\.\.//')
# version 5: the prefix check below passes vacuously for an empty PATCH_BLOB, so the parsed id must be a hex blob id of 7 to 40 digits (a missing or reformatted index line stops the script with exit 3)
printf '%s' "$PATCH_BLOB" | grep -Eq '^[0-9a-f]{7,40}$' || { log "the patch's index line for the new test file did not parse to a blob id (got '$PATCH_BLOB'): stopping"; exit 3; }
step "extract the new test file from the patch" G apply --include=$NEWTEST "$D/drain-after-terminal.patch"
ACTUAL_BLOB=$(G hash-object $NEWTEST)
case "$ACTUAL_BLOB" in "$PATCH_BLOB"*) log "new test file blob $ACTUAL_BLOB matches the patch's index line ($PATCH_BLOB)";; *) log "new test file blob $ACTUAL_BLOB does not match the patch's index line ($PATCH_BLOB): stopping"; exit 3;; esac
run N-new-test-unpatched $NEWTEST
expect_counts N-new-test-unpatched 6 3 3
expect_failing N-new-test-unpatched "$(printf '%s\n%s\n%s' "$T1" "$T3" "$T6" | sort -u)"
step "apply the first version's streamHandler.ts hunk" G apply --include=open-sse/utils/streamHandler.ts "$D/superseded-first-version.patch"
run N2-new-test-first-version $NEWTEST
expect_counts N2-new-test-first-version 6 4 2
expect_failing N2-new-test-first-version "$(printf '%s\n%s' "$T3" "$T6" | sort -u)"
step "restore streamHandler.ts" G checkout -q -- open-sse/utils/streamHandler.ts
rm -f $NEWTEST
step "apply patch under test" G apply --index "$D/drain-after-terminal.patch"
step "commit patch under test" G commit -q -m "drain patch under test"
PATCHED_TREE=$(G rev-parse 'HEAD^{tree}')
if [ "$PATCHED_TREE" = "$EXPECT_TREE" ]; then log "patched tree=$PATCHED_TREE equals the expected tree"; else log "patched tree=$PATCHED_TREE DIFFERS from the expected tree $EXPECT_TREE: stopping before the patched test stages"; exit 3; fi
run P-new-test-patched $NEWTEST
expect_counts P-new-test-patched 6 6 0
expect_failing P-new-test-patched ""
run A2-stream-tests-patched $STREAM_TESTS $NEWTEST
expect_counts A2-stream-tests-patched 1752 1751 1
expect_failing A2-stream-tests-patched "$UNRELATED_FAILURE"
run B2-calllog-usage-chat-pipeline $CALLUSE_TESTS
expect_counts B2-calllog-usage-chat-pipeline 126 126 0
expect_failing B2-calllog-usage-chat-pipeline ""
log "DONE; the scratch tree and the logs stay in $WORK (remove it with rm -rf after reading)"
