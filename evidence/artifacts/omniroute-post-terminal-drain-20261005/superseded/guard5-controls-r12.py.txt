def load(path):
    """The guard module's functions without running its main(): everything above the trailing `try: main()` block."""
    text = Path(path).read_text(encoding="utf-8")
    head = text.split("\ntry:\n    main()")[0]
    ns = {"__file__": str(path), "__name__": "guard_under_test"}
    exec(compile(head, str(path), "exec"), ns)
    return ns

def info(tx=0, rx=0):
    return REAL_INFO_IDLE.format(tx=f" bytes_sent:{tx}" if tx else "", rx=f" bytes_received:{rx}" if rx else "")


def rec(side, cip, cport, sip="127.0.0.1", sport=20128, recvq=0, sendq=0, rx=0, tx=0, owner=None, tcp_info=True, info_text=None):
    """A record of the connection (cip:cport -> sip:sport): the SERVER end (local = server) or the CLIENT end (local = client), as ss prints it. rx/tx are the END's own bytes_received and bytes_sent."""
    local, peer = (f"{sip}:{sport}", f"{cip}:{cport}") if side == "server" else (f"{cip}:{cport}", f"{sip}:{sport}")
    users = f' users:(("{owner[0]}",pid={owner[1]},fd={owner[2]}))' if owner else ""
    text = info_text if info_text is not None else (info(tx, rx) if tcp_info else "")
    return f"{recvq:<6} {sendq:<6} {local} {peer}{users} {text}".rstrip()


def to_two_line(lines):
    """The same records in guard 4's format (`ss -Htnpi`: an address line, then the info line indented by a tab)."""
    out = []
    for line in lines:
        m = re.match(r"^(\d+\s+\d+\s+\S+\s+\S+)(\s+users:\(\(.*?\)\))?\s*(.*)$", line)
        out += [m.group(1) + (m.group(2) or ""), "\t " + m.group(3)]
    return out


LISTEN_OK = "LISTEN 0      511    127.0.0.1:20128 0.0.0.0:*"


def fake_sh(listen_lines, record_lines, two_line=False, ss_rc=0, listen_rc=0):
    def sh(*a):
        if a[0] == "ss" and a[1] == "-Htln":
            return subprocess.CompletedProcess(a, listen_rc, "\n".join(listen_lines) + ("\n" if listen_lines else ""), "ss: fixture failure" if listen_rc else "")
        if a[0] == "ss":
            lines = to_two_line(record_lines) if two_line else record_lines
            return subprocess.CompletedProcess(a, ss_rc, "\n".join(lines) + ("\n" if lines else ""), "ss: fixture failure" if ss_rc else "")
        raise AssertionError(f"unexpected command {a}")
    return sh


def run_guard(ns, records, listen=(LISTEN_OK,), two_line=False, **kw):
    ns["sh"] = fake_sh(list(listen), records, two_line=two_line, **kw)
    try:
        c = ns["connections"](20128)
        return "OK", {k: (v["counters"], v["owners"]) for k, v in c.items()}, getattr(c, "unrelated", None)
    except ns["ObservationError"] as e:
        return "ObservationError", str(e)[:150], None
    except Exception as e:  # a guard that crashes is not an observation either
        return "CRASH", f"{type(e).__name__}: {str(e)[:100]}", None


def show(case, title, expect5, records, expect4=None, **kw):
    """Run guard 5 on the records (and guard 4 on the same records in its format when expect4 is given); expect5 / expect4 are 'OK' or 'ObservationError' or a predicate on the result."""
    print(f"### {case}: {title}")
    kind5, res5, unrel = run_guard(G5, records, **kw)
    ok5 = (kind5 == expect5) if isinstance(expect5, str) else bool(expect5(kind5, res5, unrel))
    print(f"    guard 5: {kind5} {res5}" + (f" (unrelated records: {unrel})" if unrel else "") + f"   [{'as expected' if ok5 else 'UNEXPECTED'}]")
    if not ok5:
        UNEXPECTED.append(case)
    if expect4 is not None:
        kind4, res4, _ = run_guard(G4, records, two_line=True, **{k: v for k, v in kw.items() if k != "listen"})
        ok4 = expect4(kind4, res4) if callable(expect4) else (kind4 == expect4)
        print(f"    guard 4 (the defect): {kind4} {res4}   [{'as expected' if ok4 else 'UNEXPECTED'}]")
        if not ok4:
            UNEXPECTED.append(case + "/v4")
    return kind5, res5

def unit_segment():
    for line in open("/proc/self/cgroup", encoding="utf-8"):
        if line.startswith("0::"):
            return line.strip()[3:].rsplit("/", 1)[-1]
    return ""


MY_UNIT = unit_segment()
MINE = ("bash", os.getppid(), 3)   # an owner pid whose cgroup is this session's own unit (visible here), used as the tolerated unit's process


def guard_e(case, title, records, expect_exit, expect_text, extra=()):
    rf = work / f"records-{case}.txt"
    rf.write_text("\n".join(records) + ("\n" if records else ""))
    env = dict(os.environ, PATH=f"{shim_dir}:{os.environ['PATH']}", SS_LISTEN=str(listen_file), SS_RECORDS=str(rf))
    cmd = [sys.executable, args.guard5, "--idle-only", "--identity-script", str(ident), "--db", str(db), "--worker-pattern", "nonexistent-process-pattern-for-the-controls", "--clean-for", "2", "--interval", "1", "--tolerate-unit", MY_UNIT, *extra]
    p = subprocess.run(cmd, capture_output=True, text=True, env=env, timeout=120)
    out = p.stdout
    last = [l for l in out.splitlines() if l.startswith(("IDLE-RESULT", "GUARD-RESULT"))][-1:]
    fails = [l.strip()[:200] for l in out.splitlines() if "the socket observation failed" in l or "ACTIVE connection" in l][:1]
    ok = p.returncode == expect_exit and (expect_text in out or not expect_text)
    print(f"### {case}: {title}\n    exit={p.returncode} {last[0] if last else p.stderr.strip()[:160]}" + (f"\n    {fails[0]}" if fails else "") + f"   [{'as expected' if ok else 'UNEXPECTED'}]")
    if not ok:
        UNEXPECTED.append(case)
