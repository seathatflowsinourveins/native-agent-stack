def utc(ts):
    return ts.astimezone(dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def now_utc():
    """The host clock: the reference of the arms' own stamps and of the guard's (importable so that a control can replace it)."""
    return dt.datetime.now(dt.timezone.utc)


def shift(ts, seconds):
    base = dt.datetime.strptime(ts, "%Y-%m-%dT%H:%M:%S.%fZ").replace(tzinfo=dt.timezone.utc)
    return utc(base + dt.timedelta(seconds=seconds))


def id8(value):
    return hashlib.sha256(str(value).encode()).hexdigest()[:8]


def parse_ts(text):
    """The UTC datetime of `2026-10-06T11:50:17Z` or `2026-10-06T11:50:17.123Z`; None for anything else."""
    if not isinstance(text, str) or not text.isascii():
        return None
    m = re.fullmatch(r"([0-9]{4}-[0-9][0-9]-[0-9][0-9]T[0-9][0-9]:[0-9][0-9]:[0-9][0-9])(?:\.([0-9]{1,6}))?Z", text)
    if not m:
        return None
    try:
        base = dt.datetime.strptime(m.group(1), "%Y-%m-%dT%H:%M:%S").replace(tzinfo=dt.timezone.utc)
    except ValueError:
        return None
    if not 2000 <= base.year <= 2100:   # a stamp that far out is not a stamp of this run (and would overflow the window arithmetic)
        return None
    return base + dt.timedelta(microseconds=int((m.group(2) or "0").ljust(6, "0")))


def valid_gateway_ts(value):
    """True for a text stamp in the gateway's exact format (YYYY-MM-DDTHH:MM:SS.mmmZ) whose calendar values are valid and whose year is 2000-2100."""
    return isinstance(value, str) and value.isascii() and GATEWAY_TS.fullmatch(value) is not None and parse_ts(value) is not None


def numbers_problem(duration, ttft):
    """None when a duration (a call row's) or latency (a usage row's) and a time to first token are plausible: each may be NULL, a present one is a finite non-negative number and the time to first token is not longer than the duration by more than 1 ms
    (measured on this host's store: 30,129 usage rows and 24,522 call rows, no exception); otherwise the reason. The scripts' own `associated()` skips a NULL time to first token and lets a NaN pass the latency comparison; this gate does not."""
    values = []
    for name, v in (("latency", duration), ("time to first token", ttft)):
        if v is None:
            values.append(None)
            continue
        try:
            x = float(v)
        except (TypeError, ValueError):
            return f"{name} {v!r} is not a number"
        if not math.isfinite(x) or x < 0:
            return f"{name} {v!r} is not a finite non-negative number"
        values.append(x)
    if values[0] is not None and values[1] is not None and values[1] > values[0] + 1.0:
        return f"time to first token {ttft!r} is longer than the latency {duration!r}"
    return None


def journal_posts(unit, since, until):
    """Every inbound request line (`tag=HTTP method=POST`, message `📥 POST ...`) of the gateway unit's journal between `since` and `until`: a list of (datetime, label). The gateway logs one such line when a request ARRIVES, whatever happens to its call and usage rows
    afterwards. Raises RuntimeError when the journal cannot be run, exits non-zero or yields output that cannot be parsed (the check then fails closed)."""
    start = since.replace(microsecond=0)
    stop = until.replace(microsecond=0) + dt.timedelta(seconds=1)
    cmd = ["journalctl", "--user", "-u", unit, "--since", start.strftime("%Y-%m-%d %H:%M:%S") + " UTC", "--until", stop.strftime("%Y-%m-%d %H:%M:%S") + " UTC", "-o", "json", "--no-pager"]
    try:
        p = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
    except (OSError, subprocess.TimeoutExpired) as error:
        raise RuntimeError(f"journalctl could not be run: {type(error).__name__}")
    if p.returncode != 0:
        raise RuntimeError(f"journalctl exited with status {p.returncode}")
    found, unreadable = [], 0
    for line in p.stdout.splitlines():
        if not line.strip():
            continue
        try:
            entry = json.loads(line)
            stamp = dt.datetime.fromtimestamp(int(entry["__REALTIME_TIMESTAMP"]) / 1e6, dt.timezone.utc)
        except (ValueError, KeyError, TypeError, OverflowError, OSError):
            unreadable += 1
            continue
        message = entry.get("MESSAGE")
        if isinstance(message, list):   # journald writes a message that holds control characters as an array of byte values
            try:
                message = bytes(message).decode("utf-8", "replace")
            except (ValueError, TypeError):
                continue
        if not isinstance(message, str) or not message.startswith("{"):
            continue
        try:
            record = json.loads(message)
        except ValueError:
            continue
        if isinstance(record, dict) and record.get("tag") == "HTTP" and record.get("method") == "POST" and str(record.get("msg", "")).startswith("\U0001F4E5 POST"):
            found.append((stamp, str(record.get("msg", ""))[:90]))
    if unreadable:
        raise RuntimeError(f"{unreadable} journal line(s) could not be parsed")
    return found


def associated(c, u):
    """True when the usage_history row u carries the same request's identity as the call_logs row c: provider, account (connection_id and api_key_id), endpoint (the call's path), latency and time to first token agree."""
    if u["provider"] != c["provider"] or u["connection_id"] != c["connection_id"] or u["api_key_id"] != c["api_key_id"] or u["endpoint"] != c["path"]:
        return False
    if c["duration"] is None or u["latency_ms"] is None or abs(float(u["latency_ms"]) - float(c["duration"])) > LATENCY_TOLERANCE_MS:
        return False
    if c["ttft_ms"] is not None and u["ttft_ms"] is not None and abs(float(u["ttft_ms"]) - float(c["ttft_ms"])) > TTFT_TOLERANCE_MS:
        return False
    return True


# ---------------------------------------------------------------- matching (importable; cross-checked in the controls)
def maximum_matching(adj, calls, skip=None):
    """Augmenting-path (Kuhn) maximum matching of the bipartite graph adj ({call id: [usage ids]}) over `calls`: {usage id: call id}. `skip` is one edge (call id, usage id) to leave out."""
    match = {}

    def augment(call, seen):
        for usage in adj[call]:
            if (call, usage) == skip or usage in seen:
                continue
            seen.add(usage)
            if usage not in match or augment(match[usage], seen):
                match[usage] = call
                return True
        return False

    for call in calls:
        augment(call, set())
    return match


def matching_status(adj, calls, usages):
    """(status, pairs) of one connected component: 'unequal' (another number of usage rows than call rows), 'none' (no perfect matching), 'multiple' (more than one) or 'unique' (exactly one; pairs holds it).
    A perfect matching M is the only one exactly when removing any single edge of M leaves no perfect matching."""
    if len(calls) != len(usages):
        return "unequal", []
    match = maximum_matching(adj, calls)
    if len(match) != len(calls):
        return "none", []
    pairs = sorted((call, usage) for usage, call in match.items())
    for call, usage in pairs:
        if len(maximum_matching(adj, calls, skip=(call, usage))) == len(calls):
            return "multiple", pairs
    return "unique", pairs


def components(edges):
    """Connected components of the bipartite graph over the call ids of `edges` ({call id: [usage ids]}) and the usage ids that have an edge: a list of (sorted call ids, sorted usage ids)."""
    parent = {}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    for call, usages in edges.items():
        parent.setdefault(("c", call), ("c", call))
        for usage in usages:
            parent.setdefault(("u", usage), ("u", usage))
            ra, rb = find(("c", call)), find(("u", usage))
            if ra != rb:
                parent[ra] = rb
    groups = {}
    for node in parent:
        groups.setdefault(find(node), []).append(node)
    out = [(sorted(i for k, i in nodes if k == "c"), sorted(i for k, i in nodes if k == "u")) for nodes in groups.values()]
    return sorted(out, key=lambda comp: (comp[0], comp[1]))

def parse_arm(path):
    """One arm's private output -> a dict: name, kind (raw or native), version, script sha, interval (start, end, source), db, requests, journal, identities, run statuses, problems."""
    arm = {"name": os.path.basename(path), "kind": None, "version": None, "script_sha": None, "db": None, "db_resolved": None, "cwd": None, "requests": [], "journal": "missing", "problems": [], "source": "", "run_status": None, "run_statuses": [],
           "identities": [], "pid": None, "unit_started": None, "wrapper_kind": None, "header_n": 0, "finished_n": 0}
    wrapper_start = wrapper_end = script_start = script_end = None
    summary_seen = False
    counts, void_seen = collections.Counter(), False
    for line in read_text(path, "arm output").splitlines():
        m = STRICT_HEADER.match(line)
        if m:
            counts["header"] += 1
        if m and wrapper_start is None:
            arm["wrapper_kind"] = m.group(2)
            wrapper_start = parse_ts(m.group(1))
            if wrapper_start is None:
                arm["problems"].append(f"unparsable wrapper start stamp {m.group(1)!r}")
            continue
        if not m and "wrapper kind=" in line:
            arm["problems"].append(f"a wrapper header line the census cannot parse (its kind and start stamp would be missed): {line[:70]!r}")
            continue
        m = re.match(r"^# finished (\S+)\s*$", line)
        if m:
            counts["finished"] += 1
            wrapper_end = parse_ts(m.group(1))
            if wrapper_end is None:
                arm["problems"].append(f"unparsable wrapper finish stamp {m.group(1)!r}")
            continue
        if LOOSE_FINISHED.match(ANSI.sub("", line)):
            arm["problems"].append(f"a `# finished` line the census cannot parse (the arm's end stamp would be missed): {line[:70]!r}")
            continue
        if line.startswith("FROZEN "):
            counts["frozen"] += 1
            if counts["frozen"] == 1:
                arm["version"] = (re.search(r"\bversion=(\S+)", line) or [None, None])[1]
                arm["script_sha"] = (re.search(r"\bscript_sha256=([0-9a-f]{64})\b", line) or [None, None])[1]
                start_token = (re.search(r"\bstart=(\S+)", line) or [None, None])[1]
                script_start = parse_ts(start_token)
                if start_token is None:
                    arm["problems"].append("the FROZEN line has no start= value (missing or empty): another stamp must not stand in for it")
                elif script_start is None:
                    arm["problems"].append(f"unparsable FROZEN start= {start_token!r}: another stamp must not stand in for it")
                arm["db"] = (re.search(r"\bdb=(.+?)(?= effort_close_after=| run_id=| baseline=| wait1=| wait2=|$)", line) or [None, None])[1]   # the path may contain spaces
                arm["cwd"] = (re.search(r"\bcwd=(.+?) start=", line) or [None, None])[1]
            continue
        m = re.match(r"^RESULT mode=(\S+) http=(\S+) .*? corr=(\S+) response_id=(\S+) read1_call_logs=", line)
        if m:
            arm["requests"].append({"kind": "raw", "mode": m.group(1), "http": m.group(2), "corr": m.group(3), "rid": m.group(4)})
            continue
        m = re.match(r"^RESULT shape=(\S+) session_id=(\S+) .*? min_rows=([0-9]{1,6}) exec_exit=(\S+)", line)
        if m:
            arm["requests"].append({"kind": "native", "shape": m.group(1), "tag": m.group(2), "min_rows": int(m.group(3)), "exec_exit": m.group(4), "judged": " read1_call_logs=" in line})
            continue
        m = re.match(r"^SUMMARY .* end=(\S+) exit=([0-9]{1,6})\s*$", line)
        if m:
            counts["summary"] += 1
            summary_seen = True
            script_end = parse_ts(m.group(1))
            if script_end is None:
                arm["problems"].append(f"unparsable SUMMARY end= {m.group(1)!r}: another stamp must not stand in for it")
            continue
        m = re.match(r"^journal_failed_to_save_call_log=(\S+)", line)
        if m:
            counts["journal"] += 1
            arm["journal"] = m.group(1)
            continue
        if line.startswith("IDENTITY "):
            pid_m, started_m = re.search(r"\bMainPID=([0-9]{1,12})", line), re.search(r"\bExecMainStartTimestamp=(\S+)", line)
            if pid_m and started_m:
                arm["identities"].append((pid_m.group(1), started_m.group(1)))
            else:
                arm["problems"].append("a wrapper identity line has no MainPID or no ExecMainStartTimestamp: the process cannot be compared with the idle receipt")
            continue
        m = re.match(r"^RUN-STATUS \S+ status=([0-9]+)\b", line)
        if m:
            arm["run_statuses"].append(int(m.group(1)))
            arm["run_status"] = int(m.group(1))
            if arm["run_status"] == 4:
                void_seen = True
            continue
        if re.match(r"^\s*IDENTITY-PROBLEM", line):
            arm["problems"].append(f"the wrapper reported an identity problem: {line[:70]!r}")
            continue
        loose = LOOSE_RESERVED.match(re.sub(r"^[^A-Za-z0-9#]+", "", ANSI.sub("", line)))   # a reserved word after a BOM, a zero-width space, a control character or any other prefix, in any letter case
        if loose:
            word = loose.group(1).upper()
            if word == "RESULT":
                arm["problems"].append(f"a RESULT line the census cannot parse (its request would go unattributed): {line[:70]!r}")
            elif word == "SUMMARY":
                arm["problems"].append(f"a SUMMARY line the census cannot parse (its end stamp would be missed): {line[:70]!r}")
            else:
                arm["problems"].append(f"a reserved record the census cannot parse (it would be skipped): {line[:70]!r}")
    arm["header_n"], arm["finished_n"] = counts["header"], counts["finished"]
    if arm["identities"]:
        arm["pid"], arm["unit_started"] = arm["identities"][0]
    if arm["version"] not in FROZEN_SCRIPT_SHA:
        arm["problems"].append(f"no FROZEN line of a frozen script (version {arm['version']!r}): not an output of raw script 2.2 or native script 2.1")
    else:
        arm["kind"] = "raw" if arm["version"] == "2.2" else "native"
        if arm["script_sha"] != FROZEN_SCRIPT_SHA[arm["version"]]:
            arm["problems"].append(f"the script of version {arm['version']} had sha256 {str(arm['script_sha'])[:12]}, not the frozen {FROZEN_SCRIPT_SHA[arm['version']][:12]}")
    if not summary_seen:
        arm["problems"].append("no SUMMARY line: a run without one is not a verdict")
    if void_seen or arm["run_status"] == 4:
        arm["problems"].append("the wrapper marked this run VOID (RUN-STATUS status=4)")
    for key, label in (("frozen", "FROZEN"), ("summary", "SUMMARY"), ("journal", "journal_failed_to_save_call_log"), ("header", "wrapper header"), ("finished", "`# finished`")):
        if counts[key] > 1:
            arm["problems"].append(f"{counts[key]} {label} lines: one run per file is expected (a hand-concatenated output is refused)")
    if script_start is None and wrapper_start is None:
        arm["problems"].append("no start time (neither FROZEN start= nor the wrapper's started stamp)")
    if script_end is None and wrapper_end is None:
        arm["problems"].append("no end time (neither SUMMARY end= nor the wrapper's # finished stamp)")
    # the script's own stamps and the wrapper's stamps must agree (the wrapper's are whole seconds, truncated)
    one_s = dt.timedelta(seconds=STAMP_RESOLUTION_S)
    if script_start is not None and script_end is not None and script_end < script_start:
        arm["problems"].append(f"the script's SUMMARY end {utc(script_end)} is before its FROZEN start {utc(script_start)}")
    if wrapper_start is not None and script_start is not None and script_start < wrapper_start - one_s:
        arm["problems"].append(f"the script started ({utc(script_start)}) before the wrapper's started stamp ({utc(wrapper_start)})")
    if wrapper_end is not None and script_end is not None and script_end > wrapper_end + one_s:
        arm["problems"].append(f"the script ended ({utc(script_end)}) after the wrapper's finished stamp ({utc(wrapper_end)})")
    starts = [t for t in (wrapper_start, script_start) if t is not None]
    ends = [t for t in (wrapper_end, script_end) if t is not None]
    arm["start"] = min(starts) if starts else None
    arm["end"] = max(ends) if ends else None
    arm["first_request_floor"] = script_start or wrapper_start   # the precise stamp when there is one: the wrapper's `date -u` stamp is whole seconds, truncated, so it can read up to 1 s BEFORE the script began
    arm["source"] = ("wrapper stamps and " if wrapper_start or wrapper_end else "") + "the script's FROZEN start= and SUMMARY end="
    # the store the script read: the scripts open `file:{db}?mode=ro` themselves (no ~ expansion, no URI escaping), a relative path against their own working directory (the FROZEN line's cwd=)
    db = arm["db"]
    if db:
        if "\x00" in db or any(ch in db for ch in "?#%") or db.startswith("~"):
            arm["problems"].append(f"the FROZEN db= {db[:60]!r} holds a character the scripts' file: URI does not handle (NUL, ?, #, %) or a leading ~ (the scripts do not expand it): the store they read is not established")
        elif os.path.isabs(db):
            arm["db_resolved"] = db
        elif arm["cwd"] and os.path.isabs(arm["cwd"]):
            arm["db_resolved"] = os.path.join(arm["cwd"], db)
        else:
            arm["problems"].append(f"the FROZEN db= {db[:60]!r} is relative and the FROZEN line has no usable cwd=: the store the script read is not established")
    if not arm["requests"]:
        arm["problems"].append("no RESULT line: no request to attribute")
    wrong = [r for r in arm["requests"] if arm["kind"] and r["kind"] != arm["kind"]]
    if wrong:
        arm["problems"].append(f"{len(wrong)} RESULT line(s) of the other script's shape in a version {arm['version']} output")
    for index, request in enumerate(arm["requests"], 1):
        if request["kind"] == "raw":
            ids = [request["corr"], request["rid"]]
            if all(i == "None" for i in ids):
                arm["problems"].append(f"request {index} ({request['mode']}) carries neither a correlation id nor a response id: its rows cannot be attributed")
            elif any(i != "None" and not full_id(i) for i in ids):
                arm["problems"].append(f"request {index} ({request['mode']}) carries a shortened id (a committed copy?): the private output is needed")
        else:
            if not (request["tag"].startswith("acceptance-") and "…" not in request["tag"] and len(request["tag"]) >= MIN_ID_LENGTH):
                arm["problems"].append(f"turn {index} ({request['shape']}) carries no full session id")
            if request["shape"] not in SHAPE_MIN_ROWS:
                arm["problems"].append(f"turn {index} has the shape {request['shape'][:30]!r}, which native script 2.1 does not define")
            elif request["min_rows"] != SHAPE_MIN_ROWS[request["shape"]]:
                arm["problems"].append(f"turn {index} ({request['shape']}) prints min_rows={request['min_rows']}, the shape needs {SHAPE_MIN_ROWS[request['shape']]}: a printed minimum must not lower the census's own")
    return arm


def parse_guard(path):
    """The guard 5 --idle-only receipt -> a dict: version, mode, start, end, clean, problems."""
    g = {"pid": None, "unit_started": None, "name": os.path.basename(path), "version": None, "mode": None, "start": None, "end": None, "clean": None, "problems": [], "clean_for": None}
    results = []
    for line in read_text(path, "guard receipt").splitlines():
        if re.match(r"^\s*GUARD(?![A-Za-z0-9_-])", line):
            g["guard_lines"] = g.get("guard_lines", 0) + 1
        if re.match(r"^\s*IDENTITY(?![A-Za-z0-9_-])", line):
            g["identity_lines"] = g.get("identity_lines", 0) + 1
        if re.match(r"^\s*(IDLE|GUARD)-RESULT\b", line):
            g["result_lines"] = g.get("result_lines", 0) + 1
        if re.match(r"^\s*CHECK\b", line) and re.search(r"(?i)\bFAIL", line):
            g["problems"].append(f"a CHECK line of the receipt says FAIL: {line[:70]!r}")
        m = re.match(r"^GUARD version=(\S+) mode=(\S+) script_sha256=([0-9a-f]{64}) start=(\S+) .*?\bclean_for=(\S+)", line)
        if m and g["version"] is None:
            g["version"], g["mode"], g["start"], g["clean_for"] = m.group(1), m.group(2), parse_ts(m.group(4)), m.group(5)
            g["script_sha"] = m.group(3)
            try:
                g["interval"] = float((re.search(r"\binterval=(\S+)", line) or [None, "nan"])[1])
            except ValueError:
                g["interval"] = float("nan")
            g["port"] = (re.search(r"\bport=([0-9]{1,6})", line) or [None, None])[1]
            g["unit"] = (re.search(r"\bunit=(\S+)", line) or [None, None])[1]
            g["tolerated"] = (re.search(r"\btolerate_units=(\[[^\]]*\])", line) or [None, None])[1]
            continue
        if re.match(r"^\s+IDENTITY ", line) and not g.get("identity_seen"):
            g["identity_seen"] = True
            pid_m, started_m = re.search(r"\bMainPID=([0-9]{1,12})", line), re.search(r"\bExecMainStartTimestamp=(\S+)", line)
            if pid_m and started_m:
                g["pid"], g["unit_started"] = pid_m.group(1), started_m.group(1)
            else:
                g["problems"].append("the receipt's identity line has no MainPID or no ExecMainStartTimestamp")
            continue
        m = re.match(r"^IDLE-RESULT clean=(True|False) failed_checks=(\[.*\]) end=(\S+)\s*$", line)
        if m:
            results.append((m.group(1) == "True", m.group(2), parse_ts(m.group(3))))
    if g.get("guard_lines", 0) != 1 or g.get("identity_lines", 0) > 1:
        g["problems"].append(f"the receipt holds {g.get('guard_lines', 0)} GUARD line(s) and {g.get('identity_lines', 0)} IDENTITY line(s): one run per file is expected (a hand-concatenated receipt is refused)")
    if g.get("result_lines", 0) != 1:
        g["problems"].append(f"{g.get('result_lines', 0)} IDLE-RESULT or GUARD-RESULT line(s) in the receipt (exactly one IDLE-RESULT is needed; another result line, even a malformed one, is refused)")
    if g["version"] == "5" and g.get("script_sha") != GUARD_SCRIPT_SHA:
        g["problems"].append(f"the receipt is from a guard script of sha256 {str(g.get('script_sha'))[:12]}, not guard 5 ({GUARD_SCRIPT_SHA[:12]})")
    if g["version"] == "5" and not (g.get("interval") is not None and 0 < g["interval"] <= GUARD_MAX_INTERVAL_S + 1e-9):
        g["problems"].append(f"the receipt's sampling interval is {g.get('interval')!r} s, not 0 < interval <= {GUARD_MAX_INTERVAL_S:.0f} s")
    if g["version"] != "5":
        g["problems"].append(f"the receipt is not from guard version 5 (GUARD version={g['version']!r})")
    if g["mode"] != "idle-only":
        g["problems"].append(f"the receipt is not an --idle-only receipt (mode={g['mode']!r})")
    try:
        observed = float(g["clean_for"])
    except (TypeError, ValueError):
        observed = None
    for key, label in (("port", "port"), ("unit", "unit"), ("tolerated", "tolerated units")):
        if g.get(key) != GUARD_EXPECT[key]:
            g["problems"].append(f"the receipt's {label} is {g.get(key)!r}, not the standard {GUARD_EXPECT[key]!r} of the 20128 canary")
    if observed is None or not math.isfinite(observed) or observed < GUARD_MIN_CLEAN_FOR_S:
        g["problems"].append(f"the idle observation lasted {g['clean_for']} s (clean_for), shorter than the {GUARD_MIN_CLEAN_FOR_S:.0f} s the protocol requires")
    if g["start"] is None:
        g["problems"].append("the receipt has no parsable start time (GUARD start=): the observed interval cannot be measured")
    if len(results) != 1:
        g["problems"].append(f"{len(results)} IDLE-RESULT line(s) (exactly one is needed)")
    else:
        clean, failed, end = results[0]
        g["clean"], g["end"] = clean, end
        if not clean or failed != "[]":
            g["problems"].append(f"the idle check was not clean (clean={clean}, failed_checks={failed})")
        if end is None:
            g["problems"].append("unparsable end time of the idle check")
    if g["start"] is not None and g["end"] is not None:   # the OBSERVED interval (start to end), not the declared clean_for: a receipt cannot claim more than it observed
        g["observed"] = (g["end"] - g["start"]).total_seconds()
        if g["observed"] < GUARD_MIN_CLEAN_FOR_S:
            g["problems"].append(f"the receipt's observed interval (GUARD start= to IDLE-RESULT end=) is {g['observed']:.1f} s, shorter than the {GUARD_MIN_CLEAN_FOR_S:.0f} s the protocol requires (zero, negative and reversed intervals included)")
        elif observed is not None and math.isfinite(observed) and observed > g["observed"] + 0.5:
            g["problems"].append(f"the receipt declares clean_for={g['clean_for']} s but its observed interval (start to end) is only {g['observed']:.1f} s")
    return g


# ---------------------------------------------------------------- the report
class Report:
    def __init__(self):
        self.lines = []
        self.reasons = []

    @staticmethod
    def clean(text):
        """Control characters of an input (a file name, a model or path string of a row) never start a line of the receipt of their own: they are written as \\xNN or \\uNNNN escapes."""
        return CONTROL_CHARS.sub(lambda m: "\\x%02x" % ord(m.group(0)) if ord(m.group(0)) < 256 else "\\u%04x" % ord(m.group(0)), str(text))

    def say(self, text=""):
        self.lines.append(self.clean(text))

    def bad(self, reason):
        reason = self.clean(reason)
        self.reasons.append(reason)
        self.lines.append("PROBLEM " + reason)

    @property
    def state(self):
        return "INCONCLUSIVE" if self.reasons else "PASS"


def row_dict(columns, values):
    return dict(zip(columns, values))


def shown_db(db):
    home = os.path.expanduser("~")
    return ("~" + db[len(home):]) if db.startswith(home + os.sep) else db


def open_store(db):
    """The store, read-only; raises sqlite3.Error (also for a missing file or a missing column) when it cannot be read."""
    con = sqlite3.connect(Path(db).absolute().as_uri() + "?mode=ro", uri=True)
    con.isolation_level = None   # explicit transaction control: ONE read transaction for every query below, so that the gateway's concurrent writes cannot tear the measurement
    con.execute("BEGIN")
    con.execute("select " + ", ".join(CALL_COLUMNS) + " from call_logs limit 1")
    con.execute("select " + ", ".join(USAGE_COLUMNS) + " from usage_history limit 1")
    return con


def evaluate(arm_paths, guard_path, db, diagnostic=False, require_arms=DEFAULT_REQUIRED_ARMS):
    """Everything the census measures and decides; returns the Report (never raises for a bad store or a bad arm: those are problems)."""
    rep = Report()
    me = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    rep.say(f"CENSUS version={VERSION} script_sha256={me} db={shown_db(db)} arms={len(arm_paths)} required_arms={require_arms} guard_receipt={'none (diagnostic run)' if guard_path is None else os.path.basename(guard_path)} diagnostic={diagnostic}")
    # ---- arms
    if len(arm_paths) != require_arms:
        rep.bad(f"{len(arm_paths)} arm output(s) given, {require_arms} required (--require-arms; the canary has four: raw, race, native, native tool-call): an omitted arm would leave its requests unattributed")
    arms = [parse_arm(p) for p in arm_paths]
    for index, arm in enumerate(arms, 1):
        rep.say(f"INPUT arm={index} file={arm['name']} kind={arm['kind']} script_version={arm['version']} script_sha256={str(arm['script_sha'])[:12]} requests={len(arm['requests'])} start={utc(arm['start']) if arm['start'] else None} end={utc(arm['end']) if arm['end'] else None} "
                f"interval_source=({arm['source']}) store_in_script_line={shown_db(arm['db']) if arm['db'] else None} journal_failed_to_save_call_log={arm['journal']}")
        for problem in arm["problems"]:
            rep.bad(f"arm {index} ({arm['name']}): {problem}")
        if not arm["db"]:
            rep.bad(f"arm {index} ({arm['name']}): the FROZEN line names no store (db=): the census cannot tell that the script read the store it reads")
        elif arm["db_resolved"] is not None and os.path.realpath(arm["db_resolved"]) != os.path.realpath(db):
            rep.bad(f"arm {index} ({arm['name']}): the script read the store {shown_db(arm['db'])}, the census reads {shown_db(db)}")
        if arm["journal"] != "0":
            rep.bad(f"arm {index} ({arm['name']}): journal_failed_to_save_call_log={arm['journal']} (0 is needed: the named check for what the census cannot exclude)")
    timed = sorted((a for a in arms if a["start"] and a["end"]), key=lambda a: a["start"])
    canary_gate = require_arms == DEFAULT_REQUIRED_ARMS and not diagnostic   # the canary contract: four wrapper arms (raw set, race set, native one-request set, native tool-call set, from `run-acceptance.sh canary`), each with the wrapper's header and identity lines
    scope = "canary" if canary_gate else "trial"
    rep.say(f"SCOPE {scope}" + ("" if canary_gate else ": not the canary's four arms (or a diagnostic run): a PASS here is a trial result, not canary acceptance"))
    if canary_gate:
        shape = [a["kind"] for a in timed]
        if shape != ["raw", "raw", "native", "native"]:
            rep.bad(f"the arms in time order are {shape}; the canary's are ['raw', 'raw', 'native', 'native']")
        found = tuple(len(a["requests"]) for a in timed)
        if found != CANARY_REQUESTS:
            rep.bad(f"the arms in time order carry {found} RESULT line(s), the canary's frozen commands make {CANARY_REQUESTS} (raw set, race set, native one-request turns, native tool-call turns): a request that has no RESULT line would go unattributed")
        for index, (a, (kind, expected)) in enumerate(zip(timed, CANARY_ARMS), 1):
            got = collections.Counter(r["mode"] for r in a["requests"] if r["kind"] == "raw") if kind == "raw" else collections.Counter((r["shape"], r["min_rows"]) for r in a["requests"] if r["kind"] == "native")
            if got != expected:
                rep.bad(f"arm {index} ({a['name']}) holds {dict(got)}, the canary's frozen command makes {dict(expected)}: the requests of the canary are fixed by its commands")
        for index, a in enumerate(arms, 1):
            if len(a["identities"]) != 2:
                rep.bad(f"arm {index} ({a['name']}) holds {len(a['identities'])} wrapper identity line(s): the wrapper reads the identity before and after the run, so a cut-off output cannot establish that the gateway stayed the same process")
            if a["run_statuses"] != [0]:
                rep.bad(f"arm {index} ({a['name']}) holds RUN-STATUS lines {a['run_statuses']}: exactly one with status 0 is needed")
            if a["finished_n"] != 1 or a["header_n"] != 1:
                rep.bad(f"arm {index} ({a['name']}) holds {a['header_n']} wrapper header(s) and {a['finished_n']} `# finished` line(s): the wrapper's output is complete only with one of each")
        for index, a in enumerate(arms, 1):
            if a["wrapper_kind"] is None:
                rep.bad(f"arm {index} ({a['name']}) has no wrapper header (a direct run of a frozen script): the canary gate accepts only the wrapper's canary outputs")
            elif a["wrapper_kind"] != "canary":
                rep.bad(f"arm {a['name']} is a wrapper `{a['wrapper_kind']}` run, not a canary run")
    for a, b in zip(timed, timed[1:]):
        if b["start"] < a["end"] - dt.timedelta(seconds=STAMP_RESOLUTION_S):
            rep.bad(f"arms overlap: {a['name']} ends {utc(a['end'])}, {b['name']} starts {utc(b['start'])} (stamps of whole seconds are tolerated: {STAMP_RESOLUTION_S} s)")
    for a in timed:
        if a["end"] < a["start"]:
            rep.bad(f"arm {a['name']} ends before it starts")
    # ---- guard and window
    guard = None
    if diagnostic:
        rep.bad("--diagnostic run: the measurements are printed, but PASS is not possible (the verdict of a window needs the guard receipt of a real acceptance run)")
    if guard_path is None:
        if not diagnostic:
            rep.bad("no guard receipt: the window's idle start is not established, so PASS is not possible")
    else:
        guard = parse_guard(guard_path)
        rep.say(f"GUARD file={guard['name']} version={guard['version']} mode={guard['mode']} clean_for={guard['clean_for']} start={utc(guard['start']) if guard['start'] else None} end={utc(guard['end']) if guard['end'] else None} clean={guard['clean']}")
        for problem in guard["problems"]:
            rep.bad(f"guard receipt: {problem}")
        compared = 0
        for index, arm in enumerate(arms, 1):
            if not arm["identities"]:
                if canary_gate:
                    rep.bad(f"arm {index} ({arm['name']}) has no usable wrapper identity line (MainPID and ExecMainStartTimestamp): the canary gate must establish that the gateway stayed the same process in every arm")
                continue
            if guard["pid"] is None:
                rep.bad(f"arm {index} ({arm['name']}) carries the wrapper's identity line but the guard receipt has none: the process cannot be compared")
                continue
            compared += 1
            for pid, started in arm["identities"]:   # every identity line of the arm (the wrapper reads it before and after the run), not only the first
                if (pid, started) != (guard["pid"], guard["unit_started"]):
                    rep.bad(f"arm {index} ({arm['name']}): the gateway was MainPID {pid} started {started}, the idle receipt saw MainPID {guard['pid']} started {guard['unit_started']}: a restart between them voids the idle claim")
        if canary_gate and guard["pid"] is None:
            rep.bad("the guard receipt has no identity line (MainPID and ExecMainStartTimestamp): the canary gate must compare the process of the idle check with that of every arm")
        if canary_gate and compared != len(arms):
            rep.bad(f"the gateway process identity was compared for {compared} of {len(arms)} arm(s): the canary gate needs every comparison to succeed")
        rep.say(f"IDENTITY-CHECK guard MainPID={guard['pid']} started={guard['unit_started']}; compared with {compared} of {len(arms)} arm(s)" + ("" if canary_gate else " (a direct run carries no identity line)"))
        if guard["end"] is not None and timed:
            gap = (timed[0]["first_request_floor"] - guard["end"]).total_seconds()
            rep.say(f"GUARD-TO-FIRST-ARM gap_s={gap:.1f} (the guard's end to the first script's own start; needed: 0 <= gap <= {GUARD_MAX_AGE_S})")
            if gap < 0:
                rep.bad(f"the guard receipt ended {-gap:.1f} s AFTER the first arm began: it does not cover the start of the window")
            elif gap > GUARD_MAX_AGE_S:
                rep.bad(f"the guard receipt ended {gap:.0f} s before the first arm began (more than {GUARD_MAX_AGE_S} s): an idle claim that does not reach the window proves nothing")
    if not timed:
        rep.bad("no arm with a usable interval: no window")
        rep.say(f"CENSUS-RESULT INCONCLUSIVE exit=2 problems={len(rep.reasons)} scope={scope}")
        return rep
    guard_start_ok = bool(guard and guard["end"] and guard["end"] <= timed[0]["first_request_floor"] and (timed[0]["first_request_floor"] - guard["end"]).total_seconds() <= GUARD_MAX_AGE_S)
    w0 = guard["end"] if guard_start_ok else timed[0]["start"]
    w1 = max(a["end"] for a in timed) + dt.timedelta(seconds=SETTLE_S)
    w0s, w1s = utc(w0), utc(w1)
    u0s, u1s = shift(w0s, -CANDIDATE_SECONDS), shift(w1s, CANDIDATE_SECONDS)
    bounds = [w0] + [a["end"] for a in timed[:-1]] + [w1]
    rep.say(f"WINDOW start={w0s} end={w1s} ({'the guard receipt end' if guard_start_ok else 'the first arm start: no usable guard receipt'} .. the last arm end + {SETTLE_S} s); usage rows read {u0s} .. {u1s} (the window padded by the {CANDIDATE_SECONDS} s of the scripts' candidate query)")
    rep.say("TILES " + " | ".join(f"{k}: {a['name']} {utc(bounds[k - 1])}..{utc(bounds[k])}" for k, a in enumerate(timed, 1)))
    rep.say("CLOCKS arm intervals: the host clock (the wrapper's date -u stamps, the scripts' datetime.now(UTC)); rows: the gateway process's own clock (new Date() in UTC), same machine; own rows must lie inside their arm's interval with a margin of "
            f"{CLOCK_MARGIN_S} s; the guard's end is the host clock")
    rep.say("IDLE-MEANING idle for the whole window = the guard was clean at the window's start AND no foreign call row and no orphan usage row exists over the whole tiled window (the guard's observation does not observe the window itself)")
    # ---- queries (printed first, so a failure below still leaves them in the receipt)
    for name, sql in (("own-raw", SQL_OWN_RAW), ("own-native", SQL_OWN_NATIVE), ("window-calls", SQL_WINDOW_CALLS), ("window-usage", SQL_WINDOW_USAGE), ("candidates", SQL_CANDIDATES), ("competing-calls", SQL_COMPETING_CALLS), ("all-call-stamps", SQL_ALL_CALL_STAMPS), ("all-usage-stamps", SQL_ALL_USAGE_STAMPS)):
        rep.say(f"QUERY {name}: {sql}")
    rep.say(f"QUERY journal: journalctl --user -u {JOURNAL_UNIT} --since <window start - {JOURNAL_PAD_S} s> UTC --until <window end + {JOURNAL_PAD_S} s> UTC -o json --no-pager (every `tag=HTTP method=POST` line whose message starts with the inbound-request marker; counted inside the window by the journal's own timestamp and compared with the number of own call rows)")
    rep.say(f"BOUNDS own-raw: (correlation id, response id) of each request as printed by the script (shown here as id8); own-native: the session id of each turn; window-calls: ({u0s}, {w1s}) (the window's start padded by the same {CANDIDATE_SECONDS} s as the usage rows); window-usage: ({u0s}, {u1s}); "
            f"candidates: (shift(call row timestamp, -{CANDIDATE_SECONDS}), shift(call row timestamp, {CANDIDATE_SECONDS}), model, tokens_in, tokens_out) per own call row; competing-calls: (shift(usage row timestamp, -{CANDIDATE_SECONDS}), shift(usage row timestamp, {CANDIDATE_SECONDS}), model, tokens_input, tokens_output) per usage row an own call row can claim; all-*-stamps: every timestamp of the two tables, each validated here (text, the gateway's format {TS_GLOB}, calendar-valid, year 2000-2100)")
    # ---- the store is read only after the whole declared interval (the window, the usage rows' pad and a settling slack) has elapsed on the host clock: a row of the window's tail cannot be missed because the census ran too early
    u1_dt = parse_ts(u1s)
    if u1_dt is None:
        rep.bad(f"the declared interval's padded end {u1s} is not a stamp of 2000-2100: nothing was measured")
        rep.say(f"CENSUS-RESULT INCONCLUSIVE exit=2 problems={len(rep.reasons)} scope={scope}")
        return rep
    ready_at = u1_dt + dt.timedelta(seconds=OBSERVE_SLACK_S)
    ahead = (ready_at - now_utc()).total_seconds()
    if diagnostic:
        rep.say(f"WAIT skipped: a diagnostic run cannot PASS; the declared interval (to {u1s}, +{OBSERVE_SLACK_S} s) {'has ended' if ahead <= 0 else 'has not ended'}")
    elif ahead > MAX_WAIT_S:
        rep.bad(f"the declared interval ends {ahead:.0f} s ahead of the host clock (the usage rows' pad ends {u1s}, +{OBSERVE_SLACK_S} s of settling): more than the {MAX_WAIT_S} s the census waits for, so the stamps or the clock are wrong or the run is not over; the store was read anyway so that the receipt shows what is there, and nothing here is a PASS")
    elif ahead > 0:
        rep.say(f"WAIT the declared interval (window end {w1s} + the {CANDIDATE_SECONDS} s usage pad + {OBSERVE_SLACK_S} s of settling) ends {utc(ready_at)}; waiting {ahead:.1f} s before the store is read")
        time.sleep(ahead)
        while now_utc() < ready_at:   # sleep() counts on the monotonic clock, the stamps are wall-clock
            time.sleep(0.05)
    read_at = now_utc()
    rep.say(f"OBSERVED-AT {utc(read_at)} (host clock) declared_interval_end={u1s} seconds_after_it_ended={(read_at - u1_dt).total_seconds():.1f} (needed: at least {OBSERVE_SLACK_S} s, waited for when less than {MAX_WAIT_S} s remain)")
    try:
        con = open_store(db)
    except (sqlite3.Error, OSError) as error:
        rep.say(f"STORE {shown_db(db)} readable=False error={type(error).__name__}")
        rep.bad(f"the gateway store cannot be read ({type(error).__name__}): nothing was measured")
        rep.say(f"CENSUS-RESULT INCONCLUSIVE exit=2 problems={len(rep.reasons)} scope={scope}")
        return rep
    rep.say(f"STORE {shown_db(db)} readable=True (opened with mode=ro; one read transaction for all queries)")
    try:
        measure(rep, con, timed, w0s, w1s, u0s, u1s, bounds)
    except Exception as error:   # whatever fails here, the receipt printed so far stays and the result is INCONCLUSIVE, never a crash without a receipt and never a PASS
        rep.bad(f"the measurement failed ({type(error).__name__}: {str(error)[:80]}): the rest was not measured, so nothing here is a PASS")
    finally:
        con.close()
    rep.say("NOT-EXCLUDED the rows alone cannot attribute a usage row to its request (the usage table has no request key), and the association tolerates numbers that are not identical (latency within 100 ms; time to first token within 1 ms, or not compared at all when either side is NULL, which is routine here: 32 of 244 associated pairs on this host have a NULL time to first token on the call row only; within 3 s of the own call row). A PASS therefore rests on the gateway's journal as the independent record: the number of inbound POST requests it logged in the window equals the number of own call rows and the lines pair with them one to one by time within 3 s (so a foreign request leaves a line even when its call row was never written), and the rows are paired one to one. What stays outside: a foreign request that the gateway did not log on arrival while the own requests' lines are all present, a lost own line balanced by a foreign line that arrives within 3 s of it, a persistence failure that is not logged (a save skipped while disk persistence is off or saves are closing, the usage insert's same-millisecond dedup), and traffic that is not an inbound POST. The wrapper's journal line per arm (journal_failed_to_save_call_log, above) covers only the call-log save's logged failure")
    rep.say(f"CENSUS-RESULT {rep.state} exit={0 if rep.state == 'PASS' else 2} problems={len(rep.reasons)} scope={scope}")
    return rep


def measure(rep, con, timed, w0s, w1s, u0s, u1s, bounds):
    # ---- timestamp format: a row whose timestamp has another format would be skipped by the string bounds, silently
    odd_calls = sum(1 for (v,) in con.execute(SQL_ALL_CALL_STAMPS) if not valid_gateway_ts(v))
    odd_usage = sum(1 for (v,) in con.execute(SQL_ALL_USAGE_STAMPS) if not valid_gateway_ts(v))
    rep.say(f"FORMAT scope=whole tables call_logs_rows_with_another_timestamp_format={odd_calls} usage_history_rows_with_another_timestamp_format={odd_usage}")
    if odd_calls or odd_usage:
        rep.bad(f"{odd_calls} call_logs and {odd_usage} usage_history row(s) have a timestamp that is not a valid gateway stamp ({TS_GLOB} with calendar-valid values, year 2000-2100: not text, no milliseconds, another offset, a leading space, or only the SHAPE of a stamp such as 2026-99-06 or xxxx-xx-xxTxx:xx:xx.xxxZ): the window's string bounds would skip them silently")
    null_call_ids = con.execute("select count(*) from call_logs where id is null or typeof(id) != 'text'").fetchone()[0]
    null_usage_ids = con.execute("select count(*) from usage_history where id is null or typeof(id) != 'integer'").fetchone()[0]
    rep.say(f"IDS call_logs_rows_without_a_text_id={null_call_ids} usage_history_rows_without_an_integer_id={null_usage_ids}")
    if null_call_ids or null_usage_ids:
        rep.bad(f"{null_call_ids} call_logs row(s) without a text id and {null_usage_ids} usage_history row(s) without an integer id: rows are told apart by their ids, and two NULL ids would be one row")
    # ---- own rows, by tag, per request
    own, claims = {}, collections.defaultdict(list)
    per_arm_own = collections.Counter()
    for k, arm in enumerate(timed, 1):
        for j, request in enumerate(arm["requests"], 1):
            if request["kind"] == "raw":
                params = (None if request["corr"] == "None" else request["corr"], None if request["rid"] == "None" else request["rid"])
                if params == (None, None):
                    continue
                rows = [row_dict(CALL_COLUMNS, r) for r in con.execute(SQL_OWN_RAW, params).fetchall()]
                label = f"request {j} ({request['mode']}, corr {'None' if request['corr'] == 'None' else id8(request['corr'])}, response {'None' if request['rid'] == 'None' else id8(request['rid'])})"
                if len(rows) != 1:
                    rep.bad(f"arm {k} {label}: {len(rows)} own call row(s) (exactly 1 is needed)")
            else:
                if not request["tag"].startswith("acceptance-"):
                    continue
                rows = [row_dict(CALL_COLUMNS, r) for r in con.execute(SQL_OWN_NATIVE, (request["tag"],)).fetchall()]
                label = f"turn {j} ({request['shape']}, session {id8(request['tag'])})"
                if not request["judged"]:
                    rep.bad(f"arm {k} {label}: the script did not judge this turn (the exec did not complete): its rows may be incomplete")
                elif len(rows) < request["min_rows"]:
                    rep.bad(f"arm {k} {label}: {len(rows)} own call row(s) (at least {request['min_rows']} are needed)")
            for row in rows:
                own[row["id"]] = row
                claims[row["id"]].append((k, j))
                per_arm_own[k] += 1
    for rid, who in claims.items():
        if len(who) > 1:
            rep.bad(f"own call row {id8(rid)} is claimed by {len(who)} requests {who}")
    rep.say(f"OWN call rows={len(own)} per arm={dict(sorted(per_arm_own.items()))}")
    # ---- the gateway's journal: an independent record of every inbound request, written when it ARRIVES (a foreign request leaves a line even when its call row was never written)
    w0, w1 = bounds[0], bounds[-1]
    try:
        posts = journal_posts(JOURNAL_UNIT, w0 - dt.timedelta(seconds=JOURNAL_PAD_S), w1 + dt.timedelta(seconds=JOURNAL_PAD_S))
    except RuntimeError as error:
        rep.say(f"JOURNAL unit={JOURNAL_UNIT} readable=False error={error}")
        rep.bad(f"the gateway journal cannot be read ({error}): the independent record of the window's requests is missing, so PASS is not possible")
        posts = None
    if posts is not None:
        inside = [(t, label) for t, label in posts if w0 <= t <= w1]
        rep.say(f"JOURNAL unit={JOURNAL_UNIT} readable=True inbound_POST_lines_in_window={len(inside)} own_call_rows={len(own)} (lines read over the window padded by {JOURNAL_PAD_S} s: {len(posts)}; counted by the journal's own timestamp inside the window)")
        if len(inside) != len(own):
            for t, label in inside[:60]:
                rep.say(f"   journal request line at {utc(t)} {label}")
            rep.bad(f"the gateway's journal shows {len(inside)} inbound POST request(s) in the window and the store holds {len(own)} own call row(s): a request that left no call row (dropped, or another client's) cannot be excluded")
        # pairing by time: a count alone can be balanced by an own request's missing line and a foreign request's line elsewhere
        arrivals = {}
        for cid, row in own.items():
            ts = parse_ts(row["timestamp"])
            try:
                duration = float(row["duration"])
            except (TypeError, ValueError):
                duration = None
            arrivals[cid] = (ts - dt.timedelta(milliseconds=duration)) if ts is not None and duration is not None and math.isfinite(duration) and duration >= 0 else None
        pair_edges = {cid: ([j for j, (t, _label) in enumerate(inside) if abs((t - arr).total_seconds()) <= JOURNAL_PAIR_S] if arr is not None else []) for cid, arr in arrivals.items()}
        stray = [j for j in range(len(inside)) if j not in {j2 for js in pair_edges.values() for j2 in js}]
        unpaired_own, unpaired_lines = 0, len(stray)
        for calls, lines in components(pair_edges):
            status, _pairs = matching_status({c: pair_edges[c] for c in calls}, calls, lines)
            if status in ("unequal", "none"):
                unpaired_own += max(0, len(calls) - len(lines)) if status == "unequal" else len(calls)
                unpaired_lines += max(0, len(lines) - len(calls)) if status == "unequal" else 0
        rep.say(f"JOURNAL-PAIRING each own call row to one inbound line within {JOURNAL_PAIR_S:.0f} s of (its timestamp - its duration): own_call_rows_without_a_line={unpaired_own} inbound_lines_without_an_own_call_row={unpaired_lines}")
        if unpaired_own or unpaired_lines:
            for j in stray[:20]:
                rep.say(f"   inbound line with no own call row near it at {utc(inside[j][0])} {inside[j][1]}")
            rep.bad(f"the gateway's journal and the own call rows do not pair one to one by time: {unpaired_own} own call row(s) have no inbound line within {JOURNAL_PAIR_S:.0f} s of their arrival and {unpaired_lines} inbound line(s) have no own call row (a lost line balanced by another client's request cannot pass)")
    # ---- clocks: own rows inside their arm's interval and inside the window
    margin = dt.timedelta(seconds=CLOCK_MARGIN_S)
    for k, arm in enumerate(timed, 1):
        leads, lags = [], []
        for rid, who in claims.items():
            if who[0][0] != k:
                continue
            ts = parse_ts(own[rid]["timestamp"])
            if ts is None:
                rep.bad(f"own call row {id8(rid)} has an unparsable timestamp {own[rid]['timestamp']!r}")
                continue
            leads.append((ts - arm["start"]).total_seconds())
            lags.append((arm["end"] - ts).total_seconds())
            if ts < arm["start"] - margin or ts > arm["end"] + margin:
                rep.bad(f"own call row {id8(rid)} (arm {k}) at {own[rid]['timestamp']} lies outside the arm's interval {utc(arm['start'])}..{utc(arm['end'])} (margin {CLOCK_MARGIN_S} s): the clocks disagree or the row is not this arm's")
        rep.say(f"CLOCK arm={k} own_rows={len(leads)} min_s_after_arm_start={min(leads):.1f} min_s_before_arm_end={min(lags):.1f}" if leads else f"CLOCK arm={k} own_rows=0")
    for rid, row in own.items():
        if not (w0s <= row["timestamp"] <= w1s):
            rep.bad(f"own call row {id8(rid)} at {row['timestamp']} lies outside the window {w0s}..{w1s}")
    # ---- window rows: foreign call rows, usage rows
    calls_win = [row_dict(CALL_COLUMNS, r) for r in con.execute(SQL_WINDOW_CALLS, (u0s, w1s)).fetchall()]   # padded at the start like the usage rows: a foreign request that finished just before the window has its usage row in the pad
    usage_win = [row_dict(USAGE_COLUMNS, r) for r in con.execute(SQL_WINDOW_USAGE, (u0s, u1s)).fetchall()]
    foreign = [c for c in calls_win if c["id"] not in own]
    rep.say(f"WINDOW-ROWS call_logs={len(calls_win)} own={len(calls_win) - len(foreign)} foreign={len(foreign)}; usage_history={len(usage_win)}")

    def tile(ts):
        for k in range(1, len(bounds)):
            if ts < utc(bounds[k]) or k == len(bounds) - 1:
                return k
        return len(bounds) - 1

    foreign_by = collections.Counter((tile(c["timestamp"]), c["path"], c["status"], c["model"]) for c in foreign)
    for (k, path, status, model), n in sorted(foreign_by.items(), key=repr):
        times = sorted(str(c["timestamp"]) for c in foreign if (tile(c["timestamp"]), c["path"], c["status"], c["model"]) == (k, path, status, model))
        rep.say(f"FOREIGN tile={k} path={path} status={status} model={model} rows={n} first={times[0]} last={times[-1]}")
    for c in foreign[:20]:
        rep.say(f"   foreign call row id8={id8(c['id'])} at {c['timestamp']} path={c['path']} status={c['status']} model={c['model']} provider={c['provider']} connection={id8(c['connection_id'])}")
    if foreign:
        rep.bad(f"{len(foreign)} foreign call row(s) in the window (any call row without an own tag, read from {CANDIDATE_SECONDS} s before the window's start): a foreign request ran, so a foreign usage row could stand in for an own one")
    # ---- edges: the scripts' candidate query per own call row, intersected with associated()
    edges, lookalikes, edge_rows = {}, 0, {}
    for cid, c in sorted(own.items(), key=lambda kv: (str(kv[1]["timestamp"]), str(kv[0]))):
        try:
            lo, hi = shift(c["timestamp"], -CANDIDATE_SECONDS), shift(c["timestamp"], CANDIDATE_SECONDS)
        except ValueError:   # a timestamp that is not the gateway's format (the FORMAT check above has reported it): the candidate window cannot be computed
            rep.bad(f"own call row {id8(cid)} has a timestamp {c['timestamp']!r} that is not {TS_GLOB}: its candidate usage rows cannot be looked up")
            edges[cid] = []
            continue
        rows = [row_dict(USAGE_COLUMNS, r) for r in con.execute(SQL_CANDIDATES, (lo, hi, c["model"], c["tokens_in"], c["tokens_out"])).fetchall()]
        strong = []
        why_c = numbers_problem(c["duration"], c["ttft_ms"])
        if why_c:
            rep.bad(f"own call row {id8(cid)}: its {why_c}: the association rests on numbers that cannot be compared")
        for u in rows:
            why_u = None if why_c else numbers_problem(u["latency_ms"], u["ttft_ms"])
            if why_c or why_u:
                if why_u:
                    rep.bad(f"usage row {id8(u['id'])} (a candidate of own call row {id8(cid)}): its {why_u}: the association would rest on numbers that cannot be compared")
                continue
            try:
                ok = associated(c, u)
            except (TypeError, ValueError):   # a latency or time to first token that is not a number: the scripts' own function would raise too; here it is a problem, not a crash
                rep.bad(f"own call row {id8(cid)} or its candidate usage row {id8(u['id'])} has a latency or time to first token that is not a number: the association cannot be evaluated")
                ok = False
            if ok:
                strong.append(u)
                edge_rows[u["id"]] = u
        lookalikes += len(rows) - len(strong)
        edges[cid] = [u["id"] for u in strong]
    # ---- the exact complement of the window check: every call row that is not this run's and could own a usage row that an own call row can claim (the candidate query seen from the usage row: same model and token counts within 3 s, and the scripts' own association)
    competing, competing_rows = 0, []
    for uid, u in sorted(edge_rows.items(), key=lambda kv: (str(kv[1]["timestamp"]), str(kv[0]))):
        try:
            lo, hi = shift(u["timestamp"], -CANDIDATE_SECONDS), shift(u["timestamp"], CANDIDATE_SECONDS)
        except ValueError:   # a usage stamp that cannot be shifted: the FORMAT check reports it too, and here it is a problem of its own
            rep.bad(f"usage row {id8(uid)} (a candidate of an own call row) has a timestamp {u['timestamp']!r} that is not {TS_GLOB}: the call rows that could own it cannot be looked up")
            continue
        for f in (row_dict(CALL_COLUMNS, r) for r in con.execute(SQL_COMPETING_CALLS, (lo, hi, u["model"], u["tokens_input"], u["tokens_output"])).fetchall()):
            if f["id"] in own:
                continue
            try:
                ok = associated(f, u) or numbers_problem(f["duration"], f["ttft_ms"]) is not None
            except (TypeError, ValueError):
                ok = True   # numbers that cannot be compared cannot show that this call row is NOT an owner
            if ok:
                competing += 1
                competing_rows.append((u, f))
    for u, f in competing_rows[:20]:
        rep.say(f"   usage row id8={id8(u['id'])} at {u['timestamp']} (a candidate of an own call row) is also associated with the NOT-own call row id8={id8(f['id'])} at {f['timestamp']} path={f['path']} status={f['status']} model={f['model']}")
    rep.say(f"COMPETING-OWNERS usage_rows_an_own_call_row_can_claim={len(edge_rows)} not_own_call_rows_that_could_own_them={competing}")
    if competing:
        rep.bad(f"{competing} call row(s) that are not this run's could own a usage row that an own call row can claim (same model and counts, within {CANDIDATE_SECONDS} s, and the scripts' association): a foreign request, wherever it lies relative to the window's edges, whose usage row could stand in for an own one")
    comps = components(edges)
    statuses = collections.Counter()
    for calls, usages in comps:
        if not usages:
            continue   # an own call row without any associated usage row: reported below
        adj = {c: edges[c] for c in calls}
        status, _pairs = matching_status(adj, calls, usages)
        statuses[status] += 1
        if status != "unique":
            rep.say(f"   component calls={[id8(c) for c in calls]} usage_rows={len(usages)} status={status}")
            rep.bad(f"a component of {len(calls)} own call row(s) and {len(usages)} usage row(s) has no unique perfect matching (status {status}): the usage rows of these calls cannot be established one to one")
    with_edges = {u for us in edges.values() for u in us}
    uncovered = [cid for cid, us in edges.items() if not us]
    for cid in uncovered:
        rep.bad(f"own call row {id8(cid)} ({own[cid]['path']}, status {own[cid]['status']}, tokens {own[cid]['tokens_in']}/{own[cid]['tokens_out']}) has no associated usage row (same model and token counts within {CANDIDATE_SECONDS} s, and provider, account, endpoint, latency, time to first token)")
    rep.say(f"MATCHING own_call_rows={len(edges)} components={len(comps)} statuses={dict(sorted(statuses.items()))} own_call_rows_without_usage_row={len(uncovered)} look_alike_usage_rows_that_do_not_associate={lookalikes}")
    # ---- a group of usage rows with identical metadata (to the millisecond and the fraction of the first-token time) is a duplicated row, or requests the association cannot tell apart
    groups = collections.Counter((u["model"], u["tokens_input"], u["tokens_output"], u["provider"], u["connection_id"], u["api_key_id"], u["endpoint"], u["latency_ms"], u["ttft_ms"]) for u in usage_win)
    duplicated = {key: n for key, n in groups.items() if n > 1}
    rep.say(f"DUPLICATES groups_of_usage_rows_with_identical_metadata={len(duplicated)} rows={sum(duplicated.values())}")
    if duplicated:
        rep.bad(f"{len(duplicated)} group(s) of usage rows with identical metadata (model, counts, account, endpoint, latency, time to first token): a duplicated row, or requests the association cannot tell apart")
    # ---- orphans: usage rows of the padded window with no edge from an own call row
    orphans = [u for u in usage_win if u["id"] not in with_edges]
    for u in orphans[:20]:
        rep.say(f"   orphan usage row id8={id8(u['id'])} at {u['timestamp']} endpoint={u['endpoint']} model={u['model']} provider={u['provider']} connection={id8(u['connection_id'])} tokens={u['tokens_input']}/{u['tokens_output']}")
    rep.say(f"ORPHANS usage_rows_in_window={len(usage_win)} associated_with_an_own_call_row={len(usage_win) - len(orphans)} orphans={len(orphans)}")
    if orphans:
        rep.bad(f"{len(orphans)} orphan usage row(s) in the window (no own call row is associated with them): foreign traffic or a call row that was never written")
    # ---- counts by account (hashed), endpoint and model
    rep.say("COUNTS own call rows by (connection, path, model): " + str(sorted(collections.Counter((id8(c["connection_id"]), c["path"], c["model"]) for c in own.values()).items(), key=repr)))
    rep.say("COUNTS usage rows of the window by (connection, endpoint, model): " + str(sorted(collections.Counter((id8(u["connection_id"]), u["endpoint"], u["model"]) for u in usage_win).items(), key=repr)))
    per_tile = collections.defaultdict(lambda: {"own_calls": 0, "foreign_calls": 0, "usage": 0, "orphans": 0})
    for c in calls_win:
        per_tile[tile(c["timestamp"])]["own_calls" if c["id"] in own else "foreign_calls"] += 1
    for u in usage_win:
        per_tile[tile(u["timestamp"])]["orphans" if u["id"] not in with_edges else "usage"] += 1
    for k in range(1, len(bounds)):
        rep.say(f"TILE {k}: " + ", ".join(f"{name}={value}" for name, value in sorted(per_tile[k].items())))


def run(argv):
    ap = Parser(add_help=False)
    ap.add_argument("--guard-receipt")
    ap.add_argument("--diagnostic", action="store_true")
    ap.add_argument("--require-arms", type=int, default=DEFAULT_REQUIRED_ARMS)
    ap.add_argument("--db", default=DEFAULT_DB)
    ap.add_argument("arms", nargs="+")
    args = ap.parse_args(argv)
    if not args.guard_receipt and not args.diagnostic:
        raise UsageError("--guard-receipt is required (the guard 5 --idle-only receipt of the pre-arm check); --diagnostic measures without one and can only be INCONCLUSIVE")
    for p in args.arms + ([args.guard_receipt] if args.guard_receipt else []):
        if not os.path.isfile(p):
            raise UsageError(f"not a readable file: {p!r}")
    rep = evaluate(args.arms, args.guard_receipt, os.path.expanduser(args.db), diagnostic=args.diagnostic, require_arms=args.require_arms)
    return (0 if rep.state == "PASS" else 2), "\n".join(rep.lines) + "\n"


def main():
    sys.setrecursionlimit(10000)
    try:
        code, text = run(sys.argv[1:])
        sys.stdout.buffer.write(text.encode("utf-8", "backslashreplace"))   # a file name that is not UTF-8 must not cost the receipt
        sys.stdout.buffer.flush()
    except UsageError as error:
        print(f"ERROR {error}", file=sys.stderr)
        sys.exit(3)
    except Exception as error:   # the contract is 0, 2 or 3: an unforeseen failure is not a verdict
        print(f"ERROR internal: {type(error).__name__}: {error}", file=sys.stderr)
        sys.exit(3)
    sys.exit(code)


if __name__ == "__main__":
    main()
