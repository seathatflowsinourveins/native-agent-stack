#!/usr/bin/env python3
"""Native `codex exec` turns through NativeStack's omniroute profile (20128), with attribution by exclusivity (frozen before the canary switch; see checks/live-acceptance-protocol.md).

A native `codex exec` cannot read the response's X-Correlation-Id, so its rows cannot be matched by id. Instead the script runs only in a QUIET window and requires that the window holds exactly the rows
the turn should produce: before the turn, no worker process of another lane is running and no call_logs row was written in the last 120 s (it polls up to --quiet-timeout seconds, else the shape is INCONCLUSIVE and
nothing is sent); after the turn, READ1 (`--wait1` s after the exec exited, longer than the 10 s disconnect grace period) and READ2 (`--wait2` s after the last exec exited) must show exactly the expected number of
call_logs rows and usage_history rows in the window. More rows than expected means other traffic: INCONCLUSIVE, never a pass. A scratch CODEX_HOME holds only the repository's omniroute profile and the documented
loopback placeholder key; nothing in a real Codex home is touched. Only metadata columns of the gateway store are read (mode=ro).
SHAPE: astra-max (one request, the final-request shape: the client closes right after response.completed) | sol-xhigh (one request) | two-request (a tool call, then the final answer: 2 requests)
--expect observe | all-logged   (observe prints what happened; all-logged gives PASS/FAIL/INCONCLUSIVE)
exit code: 0 when every verdict is PASS or OBSERVED, 1 when any is FAIL, 2 when none failed but any is INCONCLUSIVE.
"""
import argparse
import datetime as dt
import hashlib
import os
import sqlite3
import subprocess
import sys
import time
from pathlib import Path

HOME = Path.home()
DB = HOME / ".local/share/omniroute/storage.sqlite"
SCRATCH = HOME / ".cache/codex-home-omni-acceptance"
SHAPES = {
    "astra-max": (["-m", "cx/gpt-6-astra", "-c", 'model_reasoning_effort="max"'], "Reply with: ok", 1),
    "sol-xhigh": (["-m", "cx/gpt-6.1-sol", "-c", 'model_reasoning_effort="xhigh"'], "Reply with: ok", 1),
    "two-request": (["-m", "cx/gpt-6-astra", "-c", 'model_reasoning_effort="max"'], "Run the shell command `echo hi`, then reply with: ok", 2),
}


def utc(ts=None):
    return (ts or dt.datetime.now(dt.timezone.utc)).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def con_ro():
    return sqlite3.connect(f"file:{DB}?mode=ro", uri=True)


def counts(since, until=None):
    con = con_ro()
    until = until or utc()
    calls = con.execute("select timestamp, path, status, model, tokens_in, tokens_out from call_logs where timestamp >= ? and timestamp <= ? order by timestamp", (since, until)).fetchall()
    usage = con.execute("select timestamp, model, status, tokens_input, tokens_output from usage_history where timestamp >= ? and timestamp <= ? order by timestamp", (since, until)).fetchall()
    return calls, usage


def other_workers():
    out = subprocess.run(["pgrep", "-af", "omniroute-codex-sdk/worker.py|codex exec"], capture_output=True, text=True).stdout.splitlines()
    mine = str(os.getpid())
    return [line for line in out if not line.startswith(mine + " ") and "pgrep" not in line and "native-exec-acceptance" not in line]


def wait_quiet(timeout):
    end = time.time() + timeout
    while True:
        recent, _ = counts(utc(dt.datetime.now(dt.timezone.utc) - dt.timedelta(seconds=120)))
        workers = other_workers()
        if not recent and not workers:
            return True, 0, 0
        if time.time() >= end:
            return False, len(recent), len(workers)
        time.sleep(10)


def main():
    ap = argparse.ArgumentParser(add_help=False)
    ap.add_argument("--expect", choices=["observe", "all-logged"], default="observe")
    ap.add_argument("--wait1", type=float, default=12.0)
    ap.add_argument("--wait2", type=float, default=60.0)
    ap.add_argument("--quiet-timeout", type=float, default=900.0)
    ap.add_argument("--repo", default=str(HOME / "code/native-agent-stack"))
    ap.add_argument("shapes", nargs="+")
    args = ap.parse_args()
    me = open(os.path.abspath(__file__), "rb").read()
    print(f"FROZEN script_sha256={hashlib.sha256(me).hexdigest()} argv={sys.argv[1:]} cwd={os.getcwd()} start={utc()} expect={args.expect} wait1={args.wait1} wait2={args.wait2} quiet_timeout={args.quiet_timeout}")
    template = Path(args.repo) / "adoption/templates/codex.omniroute.config.toml"
    SCRATCH.mkdir(parents=True, exist_ok=True)
    (SCRATCH / "config.toml").write_text("# scratch home for the omniroute-profile acceptance\n")
    (SCRATCH / "omniroute.config.toml").write_text(template.read_text())
    env = {k: v for k, v in os.environ.items() if not k.startswith(("OPENAI", "OMNIROUTE", "ANTHROPIC", "CODEX"))}
    env.update({"CODEX_HOME": str(SCRATCH), "OMNIROUTE_API_KEY": "local-loopback"})  # the documented placeholder (docs/secret-storage.md)
    results = []
    for shape in args.shapes:
        extra, prompt, expected = SHAPES[shape]
        quiet, n_recent, n_workers = wait_quiet(args.quiet_timeout)
        if not quiet:
            print(f"RESULT shape={shape} quiet=False recent_call_logs={n_recent} other_workers={n_workers} verdict=INCONCLUSIVE (not run: the window was not quiet)")
            results.append("INCONCLUSIVE")
            continue
        since = utc()
        started = time.time()
        cmd = ["codex", "exec", "-p", "omniroute", "--skip-git-repo-check", "--ephemeral", *extra, prompt]
        proc = subprocess.run(cmd, cwd=str(SCRATCH), env=env, capture_output=True, text=True, timeout=600)
        ended = time.time()
        time.sleep(args.wait1)
        calls, usage = counts(since)
        verdict = "OBSERVED"
        if args.expect == "all-logged":
            if len(calls) == expected and len(usage) == expected and all(c[2] == 200 for c in calls):
                verdict = "PASS"
            elif len(calls) > expected or len(usage) > expected:
                verdict = "INCONCLUSIVE"
            else:
                verdict = "FAIL"
        print(
            f"RESULT shape={shape} quiet=True exec_exit={proc.returncode} elapsed_s={ended - started:.1f} expected_each={expected} read1_call_logs={len(calls)} read1_usage={len(usage)} verdict={verdict}"
        )
        print("   stdout tail:", proc.stdout.strip().replace("\n", " | ")[-200:])
        for row in calls:
            print("   call_logs:", row)
        for row in usage:
            print("   usage_history:", row)
        results.append({"shape": shape, "since": since, "ended": ended, "calls": len(calls), "usage": len(usage), "verdict": verdict})
    done = [r for r in results if isinstance(r, dict)]
    if done:
        time.sleep(max(0.0, max(r["ended"] for r in done) + args.wait2 - time.time()))
        for r in done:
            calls, usage = counts(r["since"])
            changed = (len(calls), len(usage)) != (r["calls"], r["usage"])
            if changed and r["verdict"] == "PASS":
                r["verdict"] = "FAIL"
            print(f"LATE shape={r['shape']} read2_call_logs={len(calls)} read2_usage={len(usage)} changed={changed} verdict={r['verdict']}")
    verdicts = [r["verdict"] if isinstance(r, dict) else r for r in results]
    summary = {v: verdicts.count(v) for v in sorted(set(verdicts))}
    code = 1 if "FAIL" in verdicts else (2 if "INCONCLUSIVE" in verdicts else 0)
    print(f"SUMMARY {summary} end={utc()} exit={code}")
    sys.exit(code)


main()
