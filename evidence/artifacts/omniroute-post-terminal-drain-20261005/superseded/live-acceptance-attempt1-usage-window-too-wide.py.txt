#!/usr/bin/env python3
"""Live acceptance of the post-terminal drain on NativeStack's 20128 (frozen before the canary switch; see checks/live-acceptance-protocol.md).

Each MODE sends one tiny streamed request to the gateway (documented loopback placeholder key), reads the response's X-Correlation-Id and, for Responses, the response id, closes the socket as the mode says,
and then reads only metadata columns of the gateway store (mode=ro). Rows are attributed to the request by the correlation id (call_logs) and by the same model and token counts inside the request's
own time window (usage_history has no correlation id); if more than one usage row matches, the verdict is INCONCLUSIVE, never a pass. The totals of both tables in the window are printed, so a reader can see other lanes' traffic.
Rows are read twice: READ1 `--wait1` seconds after the request ended (the default is longer than the 10 s disconnect grace period), and READ2 `--wait2` seconds after the last request ended (a late second row is visible).

MODE:
  responses:eof            /v1/responses, read the stream until the server ends it (a normal client)
  responses:early:<sec>    /v1/responses, close the socket <sec> seconds after the data line that carries response.completed (what `codex exec` does when it exits)
  chat:eof                 /v1/chat/completions (a non-Responses client), read to the end
  chat:early:<sec>         /v1/chat/completions, close <sec> seconds after the line `data: [DONE]`
  effort:<model>[:<eff>]   /v1/responses, read to the end, with client reasoning effort <eff> when given; prints the effort columns of the call-log row
--expect observe           print what is observed, no verdict (the baseline on the unpatched gateway)
--expect all-logged        verdict per mode: PASS needs exactly one call_logs row (status 200) and exactly one usage_history row, in READ1 and READ2
--baseline FILE            with all-logged, an effort mode also needs the same effort columns as the EFFORT line of the baseline file
exit code: 0 when every verdict is PASS or OBSERVED, 1 when any is FAIL, 2 when none failed but any is INCONCLUSIVE.
"""
import argparse
import datetime as dt
import hashlib
import http.client
import json
import os
import re
import sqlite3
import sys
import time

DB = os.path.expanduser("~/.local/share/omniroute/storage.sqlite")
DEFAULT_MODEL = "cx/gpt-6-astra"


def utc(ts=None):
    return (ts or dt.datetime.now(dt.timezone.utc)).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def con_ro():
    return sqlite3.connect(f"file:{DB}?mode=ro", uri=True)


def parse_mode(mode):
    parts = mode.split(":")
    if parts[0] in ("responses", "chat") and len(parts) >= 2:
        if parts[1] == "eof":
            return {"kind": parts[0], "model": DEFAULT_MODEL, "effort": None, "close_after": None}
        if parts[1] == "early" and len(parts) == 3:
            return {"kind": parts[0], "model": DEFAULT_MODEL, "effort": None, "close_after": float(parts[2])}
    if parts[0] == "effort" and len(parts) >= 2:
        model = parts[1]
        return {"kind": "responses", "model": model, "effort": parts[2] if len(parts) > 2 else None, "close_after": None}
    raise SystemExit(f"unknown mode {mode!r}")


def send(spec, port):
    conn = http.client.HTTPConnection("127.0.0.1", port, timeout=300)
    if spec["kind"] == "responses":
        path = "/v1/responses"
        body = {"model": spec["model"], "input": "Reply with: ok", "stream": True}
        body["reasoning"] = {"effort": spec["effort"] or "max"} if spec["model"] == DEFAULT_MODEL or spec["effort"] else None
        if body["reasoning"] is None:
            del body["reasoning"]
    else:
        path = "/v1/chat/completions"
        body = {"model": spec["model"], "messages": [{"role": "user", "content": "Reply with: ok"}], "stream": True}
    conn.request("POST", path, json.dumps(body), {"Content-Type": "application/json", "Authorization": "Bearer local-loopback", "Accept": "text/event-stream"})
    resp = conn.getresponse()
    out = {"http": resp.status, "corr": resp.getheader("X-Correlation-Id"), "response_id": None, "terminal": False, "closed_early": False, "path": path}
    t0 = time.time()
    while True:
        line = resp.fp.readline()
        if not line:
            break
        is_terminal = False
        if spec["kind"] == "responses" and line.startswith(b"data:") and b'"response.completed"' in line:
            is_terminal = True
            try:
                out["response_id"] = json.loads(line[5:].strip())["response"]["id"]
            except Exception:
                pass
        elif spec["kind"] == "chat" and line.strip() == b"data: [DONE]":
            is_terminal = True
        if is_terminal and not out["terminal"]:
            out["terminal"] = True
            if spec["close_after"] is not None:
                time.sleep(spec["close_after"])
                conn.close()
                out["closed_early"] = True
                break
    if not out["closed_early"]:
        conn.close()
    out["stream_s"] = round(time.time() - t0, 1)
    return out


def read_rows(req):
    con = con_ro()
    calls = con.execute(
        "select timestamp, path, status, model, requested_model, tokens_in, tokens_out, detail_state, reasoning_effort_requested, reasoning_effort_upstream from call_logs "
        "where correlation_id = ? or (response_id is not null and response_id = ?) order by timestamp",
        (req["corr"], req["response_id"]),
    ).fetchall()
    usage = []
    ambiguous = False
    if calls:
        tin, tout, model = calls[0][5], calls[0][6], calls[0][3]
        usage = con.execute(
            "select timestamp, model, status, tokens_input, tokens_output from usage_history where timestamp >= ? and timestamp <= ? and model = ? and tokens_input = ? and tokens_output = ? order by timestamp",
            (req["t_start"], req["t_read_end"], model, tin, tout),
        ).fetchall()
        ambiguous = len(usage) > 1
    candidates = con.execute(
        "select count(*) from usage_history where timestamp >= ? and timestamp <= ?", (req["t_start"], req["t_read_end"])
    ).fetchone()[0]
    window_calls = con.execute("select count(*) from call_logs where timestamp >= ? and timestamp <= ?", (req["t_start"], req["t_read_end"])).fetchone()[0]
    return {"calls": calls, "usage": usage, "ambiguous": ambiguous, "window_calls": window_calls, "window_usage": candidates}


def verdict(read, expect):
    if expect == "observe":
        return "OBSERVED"
    if read["ambiguous"]:
        return "INCONCLUSIVE"
    if len(read["calls"]) == 1 and read["calls"][0][2] == 200 and len(read["usage"]) == 1:
        return "PASS"
    return "FAIL"


def parse_baseline(path):
    base = {}
    for line in open(path, encoding="utf-8", errors="replace"):
        m = re.match(r"EFFORT mode=(\S+) model=(\S*) requested_model=(\S*) effort_requested=(\S*) effort_upstream=(\S*)", line)
        if m:
            base[m.group(1)] = m.groups()[1:]
    return base


def main():
    ap = argparse.ArgumentParser(add_help=False)
    ap.add_argument("--port", type=int, default=20128)
    ap.add_argument("--expect", choices=["observe", "all-logged"], default="observe")
    ap.add_argument("--baseline")
    ap.add_argument("--wait1", type=float, default=12.0)
    ap.add_argument("--wait2", type=float, default=60.0)
    ap.add_argument("modes", nargs="+")
    args = ap.parse_args()
    me = open(os.path.abspath(__file__), "rb").read()
    print(f"FROZEN script_sha256={hashlib.sha256(me).hexdigest()} argv={sys.argv[1:]} cwd={os.getcwd()} start={utc()} expect={args.expect} wait1={args.wait1} wait2={args.wait2} port={args.port}")
    base = parse_baseline(args.baseline) if args.baseline else {}
    reqs = []
    for mode in args.modes:
        spec = parse_mode(mode)
        t_start = utc()
        out = send(spec, args.port)
        t_end = time.time()
        time.sleep(args.wait1)
        out.update({"mode": mode, "spec": spec, "t_start": t_start, "t_read_end": utc(), "end_epoch": t_end})
        out["read1"] = read_rows(out)
        reqs.append(out)
        r = out["read1"]
        v = verdict(r, args.expect)
        if v == "PASS" and mode.startswith("effort:") and args.baseline:
            row = r["calls"][0]
            now = (str(row[3]), str(row[4]), str(row[8]), str(row[9]))
            if mode not in base:
                v = "INCONCLUSIVE"
            elif tuple(base[mode]) != now:
                v = "FAIL"
        out["verdict"] = v
        print(
            f"RESULT mode={mode} http={out['http']} terminal={out['terminal']} closed_early={out['closed_early']} stream_s={out['stream_s']} corr={out['corr']} response_id={out['response_id']} "
            f"read1_call_logs={len(r['calls'])} read1_usage={len(r['usage'])} window_call_logs={r['window_calls']} window_usage={r['window_usage']} verdict={v}"
        )
        for row in r["calls"]:
            print("   call_logs:", row)
        for row in r["usage"]:
            print("   usage_history:", row)
        if mode.startswith("effort:") and r["calls"]:
            row = r["calls"][0]
            print(f"EFFORT mode={mode} model={row[3]} requested_model={row[4]} effort_requested={row[8]} effort_upstream={row[9]}")
    last = max(r["end_epoch"] for r in reqs)
    time.sleep(max(0.0, last + args.wait2 - time.time()))
    for out in reqs:
        out["t_read_end"] = utc()
        late = read_rows(out)
        r1 = out["read1"]
        changed = (len(late["calls"]), len(late["usage"])) != (len(r1["calls"]), len(r1["usage"]))
        out["late_changed"] = changed
        if changed and out["verdict"] == "PASS":
            out["verdict"] = "FAIL"
        print(f"LATE mode={out['mode']} read2_call_logs={len(late['calls'])} read2_usage={len(late['usage'])} changed={changed} verdict={out['verdict']}")
    counts = {}
    for out in reqs:
        counts[out["verdict"]] = counts.get(out["verdict"], 0) + 1
    code = 1 if counts.get("FAIL") else (2 if counts.get("INCONCLUSIVE") else 0)
    print(f"SUMMARY {json.dumps(counts, sort_keys=True)} end={utc()} exit={code}")
    sys.exit(code)


main()
