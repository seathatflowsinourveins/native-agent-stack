+def pair_counts(edges, n_lines):

+TIME_UNITS = {"usec": 1e-6, "us": 1e-6, "\u00b5s": 1e-6, "\u03bcs": 1e-6, "msec": 1e-3, "ms": 1e-3, "seconds": 1.0, "second": 1.0, "sec": 1.0, "s": 1.0, "minutes": 60.0, "minute": 60.0, "min": 60.0, "m": 60.0,

+              "months": 2629800.0, "month": 2629800.0, "M": 2629800.0, "years": 31557600.0, "year": 31557600.0, "y": 31557600.0}   # systemd.time(7): a month is 30.4375 days, a year 365.25 days; the unit letters are case-sensitive (M is a month, m a minute)
+TIME_PART = r"[0-9]+(?:\.[0-9]+)?\s*(?:" + "|".join(re.escape(k) for k in sorted(TIME_UNITS, key=len, reverse=True)) + r")"
+TIME_SPAN = re.compile(r"\s*(?:" + TIME_PART + r"\s*)+")
+TIME_PIECE = re.compile(r"([0-9]+(?:\.[0-9]+)?)\s*(" + "|".join(re.escape(k) for k in sorted(TIME_UNITS, key=len, reverse=True)) + r")")

+def seconds_of(text):
+    """systemd.time(7) span such as `30s`, `1min 30s`, `500ms`, `1d`, `1w`, `1M`, `0` -> seconds; None when any part of it is not one. The whole value must be consumed (`30 foo` and `1.5.5s` are refused), the digits are ASCII, the unit letters
+    are case-sensitive, and a bare number (seconds) is a span only when it is the whole value."""
+    t = text.strip()
+    if re.fullmatch(r"[0-9]+(?:\.[0-9]+)?", t):
+        return float(t)
+    if not TIME_SPAN.fullmatch(t):
+        return None
+    return sum(float(number) * TIME_UNITS[unit] for number, unit in TIME_PIECE.findall(t))

+def journal_limits(unit):
+    """(StandardOutput, StandardError, interval s, burst, source, error): the unit's stream targets and the rate limit that journald applies to it. journald drops the lines of a service that logs more than RateLimitBurst messages in RateLimitIntervalSec and says so only in the SYSTEM journal (which this user cannot read);
+    the unit's own LogRateLimitIntervalSec/LogRateLimitBurst override journald.conf (merged here with `systemd-analyze cat-config`), the compiled-in defaults are 30 s and 10000."""
+    try:
+        shown = subprocess.run(["systemctl", "--user", "show", unit, "-p", "StandardOutput", "-p", "StandardError", "-p", "LogRateLimitIntervalUSec", "-p", "LogRateLimitBurst"], capture_output=True, timeout=20)
+        merged = subprocess.run(["systemd-analyze", "cat-config", "systemd/journald.conf"], capture_output=True, timeout=20)
+    except (OSError, subprocess.TimeoutExpired) as error:
+        return None, None, None, None, None, f"the unit's journal settings cannot be read ({type(error).__name__})"
+    if shown.returncode != 0 or merged.returncode != 0:
+        return None, None, None, None, None, f"the unit's journal settings cannot be read (systemctl exit {shown.returncode}, systemd-analyze exit {merged.returncode})"
+    props = dict(line.split("=", 1) for line in shown.stdout.decode("utf-8", "replace").splitlines() if "=" in line)
+    interval, burst, source = 30.0, DEFAULT_RATE_BURST, "the compiled-in default"
+    for line in merged.stdout.decode("utf-8", "replace").splitlines():
+        m = re.match(r"^\s*(RateLimitIntervalSec|RateLimitInterval|RateLimitBurst)\s*=\s*(.*?)\s*$", line)   # RateLimitInterval= is the deprecated name of RateLimitIntervalSec=
+        if not m:
+            continue
+        if m.group(1) in ("RateLimitIntervalSec", "RateLimitInterval"):

+        source = "journald.conf"

+        return None, None, None, None, None, f"the unit's LogRateLimit settings are not numbers ({raw_interval[:20]!r}, {raw_burst[:20]!r})"
+    if unit_interval or int(raw_burst):

+    return props.get("StandardOutput"), props.get("StandardError"), interval, burst, source, None

+def peak_rate(stamps, interval):
+    """The most entries inside any window of `interval` seconds (stamps: datetimes)."""
+    ordered = sorted(stamps)
+    best, low = 0, 0
+    for high, stamp in enumerate(ordered):
+        while (stamp - ordered[low]).total_seconds() > interval:
+            low += 1
+        best = max(best, high - low + 1)
+    return best

+def incarnation_unmatched(arrivals, rows, tau):
+    """The arrival lines that a maximum matching leaves without a call row. arrivals: [(time, label, path)], rows: [(estimated start, path)]. An arrival may be matched to any call row of its path whose estimated start (its timestamp minus its duration) is not more than tau
+    before the arrival, whatever the arrival's lead over that clock (the line is written before the policy work, the credential refresh and the duration clock, by a delay that nothing bounds); each row serves one arrival. The sets of rows that can serve an arrival are nested
+    (a later arrival can use only the rows that an earlier one can use), so the greedy order from the latest arrival, taking any available row, is a maximum matching: the leftovers are the exact count (the controls check it against a brute force)."""
+    leftovers = []
+    for path in sorted({p for _t, _l, p in arrivals}, key=str):
+        latest_first = sorted((a for a in arrivals if a[2] == path), key=lambda a: a[0], reverse=True)
+        starts = sorted(s for s, p in rows if p == path)
+        j, pool = len(starts), 0
+        for t, label, _p in latest_first:
+            threshold = t - dt.timedelta(seconds=tau)
+            while j > 0 and starts[j - 1] >= threshold:
+                j -= 1
+                pool += 1
+            if pool == 0:
+                leftovers.append((t, label, path))
+            else:
+                pool -= 1
+    return sorted(leftovers, key=lambda a: a[0])
