#!/usr/bin/env python3
"""Plan P post-apply checks for the two OmniRoute gateways (our integration probe, not an upstream tool).

No model call, no gateway write. GET only to 127.0.0.1:20128 / 127.0.0.1:20129 on routes whose GET handler was read at
c19983a66 and returns no credential material (never /api/settings, /api/providers, /api/providers/client, /api/keys,
/api/settings/cache-config). sqlite reads run mode=ro with PRAGMA query_only on named non-secret keys and named
provider_specific_data paths only; values that could hold secrets (custom header values, interception proxy URLs) are
compared in code and reported as booleans.

Every check derives its expectation from the set of plan steps that are applied, so a held step is checked in its
pre-apply state instead of failing:
  post_apply_checks.py --applied T01,T02,...     # exactly the steps gateway_apply.py --verify reports as applied
  post_apply_checks.py --phase pre               # nothing applied
  post_apply_checks.py --phase mid               # T01-T08 applied, T09 onwards held
  post_apply_checks.py --phase t09               # T01-T09 applied (the state observed 2026-09-30T00:25Z)
  post_apply_checks.py --phase post              # T01-T15 applied
Exit code 0 when every check matches, 1 otherwise.
"""
import argparse
import json
import sqlite3
import sys
import urllib.error
import urllib.request
from pathlib import Path

STORES = {"20128": Path.home() / ".local/share/omniroute/storage.sqlite",
          "20129": Path.home() / ".local/share/omniroute-fw/storage.sqlite"}
FORBIDDEN = ("/api/settings", "/api/providers", "/api/providers/client", "/api/keys", "/api/settings/cache-config")
NODE = "openai-compatible-responses-3393b3bd…"
CONN = "3eea0518…"
OLD_COMBOS = {"default-caveman", "allow-lossy", "fw-session-dedup", "fw-ccr", "fw-lite", "fw-rtk", "fw-codex-responses",
              "fw-headroom"}
COMBO_OF = {"T02": "gpt6-agent-safe", "T03": "gpt6-memory-safe", "T04": "gpt6-safe-lossy",
            "T12": "gpt6-agent-safe", "T13": "gpt6-memory-safe", "T14": "gpt6-safe-lossy"}
HDRS = {"x-omniroute-no-memory": "true", "x-omniroute-disabled-guardrails": "vision-bridge,audio-bridge,video-bridge",
        "x-omniroute-no-cache": "true"}
ENGINES_ON = ["caveman", "ccr", "codex-responses", "rtk"]
STEPS = [f"T{i:02d}" for i in range(1, 16)]
PHASES = {"pre": [], "mid": STEPS[:8], "t09": STEPS[:9], "post": STEPS}


def get(port, route):
    if route.split("?")[0] in FORBIDDEN:
        raise SystemExit(f"refused forbidden route {route}")
    try:
        with urllib.request.urlopen(f"http://127.0.0.1:{port}{route}", timeout=30) as r:
            return r.status, json.loads(r.read() or b"null")
    except urllib.error.HTTPError as e:
        return e.code, None
    except Exception as e:  # noqa: BLE001
        return f"error-{type(e).__name__}", None


def rows(port, sql, params=()):
    con = sqlite3.connect(STORES[port].resolve().as_uri() + "?mode=ro", uri=True, timeout=5)
    try:
        con.execute("PRAGMA query_only = ON")
        return con.execute(sql, params).fetchall()
    finally:
        con.close()


def setting(port, key):
    r = rows(port, "SELECT value FROM key_value WHERE namespace = 'settings' AND key = ?", (key,))
    return json.loads(r[0][0]) if r else "<absent>"


def flag(port, key):
    st, b = get(port, "/api/settings/feature-flags")
    for f in (b or {}).get("flags", []) if isinstance(b, dict) else []:
        if f.get("key") == key:
            return f.get("effectiveValue"), f.get("source")
    return None, st


def engines_on(c):
    return sorted(k for k, v in (c.get("engines") or {}).items() if isinstance(v, dict) and v.get("enabled"))


def safest_globals(c):
    rk, cv = c.get("rtkConfig") or {}, c.get("cavemanConfig") or {}
    observed = {"rtk.disabledFilters": len(rk.get("disabledFilters") or []), "rtk.maxCharsPerResult": rk.get("maxCharsPerResult"),
                "caveman.compressRoles": cv.get("compressRoles"), "sessionDedup": c.get("sessionDedup"),
                "headroom": c.get("headroom")}
    ok = (len(rk.get("disabledFilters") or []) == 55 and rk.get("maxCharsPerResult") == 40000
          and cv.get("compressRoles") == ["assistant"] and (c.get("sessionDedup") or {}).get("minBlockChars") == 512
          and (c.get("headroom") or {}).get("minRows") == 16)
    return ok, observed

    add("20129", "legacy output mode, Auto-Clarity bypass [T10]",

        else (lz == {"enabled": False}), {"liveZone": lz, "languageConfig": lc})

        {"row": bool(ir), "interceptSearch": (parsed or {}).get("interceptSearch"),

                 "present" if setting(port, "noAuthFallbackDisabledProviders") != "<absent>" else "absent"})

def main():
    ap = argparse.ArgumentParser()
    group = ap.add_mutually_exclusive_group()
    group.add_argument("--phase", choices=sorted(PHASES))
    group.add_argument("--applied", help="comma-separated applied step ids (use gateway_apply.py --verify to list them)")
    args = ap.parse_args()
    if args.applied is not None:
        applied = [s for s in args.applied.split(",") if s]
        unknown = sorted(set(applied) - set(STEPS))
        if unknown:
            raise SystemExit(f"unknown step ids {unknown}")
        label = "custom"
    else:
        label = args.phase or "post"
        applied = PHASES[label]
    results = checks(applied)
    bad = 0
    for port, name, ok, observed in results:
        bad += 0 if ok else 1
        print(f"{'PASS' if ok else 'FAIL'} [{port}] {name}: {json.dumps(observed, default=str)[:300]}")
    print(f"phase={label} applied={','.join(applied) or '-'} checks={len(results)} failed={bad}")
    sys.exit(1 if bad else 0)


if __name__ == "__main__":
    main()
