def stamp():
    return subprocess.check_output(['date', '-u', '+%Y-%m-%dT%H:%M:%SZ'], text=True).strip()


def report(label, expectation, result, output):
    caught = not result.wasSuccessful()
    observed = 'caught' if caught else 'survives'
    lines = [line for line in output.getvalue().splitlines()
             if line.startswith(('AssertionError', 'FAIL:', 'ERROR:')) or line.endswith((' ... ok', ' ... FAIL', ' ... ERROR'))]
    print(f'{stamp()} {label}: expected {expectation}, observed {observed}; '
          f'ran {result.testsRun}, failures {len(result.failures)}, errors {len(result.errors)}')
    for line in lines:
        print('   ' + line)
    return observed == expectation


def run_config(label, expectation, config, tests):
    output = io.StringIO()
    with tempfile.TemporaryDirectory() as directory:
        config_path = Path(directory) / 'osv-scanner.toml'
        config_path.write_text(module.CONFIG.read_text(encoding='utf-8'), encoding='utf-8')
        with patch.object(module, 'CONFIG', config_path), \
             patch.object(module.FrozenScanTests, 'ordinary_config', config), \
             patch.object(module.tomllib, 'loads', wraps=module.tomllib.loads) as loads:
            loads.side_effect = lambda text: config if text == config_path.read_text() else loads._mock_wraps(text)
            result = unittest.TextTestRunner(stream=output, verbosity=2).run(unittest.TestSuite(tests))
    return report(label, expectation, result, output)
