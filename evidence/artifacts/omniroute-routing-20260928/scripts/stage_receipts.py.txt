"""Stage the receipts of evidence/artifacts/omniroute-routing-20260928/ from private inputs.

Run from the repository root:

  SCRATCH=<session scratch directory> WF_JOURNAL=<journal.jsonl of the record workflow> \
    AB_LABELS=<peer session A>,<peer session B>,<Codex home label> \
    uv run --no-project --with jsonschema python \
    evidence/artifacts/omniroute-routing-20260928/scripts/stage_receipts.py.txt

It reads only the inputs named below and writes only this receipt directory. It prints a
JSON summary: the source and output SHA-256 of every file, the substitution counts, the
split of route-mapper-resume-parts.md and the private-content scan result. The scan reuses
scripts/validate.py's PRIVATE_CONTENT patterns and adds emails, the home and scratch
directory strings, the opaque ids of the private limiter record and the three local labels
that ab-requirements.md replaces; it reports labels and counts, never a matched value.
Every generated file is pinned in EXPECTED: a file is written only when its bytes equal the
pin, so a changed private input cannot silently replace a receipt that the record cites. It
exits 1 on any scan finding or pin mismatch. jsonschema is optional: without it the schema
check is "not run".

AB_LABELS holds the local labels of two peer Claude sessions and one Codex home directory,
comma-separated in that order, as they appear in the private ab-requirements.md. They come
from the environment so that this script does not publish them, and the substitution
summary prints a description in place of each private literal.
"""

summary = {"copies": [], "substituted": [], "split": [], "derived": [], "unpublished": []}

def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def source_name(path: Path) -> str:
    return "$SCRATCH/" + path.relative_to(SCRATCH).as_posix()


def emit(name: str, raw: bytes) -> str:
    """Write a receipt only when its bytes equal its pin in EXPECTED."""
    digest = sha(raw)
    if EXPECTED.get(name) != digest:
        mismatches.append(f"{name}: output differs from its pin; not written")
    else:
        (OUT / name).write_bytes(raw)
    return digest


def check_source(name: str, raw: bytes) -> None:
    if name in EXPECTED_SOURCE and EXPECTED_SOURCE[name] != sha(raw):
        mismatches.append(f"{name}: source differs from its pin")


def copy(src: Path, name: str) -> None:
    raw = src.read_bytes()
    summary["copies"].append({"output": name, "source": source_name(src), "sha256": emit(name, raw)})


def substitute(src: Path, name: str, rules) -> None:
    raw = src.read_bytes()
    check_source(name, raw)
    text = raw.decode("utf-8")
    counts = []
    for rule in rules:
        pattern, replacement = rule[0], rule[1]
        text, count = pattern.subn(replacement, text)
        counts.append({"pattern": rule[2] if len(rule) > 2 else pattern.pattern, "replacement": replacement,
                       "count": count})
    out = text.encode("utf-8")
    summary["substituted"].append({"output": name, "source": source_name(src), "source_sha256": sha(raw),
                                   "sha256": emit(name, out), "substitutions": counts})


def split(src: Path, name: str, rest_name: str, first_bytes: int) -> None:
    """Keep the first first_bytes bytes as name and the rest as rest_name, byte for byte."""
    raw = src.read_bytes()
    check_source(name, raw)
    first, rest = raw[:first_bytes], raw[first_bytes:]
    summary["split"].append({"source": source_name(src), "source_sha256": sha(raw), "source_bytes": len(raw),
                             "parts": [{"output": name, "bytes": len(first), "sha256": emit(name, first)},
                                       {"output": rest_name, "bytes": len(rest), "sha256": emit(rest_name, rest)}]})


def write_json(name: str, document, compact: bool) -> None:
    if compact:
        text = json.dumps(document, ensure_ascii=False, separators=(",", ":"))
    else:
        text = json.dumps(document, ensure_ascii=False, indent=1)
    raw = (text + "\n").encode("utf-8")
    summary["derived"].append({"output": name, "sha256": emit(name, raw)})


def brace_strings(value, counter):
    if isinstance(value, dict):
        return {key: brace_strings(item, counter) for key, item in value.items()}
    if isinstance(value, list):
        return [brace_strings(item, counter) for item in value]
    if isinstance(value, str):
        text, count = BRACES[0].subn(BRACES[1], value)
        counter["count"] += count
        return text
    return value


def adjudication() -> None:
    records = [json.loads(line) for line in JOURNAL.read_text(encoding="utf-8").splitlines() if line.strip()]
    labels = {item["key"]: item["label"] for item in records if item.get("type") == "started"}
    results = {labels[item["key"]]: item["result"] for item in records if item.get("type") == "result"}
    counter = {"count": 0}
    groups = [brace_strings(results[label], counter) for label in GROUP_ORDER]
    items = [item for group in groups for item in group["items"]]
    ids = [item["id"] for item in items]
    assert len(ids) == len(set(ids)) == 51, len(ids)
    document = {
        "kind": "routing_practice_adjudication",
        "subject": "GPT-6 Codex account-pool lane behind OmniRoute: 20128 shared; 20129, the full-compression lane, chained to it through the sharedgw provider node",
        "source": "Result records of the workflow stages adjudicate:affinity-cache, adjudicate:resilience-retry and adjudicate:chain-stream-extra (Claude, model opus, agent type stack-researcher, effort max), read from the workflow's private journal. The rule they followed is adjudication-rule.json.",
        "recovery_note": "The builder's task text carried this table cut at 150,000 characters, inside E9. Every item here comes from the journal's complete result records.",
        "sanitization": "One change only: CLIProxyAPI's internal/home/<file> is written internal/home/{<file>} (%d substitutions), because scripts/validate.py treats /home/<name> as a personal home path. Removing the braces restores the original text." % counter["count"],
        "counts": {
            "items": len(items),
            "converged": dict(sorted(Counter(item["converged"] for item in items).items())),
            "action": dict(sorted(Counter(item["action"] for item in items).items())),
        },

    summary["derived"][-1]["brace_substitutions"] = counter["count"]

def limiter() -> list[str]:
    record = json.loads((OF / "limiter" / "apply.json").read_text(encoding="utf-8"))
    before, after = record["rate_limits_before"]["connections"], record["rate_limits_after"]["connections"]
    assert len(before) == len(after) == 1
    queue_before = record["resilience_before"]["requestQueue"]
    queue_after = record["resilience_after"]["requestQueue"]
    other_before = {key: value for key, value in record["resilience_before"].items() if key != "requestQueue"}
    other_after = {key: value for key, value in record["resilience_after"].items() if key != "requestQueue"}
    document = {
        "kind": "gateway_setting_change",
        "instance": "OmniRoute 20129, the full-compression lane, chained to 20128 through the sharedgw provider node",
        "applied_utc": record["applied_utc"],
        "request": {"method": "PATCH", "path": "/api/resilience", "body": record["patch"]},
        "requestQueue_before": queue_before,
        "requestQueue_after": queue_after,
        "requestQueue_changed_fields": sorted(key for key in queue_before if queue_before[key] != queue_after.get(key)),
        "waitForCooldown_before": record["resilience_before"]["waitForCooldown"],
        "waitForCooldown_after": record["resilience_after"]["waitForCooldown"],
        "other_resilience_settings_unchanged": other_before == other_after,
        "sharedgw_limiter_enabled_before": before[0]["enabled"],
        "sharedgw_limiter_enabled_after": after[0]["enabled"],
        "read_back": "GET /api/resilience and GET /api/rate-limits after the PATCH. The PATCH's HTTP 200 and the absence of an environment override come from the coordinator's log, not from this file.",
        "reduction": "Reduced from the private apply record: the requestQueue and waitForCooldown objects, the one connection's limiter enabled flag and the UTC time are kept. Dropped: the connection and provider ids, the connection name, the other resilience settings (compared equal above) and the rate-limit cache statistics.",
    }
    write_json("limiter-apply-20129.json", document, compact=False)
    return [value for connection in before + after for key, value in connection.items()
            if key in {"connectionId", "provider"} and isinstance(value, str)]


def scan(forbidden: list[str]) -> list[str]:
    sys.path.insert(0, "scripts")
    from validate import PRIVATE_CONTENT
    literals = [("home directory", str(Path.home())), ("scratch directory", str(SCRATCH)),
                ("journal path", str(JOURNAL))] + [("limiter record id", value) for value in forbidden] \
        + [("ab-requirements label", value) for value in AB_LABELS]
    findings = []
    for path in sorted(OUT.rglob("*")):
        if not path.is_file():
            continue
        text = path.read_bytes().decode("utf-8")
        for description, pattern in PRIVATE_CONTENT + (("email", EMAIL),):
            if pattern.search(text):
                findings.append(f"{path.relative_to(OUT)}: {description}")
        for description, value in literals:
            if value and value in text:
                findings.append(f"{path.relative_to(OUT)}: {description}")
    return findings

    copy(REVIEW / "gpt6" / "routing-record-review" / "schema.json", "record-review-gpt6.schema.json")
    for relative, pin in UNPUBLISHED.items():
        digest = sha((OF / relative).read_bytes())
        summary["unpublished"].append({"source": source_name(OF / relative), "sha256": digest})
        if digest != pin:
            mismatches.append(f"{relative}: unpublished input differs from its pin")

    findings = scan(forbidden)

    summary["scan"] = {"files": sum(1 for path in OUT.rglob("*") if path.is_file()), "findings": findings}
    print(json.dumps(summary, indent=1))
    return 1 if findings or mismatches else 0
