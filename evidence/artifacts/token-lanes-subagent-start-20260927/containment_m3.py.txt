#!/usr/bin/env python3
"""Context containment per workflow run (the M3/M5 idea of the token-adoption preregistration, #381):
tool results over 5,120 bytes entering an agent's context, by carrier. Counts and byte shares only; no content.

Usage: containment_m3.py <workflow transcript dir> [<dir> ...]
"""
import collections
import glob
import json
import os
import sys

LIMIT = 5120


def carrier(name):
    if not name:
        return "?"
    if name.startswith("mcp__plugin_context-mode"):
        return "ctx:" + name.rsplit("__", 1)[-1]
    if name.startswith("mcp__"):
        return "mcp:" + name.split("__")[1]
    return name


for run in sys.argv[1:]:
    by_carrier = collections.defaultdict(lambda: [0, 0, 0, 0])  # results, over-limit results, bytes, over-limit bytes
    largest = 0
    for meta in glob.glob(os.path.join(run, "*.meta.json")):
        names = {}
        for line in open(meta[: -len(".meta.json")] + ".jsonl", encoding="utf-8"):
            try:
                record = json.loads(line)
            except ValueError:
                continue
            content = (record.get("message") or {}).get("content")
            if not isinstance(content, list):
                continue
            for block in content:
                if not isinstance(block, dict):
                    continue
                if block.get("type") == "tool_use":
                    names[block.get("id")] = block.get("name")
                elif block.get("type") == "tool_result":
                    body = block.get("content")
                    size = len(json.dumps(body, ensure_ascii=False).encode("utf-8")) if not isinstance(body, str) \
                        else len(body.encode("utf-8"))
                    row = by_carrier[carrier(names.get(block.get("tool_use_id")))]
                    row[0] += 1
                    row[2] += size
                    if size > LIMIT:
                        row[1] += 1
                        row[3] += size
                    largest = max(largest, size)
    total = [sum(r[i] for r in by_carrier.values()) for i in range(4)]
    print(f"== {os.path.basename(run)}: results {total[0]}, over 5 KB {total[1]} ({100 * total[1] / max(total[0], 1):.1f}%), "
          f"bytes from over-5KB results {100 * total[3] / max(total[2], 1):.1f}%, largest {largest} B")
    for name, r in sorted(by_carrier.items(), key=lambda kv: -kv[1][3])[:8]:
        print(f"   {name[:34]:34} results {r[0]:5}  over5KB {r[1]:4}  bytes {r[2]:9}  over5KB-bytes {r[3]:9}")
