#!/usr/bin/env python3
"""Retain the observations the PR #423 review found uncommitted (read-only; no ids, emails or paths kept).

Usage: retain_repair_evidence.py <scratchpad> <evidence dir> <qmd evidence dir>
"""
import glob
import hashlib
import json
import os
import re
import sqlite3
import subprocess
import sys
from collections import Counter
from pathlib import Path

SP, OUT, QOUT = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3])
DB = os.path.expanduser("~/.local/share/omniroute/storage.sqlite")
con = sqlite3.connect(f"file:{DB}?mode=ro", uri=True)
COLS = "timestamp, path, status, model, reasoning_effort_requested, reasoning_effort_upstream, tokens_in, tokens_cache_read, tokens_reasoning"


def rows(sql, args):
    return [dict(zip([c.strip() for c in COLS.split(",")], r)) for r in con.execute(sql, args)]


def dump(path, obj):
    path.parent.mkdir(parents=True, exist_ok=True)
    text = json.dumps(obj, indent=1)
    text = text.replace(str(SP), "$SCRATCH").replace(str(Path.home()), "~")
    path.write_text(text + "\n")


# 1. Effort read-back of the 17:20Z probes (the same non-sensitive columns as effort_rows.py) and account spread.
probe = rows(f"SELECT {COLS} FROM call_logs WHERE path='/v1/chat/completions' AND timestamp >= ? AND timestamp < ? "
             "AND model LIKE 'gpt-6-astra%' ORDER BY timestamp", ("2026-09-27T17:20:00", "2026-09-27T17:22:00"))
spread = {}
for name, (a, b) in {"max_probes_17_20Z": ("2026-09-27T17:20:00", "2026-09-27T17:22:00")}.items():
    n = con.execute("SELECT COUNT(DISTINCT connection_id), COUNT(*) FROM call_logs WHERE path='/v1/chat/completions' AND "
                    "timestamp >= ? AND timestamp < ? AND model LIKE 'gpt-6-astra%' AND status=200", (a, b)).fetchone()
    spread[name] = {"distinct_accounts": n[0], "calls": n[1], "session_headers": "one distinct x-omniroute-session per call"}
cg = (SP / "cg/retain2-end.txt").read_text().strip().rstrip("Z")
n = con.execute("SELECT COUNT(DISTINCT connection_id), COUNT(*) FROM call_logs WHERE model='gpt-6-astra-max' AND status=200 "
                "AND timestamp >= ? AND timestamp < ?", (cg, "2026-09-27T17:20:00")).fetchone()
spread["cognee_smoke_tool_call_attempts_3_4"] = {"distinct_accounts": n[0], "calls": n[1], "session_headers": "one shared x-omniroute-session"}
dump(OUT / "probes/effort-readback-20260927T1720Z.json", {"source": "read-only SELECT of call_logs effort and token columns (no ids)", "rows": probe})
dump(OUT / "probes/account-spread.json", {"method": "COUNT(DISTINCT connection_id) only; no ids retained", "windows": spread})

# 2. Probe outputs as JSON arrays (*.jsonl is git-ignored in this repository).
for name in ["probe-max", "probe-max-hard"]:
    lines = [json.loads(line) for line in (SP / f"omniroute-phase0/{name}.jsonl").read_text().splitlines() if line.strip()]
    dump(OUT / f"probes/{name}.json", lines)
    old = OUT / f"probes/{name}.jsonl"
    if old.exists():
        old.unlink()

# 3. Tool counts of the failed first attempt's children, from their transcripts.
tools = Counter()
tdir = Path.home() / ".claude/projects/<project>/<session>/subagents/workflows/wf_baff0093-a39"
for f in glob.glob(str(tdir / "**/*.jsonl"), recursive=True):
    if f.endswith("journal.jsonl"):
        continue
    for line in open(f):
        try:
            m = json.loads(line).get("message") or {}
        except ValueError:
            continue
        for b in m.get("content") or [] if isinstance(m.get("content"), list) else []:
            if isinstance(b, dict) and b.get("type") == "tool_use":
                name = b.get("name", "")
                tools["mcp:" + name.split("__")[1] if name.startswith("mcp__") else name] += 1
total = sum(tools.values())
dump(OUT / "phase0-attempt1-tool-counts.json", {"run": "wf_baff0093-a39", "total_tool_calls": total,
     "by_tool": {k: {"calls": v, "share_pct": round(100 * v / total, 1)} for k, v in tools.most_common()}})

# 4. Error events of the timed-out synthesis job (messages only, paths masked).
errs = []
for line in (SP / "gpt6-lane/gpt6/synth-r1/events.jsonl").read_text().splitlines():
    try:
        o = json.loads(line)
    except ValueError:
        continue
    if o.get("type") == "error" or (o.get("item") or {}).get("type") == "error":
        msg = o.get("message") or (o.get("item") or {}).get("message", "")
        errs.append(re.sub(r"in /[^ ]+", "in <lane codex home>", msg)[:300])
dump(OUT / "synth-r1-errors.json", {"job": "synth-r1", "exit": 124, "cap_seconds": 3000, "error_events": errs})

# 5. Upstream issue #14866 closure (GitHub API).
issue = json.loads(subprocess.check_output(["gh", "api", "repos/diegosouzapw/OmniRoute/issues/14866",
                                            "--jq", "{state, closed_at, title}"], text=True))
comments = json.loads(subprocess.check_output(["gh", "api", "repos/diegosouzapw/OmniRoute/issues/14866/comments",
                                               "--jq", "[.[] | {created_at, author: .user.login, body: .body[0:300]}]"], text=True))
dump(OUT / "upstream-issue-14866.json", {"fetched_by": "gh api", "issue": issue, "last_comments": comments[-3:]})

# 6. qmd embedding and reranker model identity (sha256 and size).
ident = {}
for f in sorted(Path.home().joinpath(".cache/qmd/models").glob("*.gguf")):
    h = hashlib.sha256()
    with open(f, "rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    ident[f.name] = {"sha256": h.hexdigest(), "bytes": f.stat().st_size}
dump(QOUT / "qmd-model-identity.json", {"directory": "~/.cache/qmd/models", "models": ident})
print("probe rows", len(probe), "| spread", spread, "| tools", total, "| errors", len(errs), "| issue", issue.get("state"), issue.get("closed_at"))
