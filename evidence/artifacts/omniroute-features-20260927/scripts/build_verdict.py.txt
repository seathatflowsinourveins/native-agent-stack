"""Deterministic OmniRoute feature verdict from the verified records (no model call).

Final position per feature = its proposal as amended by its adversarial refutation (refuted -> the refuter's state
and config; corrected -> the refuter's config; upheld -> the proposal). Features whose refutation has not returned
are labelled `verification: pending`.
"""
import json
import sys
from pathlib import Path

P, G, OUT = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3])
STATE = {"ON-GLOBAL": "enabled_verified", "ON-OPT-IN": "enable_after_ab", "OFF-BY-EVIDENCE": "hold", "N/A": "not_applicable"}

proposals = {}
for d in json.load(open(P / "dives-round1.json")):
    proposals[d["id"]] = ("Claude dive", d)
for name in ["dives-gpt6-H1-H6.json", "dives-gpt6-H2-H5.json", "dives-gpt6-H7.json"]:
    for d in json.load(open(P / name)):
        proposals[d["id"]] = ("GPT-6 dive", d)

refutes = {}
for g, o in json.load(open(P / "gpt6-refutations.json")).items():
    for it in o["items"]:
        refutes[it["id"]] = ("GPT-6 refutation " + g, it)
for job in sorted((G / "gpt6-final").glob("refute-RH*")):
    last = job / "last.json"
    if last.exists():
        for it in json.load(open(last))["items"]:
            refutes[it["id"]] = ("GPT-6 refutation " + job.name.split("-")[1], it)


def short(text, n=700):
    text = " ".join(str(text or "").split())
    return text if len(text) <= n else text[:n - 1] + "…"


features = []
for fid in sorted(proposals):
    who, d = proposals[fid]
    ref = refutes.get(fid)
    state = d["proposed_state"]
    config = d.get("proposed_config", "")
    sources = [c["ref"] for c in d.get("citations", [])][:8]
    if ref:
        how, it = ref
        if it["verdict"] in ("refuted", "corrected"):
            state = it["state"]
            config = it.get("config") or config
        sources += [f["citation"] for f in it["findings"] if f["material"]][:4]
        verification = f"{who}; {how}: {it['verdict']}" + (
            " — " + short(next((f['claim'] for f in it['findings'] if f['material']), ""), 240) if it["verdict"] != "upheld" else "")
    else:
        verification = f"{who}; adversarial refutation pending"
    s = STATE[state]
    features.append({
        "id": fid, "name": d["feature"], "description": short(d.get("what_it_does"), 500), "state": s,
        "lanes": d.get("applies_to", ""),
        "enable": short(config, 900) if s in ("enable_after_ab", "enable_now", "enabled_verified") else "",
        "ab_design": short(d.get("experiment_needed"), 700) if s == "enable_after_ab" else "",
        "hold_reason": short(d.get("quality_risk"), 700) if s in ("hold", "not_applicable") else "",
        "keep_config": short(config, 700) if s in ("hold", "not_applicable") else "",
        "cache_effect": short(d.get("cache_effect"), 400), "overturn": short(d.get("overturn_condition"), 500),
        "sources": list(dict.fromkeys(sources)), "verification": verification})

counts = {}
for f in features:
