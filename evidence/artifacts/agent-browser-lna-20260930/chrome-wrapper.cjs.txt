#!/usr/bin/env node
// Keep agent-browser's supported --executable-path launch, with its actual
// arguments, but move CDP onto the page server plus Chromium's native pipe.
// Native pipe framing is from Playwright 1.63.0 src/server/pipeTransport.ts.
const fs = require('fs');
const path = require('path');
const http = require('http');
const args = process.argv.slice(2);
if (args.includes('--version')) {
  console.log('Google Chrome for Testing 153.0.8010.12');
  process.exit(0);
}
function request(route, data) {
  return new Promise((resolve,reject) => {
    const req = http.request({hostname:'127.0.0.1',port:45871,path:route,
      method:data?'POST':'GET',headers:data?{'Content-Type':'application/json'}:{}}, res => {
      let body=''; res.on('data', x=>body+=x); res.on('end',()=>{
        try { resolve(JSON.parse(body)); } catch(e) { reject(e); }
      });
    });
    req.on('error',reject); req.end(data?JSON.stringify(data):undefined);
  });
}
(async()=>{
  const info = await request('/launch', {args});
  if (info.error) throw new Error(info.error);
  const profile = args.find(x=>x.startsWith('--user-data-dir='))?.split('=').slice(1).join('=');
  if (profile) fs.writeFileSync(path.join(profile,'DevToolsActivePort'), `45871\n/devtools/browser/lna\n`);
  console.error('DevTools listening on ws://127.0.0.1:45871/devtools/browser/lna');
  const timer = setInterval(async()=>{
    try { if (!(await request('/alive')).alive) { clearInterval(timer); process.exit(0); } }
    catch { clearInterval(timer); process.exit(0); }
  },500);
})().catch(e=>{ console.error(e); process.exit(1); });
