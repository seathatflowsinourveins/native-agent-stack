#!/usr/bin/env python3
"""Create OmniRoute's private data directory and its server environment file with generated secrets.

Follows upstream .env.example at release/v3.8.51 (a58000c7): JWT_SECRET (openssl rand -base64 48),
API_KEY_SECRET (openssl rand -hex 32), STORAGE_ENCRYPTION_KEY (openssl rand -hex 32; encrypts the SQLite DB),
INITIAL_PASSWORD (bootstrap-only; the operator replaces it with `omniroute reset-password` in their own terminal).
Secret values are generated with the secrets module, written once with O_EXCL and mode 0600, and never printed.
Usage: make_server_env.py DATA_DIR PORT
"""
import base64
import os
import secrets
import sys
from pathlib import Path

data = Path(sys.argv[1]).expanduser().resolve()
port = int(sys.argv[2])
if not data.is_absolute():
    sys.exit("DATA_DIR must be absolute")
data.mkdir(mode=0o700, parents=True, exist_ok=True)
os.chmod(data, 0o700)
env_path = data / "server.env"
values = {
    "JWT_SECRET": base64.b64encode(secrets.token_bytes(48)).decode(),
    "API_KEY_SECRET": secrets.token_hex(32),
    "STORAGE_ENCRYPTION_KEY": secrets.token_hex(32),
    "STORAGE_ENCRYPTION_KEY_VERSION": "v1",
    "INITIAL_PASSWORD": secrets.token_urlsafe(24),
    "DATA_DIR": str(data),
    "PORT": str(port),
    "OMNIROUTE_SERVER_HOST": "127.0.0.1",
    "REQUIRE_API_KEY": "true",
}
fd = os.open(env_path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)  # refuses to overwrite an existing file
with os.fdopen(fd, "w", encoding="utf-8") as handle:
    handle.write("".join(f"{k}={v}\n" for k, v in values.items()))
print(f"created {env_path.name} in the data dir: {len(values)} entries, mode {oct(env_path.stat().st_mode & 0o777)}; "
      f"data dir mode {oct(data.stat().st_mode & 0o777)}; values not printed")
