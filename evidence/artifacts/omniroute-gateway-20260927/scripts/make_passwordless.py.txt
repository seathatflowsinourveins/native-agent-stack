#!/usr/bin/env python3
"""Switch the local OmniRoute to upstream's passwordless, loopback-only mode (user decision 2026-09-27).

Upstream (release/v3.8.51 a58000c7):
- src/lib/db/settings.ts:301-311 forces requireLogin=true when INITIAL_PASSWORD is set, treating it as a headless deploy.
- bin/cli/commands/setup.mjs:44-53: `omniroute setup` with no password writes requireLogin=false when no password is stored.
- REQUIRE_API_KEY=false keeps /v1 open on the loopback bind; Codex's env_key then accepts a placeholder
  (docs/guides/CODEX-CLI-CONFIGURATION.md, "Local unauthenticated OmniRoute").
This removes INITIAL_PASSWORD and sets REQUIRE_API_KEY=false in both env files (our server.env and the bootstrap
copy .env), keeping every other line. Writes atomically with mode 0600 and prints key names only, never values.
"""
import os
import sys
import tempfile
from pathlib import Path

data = Path(sys.argv[1])
for name in ("server.env", ".env"):
    path = data / name
    if not path.exists():
        print(f"{name}: absent")
        continue
    lines = path.read_text(encoding="utf-8").splitlines()
    kept, removed, set_flag = [], [], False
    for line in lines:
        key = line.split("=", 1)[0].strip()
        if key == "INITIAL_PASSWORD":
            removed.append(key)
            continue
        if key == "REQUIRE_API_KEY":
            kept.append("REQUIRE_API_KEY=false")
            set_flag = True
            continue
        kept.append(line)
    if not set_flag:
        kept.append("REQUIRE_API_KEY=false")
    fd, tmp = tempfile.mkstemp(dir=data, prefix=f".{name}.", suffix=".tmp")
    with os.fdopen(fd, "w", encoding="utf-8") as handle:
        handle.write("\n".join(kept) + "\n")
    os.chmod(tmp, 0o600)
    os.replace(tmp, path)
    print(f"{name}: removed {removed or 'nothing'}; REQUIRE_API_KEY=false; {len(kept)} lines kept; mode 600")
