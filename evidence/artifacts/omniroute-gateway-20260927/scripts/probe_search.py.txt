#!/usr/bin/env python3
"""Web-search probe through the OmniRoute gateway: Codex standalone web.run -> /v1/alpha/search.

Uses the same keys as the landscape-sweep GPT-6 lane (#387): provider supports_standalone_web_search, the
standalone_web_search feature, web_search live, shell_snapshot off. Prints item types, the final answer and the
gateway's /alpha/search rows (path, status, time only). Never prints headers, accounts or connection ids.
Usage: probe_search.py <out_dir>
"""
import datetime
import json
import os
import subprocess
import sys
import time
import urllib.request

LAUNCHER = os.path.expanduser("~/.local/share/codex-ecosystem/bin/codex-omniroute")
out_dir = sys.argv[1]
os.makedirs(out_dir, exist_ok=True)
stamp = datetime.datetime.now(datetime.timezone.utc).strftime("%Y%m%dT%H%M%SZ")
started = datetime.datetime.now(datetime.timezone.utc).isoformat()[:19]
prompt = ("Use web search to find the most recent release tag published for the GitHub repository "
          "diegosouzapw/OmniRoute. Reply with one line: the tag, then the URL of the page you used.")
events_path = os.path.join(out_dir, f"search-events-{stamp}.jsonl")
t0 = time.time()
overrides = [] if os.environ.get("PROBE_PROFILE_ONLY") == "1" else [
    "-c", "model_providers.omniroute.supports_standalone_web_search=true",
    "-c", "features.standalone_web_search=true", "-c", 'web_search="live"',
    "-c", "features.shell_snapshot=false"]
print("mode:", "profile-only (no -c overrides)" if not overrides else "with -c overrides")
with open(events_path, "w", encoding="utf-8") as events:
    proc = subprocess.run(
        [LAUNCHER, "exec", "--skip-git-repo-check", "-s", "read-only", "--json", *overrides, prompt],
        stdin=subprocess.DEVNULL, stdout=events, stderr=subprocess.PIPE, text=True, timeout=900)
item_types, final_text, usage, errors = {}, None, None, []
for line in open(events_path, encoding="utf-8"):
    try:
        event = json.loads(line)
    except ValueError:
        continue
    if event.get("type") == "item.completed":
        item = event.get("item", {})
        item_types[item.get("type")] = item_types.get(item.get("type"), 0) + 1
        if item.get("type") == "agent_message":
            final_text = item.get("text")
    elif event.get("type") == "turn.completed":
        usage = event.get("usage")
    elif event.get("type") in ("error", "turn.failed"):
        errors.append(str(event.get("message") or event.get("error"))[:200])
print(json.dumps({"exec_rc": proc.returncode, "elapsed_s": round(time.time() - t0, 1), "item_types": item_types,
                  "final": (final_text or "")[:300], "usage": usage, "errors": errors[:5]}, indent=1))
time.sleep(3)
with urllib.request.urlopen("http://127.0.0.1:20128/api/usage/call-logs?limit=40", timeout=30) as response:
    rows = json.load(response)
rows = rows if isinstance(rows, list) else rows.get("logs") or rows.get("data") or []
search_rows = [{"ts": str(r.get("timestamp"))[11:19], "path": r.get("path"), "status": r.get("status")}
               for r in rows if isinstance(r, dict) and "alpha/search" in str(r.get("path"))
               and str(r.get("timestamp") or "") >= started]
print("gateway /alpha/search rows:", json.dumps(search_rows))
