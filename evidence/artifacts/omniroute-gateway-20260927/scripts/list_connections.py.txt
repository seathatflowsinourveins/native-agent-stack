#!/usr/bin/env python3
"""List OmniRoute provider connections on the loopback gateway: provider ids and counts only.

Never prints tokens, emails, account ids or other connection fields.
"""
import json
import urllib.request

with urllib.request.urlopen("http://127.0.0.1:20128/api/providers", timeout=10) as response:
    data = json.load(response)
items = data if isinstance(data, list) else next(
    (v for v in (data.get("connections"), data.get("providers"), data.get("data"), data.get("items")) if isinstance(v, list)),
    [])
counts = {}
for item in items:
    if isinstance(item, dict):
        provider = str(item.get("provider") or item.get("providerId") or item.get("type") or "?")[:32]
        active = item.get("isActive", item.get("enabled", True))
        counts.setdefault(provider, [0, 0])
        counts[provider][0] += 1
        counts[provider][1] += 1 if active else 0
print("top-level keys:", sorted(data)[:8] if isinstance(data, dict) else "list")
print("connections:", len(items), "| by provider (total, active):", {k: tuple(v) for k, v in sorted(counts.items())})
