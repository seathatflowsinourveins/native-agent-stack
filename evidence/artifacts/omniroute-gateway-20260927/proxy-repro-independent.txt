// Corrected reproduction for OmniRoute defect 1: every arm builds its own Request, so no arm can see a body that an
// earlier arm consumed. The Proxy handler is the one in the original proxy_repro.js. The rebuild call mirrors
// withDeadlineSignal (open-sse/utils/earlyStreamKeepalive.ts:341 at a58000c7): new Request(input, {signal, headers}).
const make = (method) => new Request("http://x/v1/responses", method === "POST"
  ? {method, body: "{}", headers: {"content-type": "application/json"}}
  : {method});
const proxied = (target) => new Proxy(target, {get(t, p) { const v = Reflect.get(t, p, t); return typeof v === "function" ? v.bind(t) : v; }});
const out = [];
for (const method of ["POST", "GET"]) {
  for (const [label, build] of [["raw", () => make(method)], ["proxy", () => proxied(make(method))]]) {
    try {
      const input = build();
      new Request(input, {signal: new AbortController().signal, headers: new Headers(input.headers)});
      out.push(`${method} ${label}=ok`);
    } catch (e) {
      out.push(`${method} ${label}=${e.constructor.name}: ${e.message}`);
    }
  }
}
console.log(process.version, "undici", process.versions.undici, "|", out.join(" | "));
