"""Probe gate for the 20128 restart onto the upstream PR 15167 prefix: a few tiny live requests through the gateway and, for each, a read-only look at the
effort the gateway sent upstream (the call_logs row's two effort columns, filled for rows whose response carried encrypted reasoning).
Values-free: the gateway is keyless on loopback, so no credential is read or sent; the output holds status codes, response event types, the served
model, token counts and the metadata columns named below, never a request or response body.
Usage: probe_gate.py <label> [--lane]   (the label only names the run in the output, for example before-B2 or after-B3)
--lane adds one request through the framework lane (20129's sharedgw node), and it is sent only when the bare-id request to 20128 answered 200: the node
forwards the id without its provider prefix, and an id that 20128 cannot place answers 401, which 20129 reads as the node's own credential failing and
answers by parking the node's only connection as expired for every caller (2026-09-30 05:50Z). So no unknown id goes through the lane."""
import datetime
import json
import sqlite3
import sys
import time
import urllib.error
import urllib.request
import uuid
from pathlib import Path

LABEL = sys.argv[1]
WITH_LANE = "--lane" in sys.argv
STORE = Path.home() / ".local/share/omniroute/storage.sqlite"
PROMPT = "Compute 48271 * 3917 and reply with only the number."   # a small sum, so that the model emits a reasoning item and the effort columns fill
UA_0157 = {"User-Agent": "codex_cli_rs/0.157.1 (Linux; x86_64)", "Version": "0.157.1", "originator": "codex_cli_rs"}


def now():
    return datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def call(port, model, effort, extra_headers=None, timeout=240):
    correlation = f"probe-{LABEL}-{uuid.uuid4().hex[:12]}"
    body = {"model": model, "instructions": "You are a helpful assistant.", "input": PROMPT, "stream": True, "store": False}
    if effort:
        body["reasoning"] = {"effort": effort}
    headers = {"Content-Type": "application/json", "Accept": "text/event-stream", "X-Correlation-Id": correlation, **(extra_headers or {})}
    request = urllib.request.Request(f"http://127.0.0.1:{port}/v1/responses", data=json.dumps(body).encode(), headers=headers, method="POST")
    started = time.time()
    result = {"port": port, "model": model, "body_effort": effort, "extra_headers": sorted((extra_headers or {}).keys()), "correlation_id": correlation}
    try:
        with urllib.request.urlopen(request, timeout=timeout) as response:
            result["status"] = response.status
            types, served, usage, response_id = [], None, None, None
            for raw in response:
                line = raw.decode("utf-8", "replace").strip()
                if not line.startswith("data:"):
                    continue
                payload = line[5:].strip()
                if not payload or payload == "[DONE]":
                    continue
                try:
                    event = json.loads(payload)
                except ValueError:
                    continue
                kind = event.get("type")
                if kind and kind not in types:
                    types.append(kind)
                if kind in ("response.created", "response.completed"):
                    response_id = (event.get("response") or {}).get("id") or response_id
                if kind == "response.completed":
                    inner = event.get("response", {})
                    served = inner.get("model")
                    usage = inner.get("usage")
            result["event_types"] = types
            result["_response_id"] = response_id
            result["served_model"] = served
            if isinstance(usage, dict):
                result["usage"] = {"input": usage.get("input_tokens"), "output": usage.get("output_tokens"),
                                   "reasoning": (usage.get("output_tokens_details") or {}).get("reasoning_tokens")}
    except urllib.error.HTTPError as error:
        result["status"] = error.code
        text = error.read().decode("utf-8", "replace")
        try:
            message = json.loads(text)
            message = message.get("error", message)
            message = message.get("message", message) if isinstance(message, dict) else message
        except ValueError:
            message = text
        result["error_message"] = str(message)[:240]
    except Exception as error:  # noqa: BLE001 - record the type and stop for this request only
        result["status"] = None
        result["error"] = type(error).__name__
    result["seconds"] = round(time.time() - started, 1)
    return result


def log_row(response_id):
    """The call_logs row of one request, found by the Responses API id the gateway logged (the correlation_id column holds a gateway-made UUID, not the
    client's X-Correlation-Id header)."""
    if not STORE.exists() or not response_id:
        return None
    for _ in range(10):
        connection = sqlite3.connect(f"file:{STORE}?mode=ro", uri=True, timeout=10)
        connection.execute("pragma query_only = on")
        row = connection.execute(
            "select provider, model, requested_model, status, reasoning_source, reasoning_effort_requested, reasoning_effort_upstream "
            "from call_logs where response_id = ? order by timestamp desc limit 1", (response_id,)).fetchone()
        connection.close()
        if row:
            names = ("provider", "model", "requested_model", "status", "reasoning_source", "effort_requested", "effort_upstream")
            return dict(zip(names, row))
        time.sleep(2)
    return None


PLAN = [
    # (port, model, body effort, extra headers, what it shows)
    ("20128", "cx/gpt-6.1-sol", "max", None, "native-style max request on the base id"),
    ("20128", "cx/gpt-6.1-sol-max", None, None, "model-id suffix max (parsed only when the base id is in the alias sets)"),
    ("20128", "cx/gpt-6.1-sol-xhigh", None, None, "suffix xhigh (parsed for every model)"),
    ("20128", "gpt-6.1-sol-max", None, None, "the bare id exactly as the framework lane's node forwards it (no provider prefix): must resolve to codex"),
    ("20128", "cx/gpt-6-sol", "low", UA_0157, "a request carrying the 0.157.1 User-Agent and Version header, on a model that client can use (Gate A owner's probe gate)"),
    ("20128", "cx/gpt-6-astra-max", None, None, "the astra lane model at max: must stay max"),
    ("20128", "cx/gpt-6.1-sol-nonexistent-probe", "low", None, "a made-up id must answer 4xx"),
]
LANE_STEP = ("20129", "sharedgw/gpt-6.1-sol-max", None, None, "the framework lane's node forwarding the max suffix to 20128")

out = {"label": LABEL, "started_utc": now(), "prompt": PROMPT, "runs": []}
bare_ok = False
for port, model, effort, headers, note in PLAN:
    result = call(port, model, effort, headers)
    if model == "gpt-6.1-sol-max" and port == "20128":
        bare_ok = result.get("status") == 200
    result["shows"] = note
    rid = result.pop("_response_id", None)
    result["response_id_seen"] = bool(rid)
    result["call_log"] = log_row(rid) if port == "20128" else "20129 rows are not read here (its store is separate and unchanged)"
    out["runs"].append(result)
if WITH_LANE:
    if bare_ok:
        port, model, effort, headers, note = LANE_STEP
        result = call(port, model, effort, headers)
        result["shows"] = note
        result.pop("_response_id", None)
        result["call_log"] = "20129 rows are not read here (its store is separate and unchanged)"
        out["runs"].append(result)
    else:
        out["runs"].append({"port": "20129", "model": LANE_STEP[1], "skipped": "the bare id did not answer 200 on 20128, so the id is not sent through the lane"})
out["finished_utc"] = now()
print(json.dumps(out, indent=1))
