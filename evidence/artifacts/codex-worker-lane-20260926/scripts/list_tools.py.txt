# As run on 2026-09-26 (paths masked); a record, not a maintained tool.
"""Scratch probe: Codex's own view of MCP tool annotations (app-server mcpServerStatus/list).

Builds a sandbox CODEX_HOME whose config.toml holds only the named MCP server tables copied from the host's
user config (no otel, plugins, hooks, trust or auth), starts `codex app-server` there with an empty working
directory, and prints each server's tools with their MCP annotations. Nothing under the host's Codex home is
written. usage: list_tools.py <sandbox-dir> <server> [<server> ...]
"""
import json
import os
import select
import shutil
import subprocess
import sys
import time
import tomllib
from pathlib import Path

sandbox = Path(sys.argv[1]).resolve()
servers = sys.argv[2:]
host_cfg = tomllib.loads((Path.home() / ".codex" / "config.toml").read_text())


def toml_value(v):
    if isinstance(v, bool):
        return "true" if v else "false"
    if isinstance(v, (int, float)):
        return repr(v)
    if isinstance(v, str):
        return json.dumps(v)
    if isinstance(v, list):
        return "[" + ", ".join(toml_value(x) for x in v) + "]"
    raise TypeError(type(v))


lines = []
for name in servers:
    table = host_cfg["mcp_servers"][name]
    lines.append(f"[mcp_servers.{name}]")
    lines.append("startup_timeout_sec = 120") if "startup_timeout_sec" not in table else None
    for k, v in table.items():
        if not isinstance(v, dict):
            lines.append(f"{k} = {toml_value(v)}")
    for k, v in table.items():
        if isinstance(v, dict):
            lines.append(f"[mcp_servers.{name}.{k}]")
            for kk, vv in v.items():
                lines.append(f"{json.dumps(kk)} = {toml_value(vv)}")
    lines.append("")
shutil.rmtree(sandbox, ignore_errors=True)
(home := sandbox / "home").mkdir(parents=True)
(codex_home := home / ".codex").mkdir()
(work := sandbox / "cwd").mkdir()
(codex_home / "config.toml").write_text("\n".join(lines))
child_env = {"HOME": str(home), "CODEX_HOME": str(codex_home), "LANG": "C.UTF-8",
             "PATH": os.defpath + ":" + os.path.dirname(shutil.which("codex"))}
p = subprocess.Popen([shutil.which("codex"), "app-server", "--listen", "stdio://"], cwd=work, env=child_env,
                     stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, bufsize=1)


def send(o):
    p.stdin.write(json.dumps(o) + "\n")
    p.stdin.flush()


def recv(want_id, timeout=120):
    end = time.time() + timeout
    while time.time() < end:
        r, _, _ = select.select([p.stdout], [], [], 1)
        if not r:
            continue
        line = p.stdout.readline()
        if not line:
            break
        msg = json.loads(line)
        if msg.get("id") == want_id and ("result" in msg or "error" in msg):
            return msg
    return None


send({"id": 1, "method": "initialize",
      "params": {"clientInfo": {"name": "nas_tools_probe", "version": "0.0.1"}, "capabilities": {"experimentalApi": True}}})
recv(1)
send({"method": "initialized"})
time.sleep(1)
send({"id": 2, "method": "mcpServerStatus/list", "params": {"detail": "toolsAndAuthOnly"}})
r = recv(2, timeout=180)
out = {}
for s in (r or {}).get("result", {}).get("data", []):
    tools = {}
    for tname, t in sorted(s.get("tools", {}).items()):
        ann = t.get("annotations") or {}
        tools[tname] = {k: ann.get(k) for k in ("readOnlyHint", "destructiveHint", "openWorldHint", "idempotentHint")}
        tools[tname]["description"] = (t.get("description") or "")[:260]
    out[s["name"]] = {"tools_error": s.get("toolsError"), "tools": tools}
print(json.dumps(out, indent=1, sort_keys=True))
if r is None or "error" in (r or {}):
    print("RESPONSE", json.dumps(r)[:600])
p.stdin.close()
try:
    p.wait(timeout=15)
except subprocess.TimeoutExpired:
    p.kill()
print("rc", p.returncode, "stderr tail:", p.stderr.read()[-400:])
