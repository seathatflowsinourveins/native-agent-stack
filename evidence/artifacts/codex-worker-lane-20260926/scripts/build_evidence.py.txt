UUID = re.compile(r"\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b")


def clean(text: str) -> str:
    text = text.replace(str(SP), "$SCRATCH").replace("$TMP/codex-lane-", "$TMP/codex-lane-")
    text = re.sub(r"/tmp/claude-\d+/[^\s\"')]*", "$SCRATCH/...", text)  # a path cut short in a printed excerpt
    text = UUID.sub("<uuid>", text)
    return host_receipts.sanitize(text)


def put(name: str, text: str) -> None:
    path = OUT / name
    path.parent.mkdir(parents=True, exist_ok=True)
    lines = [line.rstrip() for line in clean(text).rstrip("\n").splitlines()]  # no trailing blanks (git diff --check)
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")
