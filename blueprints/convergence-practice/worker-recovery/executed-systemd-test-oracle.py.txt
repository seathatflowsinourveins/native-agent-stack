"""A repeated or unauthorized effect must fail before a native worker runs."""
import importlib.util
import hashlib
import json
from pathlib import Path
import tempfile
import unittest

HERE = Path(__file__).resolve().parents[1] / 'blueprints/convergence-practice/worker-recovery'


class WorkerRecoveryFixtureTests(unittest.TestCase):
    def setUp(self):
        spec = importlib.util.spec_from_file_location('worker_recovery_stage', HERE / 'stage.py')
        self.module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(self.module)
        temp = tempfile.TemporaryDirectory()
        self.addCleanup(temp.cleanup)
        self.root = Path(temp.name)
        (self.root / 'input.json').write_text('{"records":[{"id":"b","value":7},{"id":"a","value":5}]}')

    def test_checkpoint_is_exact_and_repetition_is_rejected(self):
        self.module.execute(self.root, 'checkpoint')
        self.assertEqual((self.root / 'checkpoint.json').read_bytes(), b'{"ids":["a","b"],"total":12}\n')
        with self.assertRaisesRegex(ValueError, 'order'):
            self.module.execute(self.root, 'checkpoint')

    def test_finalize_refuses_missing_interruption_or_supervisor_authorization(self):
        self.module.execute(self.root, 'checkpoint')
        with self.assertRaisesRegex(ValueError, 'order'):
            self.module.execute(self.root, 'finalize')
        (self.root / 'actions.jsonl').write_text('\n'.join(json.dumps({'action':x}) for x in ['checkpoint','cancel-wait','crash-wait'])+'\n')
        with self.assertRaisesRegex(ValueError, 'authorization'):
            self.module.execute(self.root, 'finalize')

    def test_finalize_writes_one_effect_and_keeps_checkpoint(self):
        self.module.execute(self.root, 'checkpoint')
        before = (self.root / 'checkpoint.json').read_bytes()
        (self.root / 'actions.jsonl').write_text('\n'.join(json.dumps({'action':x}) for x in ['checkpoint','cancel-wait','crash-wait'])+'\n')
        (self.root / 'resume-authorized.json').write_text('{}')
        self.module.execute(self.root, 'finalize')
        self.assertEqual((self.root / 'checkpoint.json').read_bytes(), before)
        self.assertEqual(json.loads((self.root / 'final.json').read_text())['execution_count'], 1)
        with self.assertRaisesRegex(ValueError, 'order'):
            self.module.execute(self.root, 'finalize')

    def test_changed_checkpoint_or_input_is_not_accepted(self):
        (self.root / 'input.json').write_text('{"records":[]}')
        with self.assertRaisesRegex(ValueError, 'oracle'):
            self.module.execute(self.root, 'checkpoint')
        self.assertFalse((self.root / 'checkpoint.json').exists())

    def test_changed_checkpoint_cannot_be_finalized(self):
        self.module.execute(self.root, 'checkpoint')
        (self.root / 'actions.jsonl').write_text('\n'.join(json.dumps({'action':x}) for x in ['checkpoint','cancel-wait','crash-wait'])+'\n')
        (self.root / 'resume-authorized.json').write_text('{}')
        (self.root / 'checkpoint.json').write_text('changed')
        with self.assertRaisesRegex(ValueError, 'checkpoint_changed'):
            self.module.execute(self.root, 'finalize')
        self.assertFalse((self.root / 'final.json').exists())


class NativeStreamSummaryTests(unittest.TestCase):
    def setUp(self):
        spec = importlib.util.spec_from_file_location('worker_recovery_runner', HERE / 'run.py')
        self.module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(self.module)

    def test_repeated_native_init_and_terminal_events_keep_same_identity(self):
        events = [{'type':'system','subtype':'init','session_id':'same','model':'claude-opus-5'},
                  {'type':'result','subtype':'success','is_error':False},
                  {'type':'system','subtype':'init','session_id':'same','model':'claude-opus-5'},
                  {'type':'result','subtype':'success','is_error':False}]
        self.assertEqual(self.module.native_summary(events, 'same'), (True, True, True))
        events[-2]['session_id'] = 'different'
        self.assertEqual(self.module.native_summary(events, 'same'), (False, True, True))
        events[-1]['is_error'] = True
        self.assertEqual(self.module.native_summary(events, 'same'), (False, True, False))

    def test_missing_native_init_and_terminals_are_unknown(self):
        self.assertEqual(self.module.native_summary([], 'same'), (False, False, False))


class PublishedEffectTests(unittest.TestCase):
    def test_retained_outputs_satisfy_single_effect_oracle(self):
        accepted = HERE / 'accepted'
        expected = b'{"ids":["a","b"],"total":12}\n'
        self.assertEqual((accepted / 'checkpoint.json').read_bytes(), expected)
        self.assertEqual([json.loads(line)['action'] for line in (accepted / 'actions.jsonl').read_text().splitlines()],
                         ['checkpoint', 'cancel-wait', 'crash-wait', 'finalize'])
        self.assertEqual(json.loads((accepted / 'final.json').read_text()),
                         {'checkpoint_sha256':hashlib.sha256(expected).hexdigest(), 'execution_count':1, 'status':'complete'})


class OwnedServiceCommandTests(unittest.TestCase):
    def setUp(self):
        spec = importlib.util.spec_from_file_location('owned_service', HERE / 'service.py')
        self.module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(self.module)

    def test_service_keeps_native_arguments_and_environment_values_out_of_argv(self):
        command = self.module.service_command('foundation-recovery-0123456789ab-initial',
                                              ['claude', '-p', 'literal $text'], ['PATH', 'EXAMPLE_PRIVATE'])
        self.assertEqual(command[-3:], ['claude', '-p', 'literal $text'])
        self.assertIn('--expand-environment=no', command)
        self.assertIn('--setenv=EXAMPLE_PRIVATE', command)
        self.assertIn('--property=KillMode=control-group', command)
        self.assertIn('--property=Restart=no', command)

    def test_shared_unit_names_are_refused_before_any_process_operation(self):
        for name in ['dbus', 'default.target', 'foundation-recovery-../other', '']:
            with self.subTest(name=name), self.assertRaises(ValueError):
                self.module.validate_unit(name)


if __name__ == '__main__':
    unittest.main()
