#!/usr/bin/env bash
# Documentary supported pnpm 12.4.2 commands; use owned durable scratch.
# Sources: https://pnpm.io/cli/update and https://pnpm.io/cli/install;
# native installed update help rejects versioned indirect selectors.
# Advisory/release select exact resolved sharp 0.35.5, current published latest.
set -euo pipefail
P=$1; R=$2; RTK=$3; COREPACK=$4
export COREPACK_HOME=$R/corepack TMPDIR=$R/tmp NEXT_TELEMETRY_DISABLED=1
cd "$P"
"$RTK" "$COREPACK" pnpm update sharp --no-save --lockfile-only --ignore-scripts --store-dir "$R/pnpm-store" --state-dir "$R/pnpm-state"
"$RTK" "$COREPACK" pnpm install --frozen-lockfile --ignore-scripts --store-dir "$R/pnpm-store" --state-dir "$R/pnpm-state"
"$RTK" node -e 'const r=require("module").createRequire(require.resolve("next/package.json"));const s=r("sharp");if(s.versions.sharp!=="0.35.5"||s.versions.rsvg!=="2.63.2")process.exit(1)'
"$RTK" "$COREPACK" pnpm peers check
"$RTK" "$COREPACK" pnpm typecheck
"$RTK" "$COREPACK" pnpm build
# Full make verify also requires its dedicated PostgreSQL DB and free ports.
