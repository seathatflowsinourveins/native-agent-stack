"""Native controls for the split scan of .github/workflows/security-scan.yml (run from the repository root):
usage: osv_split_controls.py <osv-scanner binary> <out dir>

1. The two lockfile lists the workflow builds with jq are built here with the workflow's own jq lines (extracted from the file, so they cannot drift): their sizes,
   that together they are the inventory, and the sha256 of each list.
2. The workflow's scan step itself (its `run:` block, extracted from the file and run by bash with the pinned binary as $RUNNER_TEMP/osv-scanner/osv-scanner),
   once with WRITE_SARIF=false (the pull_request form) and once with WRITE_SARIF=true (the other events), with its exit status, the ids each config filtered out
   and the two SARIF reports it writes (size and sha256; host paths masked).
3. Native negative controls on scratch copies and on the frozen lock with the pinned binary: the frozen lock under the ordinary config (must still report next),
   the frozen config and the ordinary config over inputs that hide a vulnerable next in YAML the scanner reads (two inputs found by the Codex root's verifier and one
   more), and the live application-delivery lock under the ordinary config.
Environment for the subprocesses is built here (no inherited variables); nothing is sent over the network by this script."""
import hashlib
import json
import re
import subprocess
import sys
import tempfile
from pathlib import Path

osv, out_dir = Path(sys.argv[1]).resolve(), Path(sys.argv[2]).resolve()
root = Path.cwd()
workflow = (root / ".github/workflows/security-scan.yml").read_text(encoding="utf-8")
FROZEN = "evidence/artifacts/macos-application-20260924/variant/pnpm-lock.yaml"
LIVE = "blueprints/convergence-practice/application-delivery/pnpm-lock.yaml"
ORDINARY_CONFIG, FROZEN_CONFIG = ".github/osv-scanner.toml", ".github/osv-scanner-frozen-macos.toml"


def env_for(runner_temp=None, write_sarif=None):
    env = {"PATH": "/usr/local/bin:/usr/bin:/bin", "HOME": str(out_dir), "LANG": "C.UTF-8"}
    if runner_temp is not None:
        env.update(RUNNER_TEMP=str(runner_temp), WRITE_SARIF=write_sarif)
    return env


def step_script():
    start = workflow.index("      - name: Scan every listed lockfile and manifest")
    run = workflow.index("        run: |\n", start) + len("        run: |\n")
    end = workflow.index("      - name: Keep the OSV-Scanner SARIF for the upload job", run)
    return "\n".join(line[10:] if line.startswith(" " * 10) else line for line in workflow[run:end].splitlines()) + "\n"


def sha(data):
    return hashlib.sha256(data if isinstance(data, bytes) else data.encode()).hexdigest()


def ids_in(stdout):
    try:
        doc = json.loads(stdout)
    except ValueError:
        return f"unparsable output {stdout[:60]!r}"
    found = {f"{p['package']['name']}=={p['package']['version']}": sorted({v["id"] for v in p.get("vulnerabilities", [])})
             for r in doc.get("results", []) for p in r.get("packages", []) if p.get("vulnerabilities")}
    return found or "no findings"


def scan(config, lockfile_args, extra=()):
    return subprocess.run([str(osv), "scan", "source", "--config", str(config), "--no-resolve", *lockfile_args, "--format", "json", *extra], capture_output=True,
                          text=True, cwd=root, env=env_for())


# ---- 1. the two lists, with the workflow's own jq lines
lists_script = "inventory=.github/osv-scanner-lockfiles.json\nfrozen_config=" + FROZEN_CONFIG + "\n" + "\n".join(
    line.strip() for line in step_script().splitlines() if line.strip().startswith("mapfile -t")) + "\n" + (
    'echo "ordinary: ${#lockfiles[@]} entries, sha256 of the list: $(printf \'%s\\n\' "${lockfiles[@]}" | sha256sum | cut -c1-64)"\n'
    'echo "frozen: ${#frozen[@]} entries, sha256 of the list: $(printf \'%s\\n\' "${frozen[@]}" | sha256sum | cut -c1-64)"\n'
    'echo "inventory: $(jq \'.lockfiles | length\' "$inventory") entries"\n'
    'printf \'frozen entry: %s\\n\' "${frozen[@]}"\n'
    'echo "sum check: $(( ${#lockfiles[@]} + ${#frozen[@]} )) == $(jq \'.lockfiles | length\' "$inventory")"\n')
print("== 1. the lists the workflow builds (its own jq lines)")
lists = subprocess.run(["bash", "-c", lists_script], capture_output=True, text=True, cwd=root, env=env_for())
print(lists.stdout.rstrip(), "\nexit:", lists.returncode, lists.stderr.strip()[:200])

# ---- 2. the workflow's scan step itself
script = step_script()
print("\n== 2. the workflow's scan step itself, run by bash with the pinned binary (the step's run: block extracted from .github/workflows/security-scan.yml)")
print("step script sha256:", sha(script))
for write_sarif in ("false", "true"):
    with tempfile.TemporaryDirectory() as runner_temp:
        bin_dir = Path(runner_temp) / "osv-scanner"
        bin_dir.mkdir()
        (bin_dir / "osv-scanner").symlink_to(osv)
        run = subprocess.run(["bash", "-c", script], capture_output=True, text=True, cwd=root, env=env_for(runner_temp, write_sarif))
        filtered = [line[:150] for line in (run.stdout + run.stderr).splitlines() if "has been filtered out" in line or line.startswith("Filtered ")]
        ids = [line.split(" ")[0] for line in filtered if "has been filtered out" in line]
        print(f"WRITE_SARIF={write_sarif}: step exit {run.returncode}; ids filtered out per scan: {ids}; "
              f"'Filtered N vulnerabilities' lines: {[l for l in filtered if l.startswith('Filtered ')]}")
        print(f"    scanned-file lines of both scans: {sum(1 for l in (run.stdout + run.stderr).splitlines() if l.startswith('Scanned '))}")
        for report in ("osv-scanner.sarif", "osv-scanner-frozen-macos.sarif"):
            path = bin_dir / report
            if path.exists():
                data = path.read_bytes().replace(str(root).encode(), b"<scan-root>")
                (out_dir / f"relock-2026-09-30-urllib3.split-{report}").write_bytes(data)
                sarif = json.loads(data)
                print(f"    {report}: {len(data)} bytes, sha256 {sha(data)[:16]}..., runs {len(sarif.get('runs', []))}, "
                      f"results {sum(len(r.get('results', [])) for r in sarif.get('runs', []))}")
            else:
                print(f"    {report}: not written")

# ---- 3. native negative controls
print("\n== 3. native controls with the pinned binary")
empty = Path(tempfile.mkstemp(suffix=".toml")[1])
run = scan(ORDINARY_CONFIG, [f"--lockfile=pnpm-lock.yaml:{FROZEN}"])
print(f"frozen lock under the ORDINARY config: exit {run.returncode}; {ids_in(run.stdout)}")
run = scan(FROZEN_CONFIG, [f"--lockfile=pnpm-lock.yaml:{FROZEN}"])
print(f"frozen lock under the FROZEN config: exit {run.returncode}; {ids_in(run.stdout)}; filtered out: {[l.split(' ')[0] for l in run.stderr.splitlines() if 'has been filtered out' in l]}")
run = scan(ORDINARY_CONFIG, [f"--lockfile=pnpm-lock.yaml:{LIVE}"])
print(f"live application-delivery lock (next 16.3.6) under the ORDINARY config: exit {run.returncode}; {ids_in(run.stdout)}")
CASES = {
    "tagged packages key (`!!str packages:`)": "lockfileVersion: '9.0'\n!!str packages:\n  next@16.3.5:\n    resolution: {}\n",
    "packages through a YAML merge key (`<<: *base`)": "lockfileVersion: '9.0'\nbase: &base\n  packages:\n    next@16.3.5:\n      resolution: {}\n<<: *base\n",
    "Unicode line separator after `packages:`": "lockfileVersion: '9.0'\npackages:   next@16.3.5:\n    resolution: {}\n",
}
with tempfile.TemporaryDirectory() as scratch:
    for name, text in CASES.items():
        lock = Path(scratch) / "pnpm-lock.yaml"
        lock.write_text(text, encoding="utf-8")
        ordinary = scan(ORDINARY_CONFIG, [f"--lockfile=pnpm-lock.yaml:{lock}"])
        frozen = scan(FROZEN_CONFIG, [f"--lockfile=pnpm-lock.yaml:{lock}"])
        print(f"{name}: as an ordinary lock (ordinary config) exit {ordinary.returncode}, {ids_in(ordinary.stdout)}; "
              f"under the frozen config exit {frozen.returncode}, {ids_in(frozen.stdout)}")
empty.unlink()
