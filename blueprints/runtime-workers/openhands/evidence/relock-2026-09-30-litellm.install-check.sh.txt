#!/usr/bin/env bash
# Published copy of the script as it ran on 2026-09-30 (it takes its two host paths as arguments; the returned outputs mask them as <scratch-relock> and <worktree>). It needs the
# uv 0.12.17 binary, a CPython 3.13 and network access (PyPI), and is documentation of what ran, not a supported tool.
# The recipe's install-container.sh sequence (blueprints/runtime-workers/openhands/install-container.sh) run on the host in a scratch venv (UV_NO_CONFIG=1 as there), on the new lock,
# with no model request. usage: install_check_litellm.sh <scratch-relock> <worktree>
set -uo pipefail
R=$1; W=$2; OH=$W/blueprints/runtime-workers/openhands; V=$R/u3/venv
export UV_CACHE_DIR=$R/uv-cache UV_PYTHON_DOWNLOADS=never UV_NO_CONFIG=1
rm -rf "$V"
echo "[$(date -u +%FT%TZ)] uv venv"; uv venv --python 3.13 "$V" 2>&1 | tail -1; echo "venv exit: ${PIPESTATUS[0]}"
uv pip install --python "$V/bin/python" --require-hashes --no-deps --only-binary :all: --index-url https://pypi.org/simple -r "$OH/build-requirements.lock" 2>&1 | tail -2; echo "build-requirements install exit: ${PIPESTATUS[0]}"
uv pip install --python "$V/bin/python" --require-hashes --no-deps --no-build-isolation --index-url https://pypi.org/simple -r "$OH/requirements.lock" 2>&1 | tail -2; echo "runtime lock install exit: ${PIPESTATUS[0]}"
uv pip check --python "$V/bin/python" 2>&1 | tail -2; echo "uv pip check exit: ${PIPESTATUS[0]}"
"$V/bin/python" -c 'import importlib.metadata as m; assert m.version("openhands-sdk") == m.version("openhands-tools") == "1.49.6"; assert m.version("litellm") == "1.93.2"; import litellm; from openhands.sdk import LLM, Agent, Conversation; from openhands.tools.terminal import TerminalTool; print("openhands-sdk/tools", m.version("openhands-sdk"), "| litellm", m.version("litellm"), "| imports ok")' 2>&1 | tail -2; echo "sdk import exit: ${PIPESTATUS[0]}"
"$V/bin/python" -c 'import importlib.metadata as m; print("litellm", m.version("litellm"), "| urllib3", m.version("urllib3"), "| pyjwt", m.version("pyjwt"), "| oauthlib", m.version("oauthlib"), "| anyio", m.version("anyio"), "| click", m.version("click"), "| pypdf", m.version("pypdf"), "| soupsieve", m.version("soupsieve"))'
echo "[$(date -u +%FT%TZ)] done"
