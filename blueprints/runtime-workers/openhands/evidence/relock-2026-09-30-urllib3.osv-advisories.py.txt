"""Facts of the five advisories of 2026-09-30 from the OSV API (https://api.osv.dev/v1/vulns/<id>): the OSV record's published time, the GitHub Advisory
Database review time and NVD publication time it carries, aliases, database severity, CVSS vector, affected package and the introduced and fixed versions, one
line per advisory and then its summary; then the OSV query result for the locked and the fixed versions (https://api.osv.dev/v1/query); then the upload times
and yanked flags of the urllib3 2.8.0 and PyJWT 2.14.0 / 2.15.0 / 2.15.1 files on PyPI (https://pypi.org/pypi/<name>/<version>/json). Read-only network access,
no credential."""
import json
import urllib.request

for vuln in ("GHSA-8988-9cw3-xx77", "GHSA-vxq7-64xx-v4gw", "GHSA-gh4c-6fx4-qh6g", "GHSA-42vr-xj54-vc7v", "GHSA-vcvr-r3jv-pc5j"):
    d = json.load(urllib.request.urlopen(f"https://api.osv.dev/v1/vulns/{vuln}", timeout=30))
    affected = d["affected"][0]
    events = [e for r in affected.get("ranges", []) for e in r.get("events", [])]
    specific = d.get("database_specific") or {}
    print(f"{vuln}\tosv published {d.get('published')}\tgithub_reviewed_at {specific.get('github_reviewed_at')}\tnvd_published_at {specific.get('nvd_published_at')}\t"
          f"aliases {d.get('aliases') or '-'}\t{specific.get('severity')}\t{[s.get('score') for s in d.get('severity', [])]}\t"
          f"{affected['package']['ecosystem']} {affected['package']['name']}\tintroduced {[e['introduced'] for e in events if 'introduced' in e]} "
          f"fixed {[e['fixed'] for e in events if 'fixed' in e]}")
    print(f"    {d.get('summary')}")
    if vuln == "GHSA-vcvr-r3jv-pc5j":
        details = " ".join((d.get("details") or "").split())
        start = details.find("Affected applications")
        print(f"    details: {details[start:start + 620]}")

for name, ecosystem, version in (("urllib3", "PyPI", "2.7.0"), ("urllib3", "PyPI", "2.8.0"), ("PyJWT", "PyPI", "2.14.0"), ("PyJWT", "PyPI", "2.15.0"),
                                 ("PyJWT", "PyPI", "2.15.1"), ("next", "npm", "16.3.5"), ("next", "npm", "16.3.6")):
    request = urllib.request.Request("https://api.osv.dev/v1/query", headers={"Content-Type": "application/json"},
                                     data=json.dumps({"package": {"name": name, "ecosystem": ecosystem}, "version": version}).encode())
    found = sorted(v["id"] for v in json.load(urllib.request.urlopen(request, timeout=30)).get("vulns", []))
    print(f"query {ecosystem} {name}=={version}: {found if found else 'no advisories'}")

for name, version in (("urllib3", "2.8.0"), ("PyJWT", "2.14.0"), ("PyJWT", "2.15.0"), ("PyJWT", "2.15.1")):
    release = json.load(urllib.request.urlopen(f"https://pypi.org/pypi/{name}/{version}/json", timeout=30))
    for f in release["urls"]:
        print(f"pypi {name} {version} file {f['filename']}: uploaded {f['upload_time_iso_8601']}, {'yanked' if f.get('yanked') else 'not yanked'}, sha256 {f['digests']['sha256']}")
