#!/bin/bash
# Published copy of the script as it ran on 2026-09-30. Host paths are placeholders (<scratch>, <worktree>, <scratch-relock>); it needs the SDK archive
# sdk-fcc102a.tar.gz of pins.json, uv 0.12.17 and network access, and is documentation of what ran, not a supported tool.
# relock_urllib3.sh: the recipe's runtime-lock relock (research.md "Runtime lock security relock") with urllib3 added and PyJWT moved from 2.14.0 to 2.15.0.
# Control first: the five upgrades of the 2026-09-30 relock (pyjwt==2.14.0) must reproduce the committed lock byte for byte; then the new lock: the same upgrades with
# pyjwt==2.15.0 plus urllib3==2.8.0. PyJWT 2.15.0 (uploaded 2026-09-23T16:56:00Z) is inside the workspace's exclude-newer = "7 days" window until 2026-09-30T16:56:01Z.
set -u
R=<scratch-relock>; W=<worktree>; OH=blueprints/runtime-workers/openhands; U=$R/u3
export UV_CACHE_DIR=$R/uv-cache
stamp() { date -u +%FT%TZ; }
git -C $W show origin/main:$OH/requirements.lock > $U/base.lock
echo "[$(stamp)] uv $(uv --version)"
echo "[$(stamp)] base lock (origin/main $(git -C $W rev-parse --short origin/main)) sha256: $(sha256sum $U/base.lock | cut -d' ' -f1)"
echo "[$(stamp)] sdk archive sha256: $(sha256sum $R/sdk-fcc102a.tar.gz | cut -d' ' -f1) (pins.json source_archive.sha256: $(jq -r .source_archive.sha256 $W/$OH/pins.json))"
for variant in control new; do
  D=$U/$variant; rm -rf $D; mkdir -p $D
  (cd $D && tar -xzf $R/sdk-fcc102a.tar.gz --strip-components=1 && sed -i "/^\[tool\.uv\]\$/a environments = [\"sys_platform == 'linux' and platform_machine == 'x86_64'\"]" pyproject.toml)
  jwt=2.14.0; extra=""; [ $variant = new ] && { jwt=2.15.0; extra="--upgrade-package urllib3==2.8.0"; }
  echo "[$(stamp)] $variant: exclude-newer in the workspace: $(grep -h -E '^exclude-newer = ' $D/pyproject.toml | cut -c1-60)"
  echo "[$(stamp)] $variant: uv lock --upgrade-package anyio==4.14.2 --upgrade-package click==8.5.0 --upgrade-package pypdf==6.19.0 --upgrade-package soupsieve==2.9.2 --upgrade-package pyjwt==$jwt $extra"
  (cd $D && uv lock --upgrade-package anyio==4.14.2 --upgrade-package click==8.5.0 --upgrade-package pypdf==6.19.0 --upgrade-package soupsieve==2.9.2 --upgrade-package pyjwt==$jwt $extra > lock.out 2>&1; echo "lock exit: $?" >> lock.out
    uv lock --check > check.out 2>&1; echo "check exit: $?" >> check.out
    uv export --frozen --format requirements.txt --no-header --no-annotate --package openhands-tools --no-emit-workspace > export.txt 2> export.err; echo "export exit: $?" > export.rc
    jq -r '.wheels | to_entries[] | "\(.key) @ \(.value.url) \\\n    --hash=sha256:\(.value.sha256)"' "$W/$OH/pins.json" > wheels.txt
    cat export.txt wheels.txt > requirements.lock.repro)
  echo "[$(stamp)] $variant: $(tail -1 $D/lock.out) | $(tail -1 $D/check.out) | $(cat $D/export.rc) | lines $(wc -l < $D/requirements.lock.repro) | sha256 $(sha256sum $D/requirements.lock.repro | cut -d' ' -f1)"
  grep -E "^Updated|^Resolved" $D/lock.out | head -16 | sed 's/^/    /'
done
echo "[$(stamp)] committed (base) lock sha256: $(sha256sum $U/base.lock | cut -d' ' -f1)"
cmp $U/control/requirements.lock.repro $U/base.lock && echo "control reproduces the committed lock byte for byte"
echo "--- diff base lock vs new lock (changed lines only, hashes abbreviated)"
diff <(sed -E 's/--hash=sha256:([0-9a-f]{8})[0-9a-f]+/--hash=sha256:\1.../' $U/base.lock) <(sed -E 's/--hash=sha256:([0-9a-f]{8})[0-9a-f]+/--hash=sha256:\1.../' $U/new/requirements.lock.repro) | grep '^[<>]' | head -30
echo "[$(stamp)] done"
