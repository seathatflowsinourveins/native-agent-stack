=== script: <red-logs>/ci_pull_request_audit.py ===
"""Which workflows at a commit run on pull_request, and with what token scope and secrets.

Read-only: `git show <commit>:<path>` for every file under .github/workflows, parsed with PyYAML
(the YAML 1.1 key `on` loads as True). For each pull_request-triggered workflow, each job's
effective permissions are its own `permissions`, else the workflow's, else "repository default".
Run from a checkout: python3 <this file> <commit>.
"""
import subprocess
import sys

import yaml

commit = sys.argv[1]
names = subprocess.run(["git", "ls-tree", "--name-only", commit, ".github/workflows/"], capture_output=True,
                       text=True, check=True).stdout.split()
print(f"workflows at {commit[:8]}: {len(names)}")
for name in sorted(names):
    text = subprocess.run(["git", "show", f"{commit}:{name}"], capture_output=True, text=True, check=True).stdout
    data = yaml.safe_load(text)
    on = data.get(True, data.get("on"))
    triggers = [on] if isinstance(on, str) else list(on or [])
    if not any(trigger in ("pull_request", "pull_request_target") for trigger in triggers):
        continue
    top = data.get("permissions", "repository default")
    print(f"\n{name.split('/')[-1]}: triggers={sorted(map(str, triggers))} workflow permissions={top}")
    for job, spec in (data.get("jobs") or {}).items():
        spec = spec or {}
        effective = spec.get("permissions", top)
        writes = sorted(key for key, value in effective.items() if value == "write") if isinstance(effective, dict) \
            else effective
        condition = spec.get("if")
        print(f"  job {job}: permissions={effective} write_scopes={writes or []}"
              + (f" if={condition!r}" if condition else ""))
    secrets = sorted({line.strip() for line in text.splitlines() if "secrets." in line and "GITHUB_TOKEN" not in line})
    print(f"  secrets other than GITHUB_TOKEN: {secrets or 'none'}")
