"""Static OAuthlib review of an installed, hash-verified wheel closure.

Adapted from evidence/relock-2026-09-30.jwt-callers.py.txt: scan every installed
Python source, retain alias/from-import bindings, search outside the affected
package and then inspect the recipe. Python's ast parser extends that maintained
method to comma-separated and multiline literal imports. No scanned module is
imported and no credential, OAuth endpoint, model, API or store is invoked.
This is local integration evidence, not an upstream test or execution trace.
Dynamic imports, runtime-assembled names and non-Python callers remain unknown.
Usage: python <this-file> <site-packages> <recipe-directory>
"""

import ast
import collections
import hashlib
import re
import sys
from pathlib import Path

SITE, RECIPE = map(Path, sys.argv[1:3])
SKIP = {"oauthlib", "pip", "setuptools", "pkg_resources", "_distutils_hack"}
# func-timeout 4.3.5 ships this five-line Python 2 raise helper. Its complete
# reviewed source has no imports or OAuthlib references; dafunc.py:21-26 selects
# py3_raise on this Python 3.13 target. Keep the source in the text sweep and
# accept this one parser limitation only when the reviewed bytes still match.
TEXT_ONLY = {"func_timeout/py2_raise.py":
             "19d989963807de3847a00bbe90337fd9100a79cee4327683b601e4bc96225b77"}
PATTERN = re.compile(
    r"\benable_jsonp\b|\bcreate_revocation_response\b|\bRevocationEndpoint\b|"
    r"\bAuthorizationCodeGrant\b|\bcode_challenge_method_(?:s256|plain)\b|"
    r"\bvalidate_code_challenge\b|oauthlib\.(?:oauth2|openid).*"
    r"(?:endpoints|grant_types|pre_configured)"
)


def qualified(node):
    if isinstance(node, ast.Name):
        return node.id
    if isinstance(node, ast.Attribute):
        base = qualified(node.value)
        return base + "." + node.attr if base else ""
    return ""


def inspect(label, files):
    importers, references = collections.defaultdict(list), collections.defaultdict(list)
    parse_errors, text_only = [], []
    count = 0
    for path, relative in files:
        count += 1
        text = path.read_text(encoding="utf-8", errors="replace")
        lines = text.splitlines()
        try:
            tree = ast.parse(text, filename=relative)
        except SyntaxError as error:
            digest = hashlib.sha256(path.read_bytes()).hexdigest()
            if TEXT_ONLY.get(relative) == digest:
                text_only.append((relative, error.lineno, error.msg, digest, text))
            else:
                parse_errors.append((relative, error.lineno, error.msg))
            tree = None
        bound, imports = {}, []
        if tree is not None:
            for node in ast.walk(tree):
                if isinstance(node, ast.Import):
                    for item in node.names:
                        if item.name == "oauthlib" or item.name.startswith("oauthlib."):
                            name = item.asname or "oauthlib"
                            bound[name] = item.name if item.asname else "oauthlib"
                            imports.append((node.lineno, item.name, name))
                elif isinstance(node, ast.ImportFrom) and node.module and (
                    node.module == "oauthlib" or node.module.startswith("oauthlib.")
                ):
                    for item in node.names:
                        name = item.asname or item.name
                        bound[name] = node.module + "." + item.name
                        imports.append((node.lineno, node.module + "." + item.name, name))
            calls = []
            for node in ast.walk(tree):
                if isinstance(node, ast.Call):
                    name = qualified(node.func)
                    root, _, suffix = name.partition(".")
                    if root in bound:
                        calls.append((node.lineno, bound[root] + ("." + suffix if suffix else "")))
            if imports:
                importers[relative.split("/")[0]].append((relative, imports, sorted(calls)))
        for number, line in enumerate(lines, 1):
            if PATTERN.search(line):
                references[relative.split("/")[0]].append((relative, number, line.strip()))
    print(f"# {label}: {count} Python sources searched; parse errors: {len(parse_errors)}")
    for entry in parse_errors:
        print("PARSE ERROR", entry)
    for relative, line, message, digest, source in text_only:
        print(f"REVIEWED TEXT ONLY {relative}:{line}: {message}; sha256 {digest}")
        print(source)
    print("## OAuthlib literal importers with aliases, from-imports and bound call sites")
    for package, entries in sorted(importers.items()):
        print(package, "files:", len(entries))
        for relative, imports, calls in entries:
            print(relative)
            for line, target, alias in imports:
                print(f"  {line}: {target} bound as {alias}")
            for line, target in calls:
                print(f"  call {line}: {target}")
    print("## Advisory-related source references outside OAuthlib")
    for package, entries in sorted(references.items()):
        print(package, "reference lines:", len(entries))
        for relative, line, source in entries:
            print(f"  {relative}:{line}: {source}")
    if parse_errors:
        raise SystemExit("Static scan incomplete: inspect parse failures before any absence inference")


files = [(path, path.relative_to(SITE).as_posix()) for path in sorted(SITE.rglob("*.py"))
         if path.relative_to(SITE).parts[0] not in SKIP]
inspect("installed closure excluding OAuthlib itself and installer tooling", files)
inspect("recipe", [(path, path.relative_to(RECIPE).as_posix()) for path in sorted(RECIPE.rglob("*.py"))])

print("# Entered namespaces and advisory implementation: file hashes and source lines")
WINDOWS = {
    "oauthlib/__init__.py": [(1, 100)],
    "oauthlib/oauth1/__init__.py": [(1, 100)],
    "oauthlib/oauth2/__init__.py": [(1, 140)],
    "oauthlib/common.py": [(185, 223)],
    "oauthlib/oauth2/rfc6749/endpoints/revocation.py": [(1, 96)],
    "oauthlib/oauth2/rfc6749/grant_types/authorization_code.py": [(18, 65), (490, 513), (531, 550)],
}
for relative, windows in WINDOWS.items():
    path = SITE / relative
    data = path.read_bytes()
    print(relative, "sha256", hashlib.sha256(data).hexdigest())
    for start, stop in windows:
        for number, line in enumerate(data.decode("utf-8").splitlines(), 1):
            if start <= number <= stop:
                print(f"{relative}:{number}: {line}")
print("# RevocationEndpoint construction in OAuthlib's own preconfigured servers")
for relative in ("oauthlib/oauth2/rfc6749/endpoints/pre_configured.py",
                 "oauthlib/openid/connect/core/endpoints/pre_configured.py"):
    data = (SITE / relative).read_bytes()
    print(relative, "sha256", hashlib.sha256(data).hexdigest())
    lines = data.decode("utf-8").splitlines()
    for index, line in enumerate(lines):
        if "RevocationEndpoint.__init__" in line:
            for offset in range(index, min(index + 4, len(lines))):
                print(f"{relative}:{offset + 1}: {lines[offset]}")
