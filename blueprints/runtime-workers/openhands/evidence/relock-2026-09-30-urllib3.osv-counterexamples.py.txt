"""Inputs the guard must not read as "no next": for each, what the pinned OSV-Scanner finds with an EMPTY config (the ignores off), what the reader of the
first repaired head (tests/test_osv_lockfile_coverage.py at commit bd48d924, taken from git) returned, and what the current reader returns.
The first two are the counterexamples an independent verifier found against that head with the pinned scanner (a tagged `packages` key and `packages`
supplied by a YAML merge key); the third is a Unicode line separator.
Usage, from the worktree root: osv_counterexamples.py <osv-scanner binary> <old head commit>. Reads git objects, runs the scanner on scratch files; no network."""
import importlib.util
import json
import subprocess
import sys
import tempfile
from pathlib import Path

osv, old_commit = sys.argv[1], sys.argv[2]
root = Path.cwd()
sys.path.insert(0, str(root))
from tests import test_osv_lockfile_coverage as new  # noqa: E402

CASES = {
    "tagged packages key": "lockfileVersion: '9.0'\n!!str packages:\n  next@16.3.5:\n    resolution: {}\n",
    "packages through a YAML merge key": "lockfileVersion: '9.0'\nbase: &base\n  packages:\n    next@16.3.5:\n      resolution: {}\n<<: *base\n",
    "Unicode line separator": "lockfileVersion: '9.0'\npackages:   next@16.3.5:\n    resolution: {}\n",
}

with tempfile.TemporaryDirectory() as scratch:
    scratch = Path(scratch)
    (scratch / "tests").mkdir()
    (scratch / ".github").symlink_to(root / ".github")
    source = subprocess.run(["git", "show", f"{old_commit}:tests/test_osv_lockfile_coverage.py"], capture_output=True, text=True, check=True).stdout
    (scratch / "tests/old_reader.py").write_text(source, encoding="utf-8")
    spec = importlib.util.spec_from_file_location("old_reader", scratch / "tests/old_reader.py")
    old = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(old)
    empty = scratch / "empty.toml"
    empty.write_text("", encoding="utf-8")
    for name, text in CASES.items():
        lock = scratch / "pnpm-lock.yaml"
        lock.write_text(text, encoding="utf-8")
        run = subprocess.run([osv, "scan", "source", "--config", str(empty), "--no-resolve", f"--lockfile=pnpm-lock.yaml:{lock}", "--format", "json"],
                             capture_output=True, text=True)
        try:
            doc = json.loads(run.stdout)
            found = {f"{p['package']['name']}=={p['package']['version']}": sorted({v["id"] for v in p.get("vulnerabilities", [])})
                     for r in doc.get("results", []) for p in r.get("packages", []) if p.get("vulnerabilities")}
        except ValueError:
            found = f"unparsable output: {run.stdout[:80]!r}"
        print(f"{name}: scanner (empty config) exit {run.returncode} finds {found or 'nothing'}")
        print(f"    reader at {old_commit[:8]}: {old.pnpm_next_pins(text)!r} | current reader: {new.pnpm_next_pins(text)!r}")
