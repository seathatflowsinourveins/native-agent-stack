#!/usr/bin/env bash
# Published copy of the script as it ran on 2026-09-30 (it takes its host paths as arguments). It builds relock-2026-09-30-litellm.txt: it runs each command of the sections below from
# the published copies of the other scripts of this directory, prints the command as run, its output and its exit code, and masks host paths. It is documentation of what ran, not a supported tool.
# usage: builder.sh <scratch-relock> <worktree> <main-checkout> <osv-scanner binary> <out-file>
#   <scratch-relock> holds sdk-fcc102a.tar.gz (pins.json source_archive), uv-cache/, u3/ and the OSV binary's directory; <worktree> is a checkout of this repository on origin/main 29458b43
#   with this change applied; <main-checkout> is a read-only checkout of origin/main 29458b43.
set -u
R=$1; W=$2; MAIN=$3; OSV=$4; OUT=$5
E=$W/blueprints/runtime-workers/openhands/evidence
OH=blueprints/runtime-workers/openhands
SRC=$R/litellm-src
V=$R/u3/venv
export R W MAIN OSV E OH SRC V
mkdir -p "$R/tmp"
mask() { sed -e "s#$E#<evidence-dir>#g" -e "s#$W#<worktree>#g" -e "s#$MAIN#<main-checkout>#g" -e "s#$OSV#<osv-bin>#g" -e "s#$SRC#<litellm-src>#g" -e "s#$R/sdk-main#<sdk-main>#g" -e "s#$R#<scratch-relock>#g" -e "s#$HOME#~#g"; }
show() { printf '%s' "$1" | sed -e 's#"\$E"#<evidence-dir>#g; s#\$E#<evidence-dir>#g; s#"\$W"#<worktree>#g; s#\$W#<worktree>#g; s#"\$MAIN"#<main-checkout>#g; s#\$MAIN#<main-checkout>#g; s#"\$OSV"#<osv-bin>#g; s#\$OSV#<osv-bin>#g; s#"\$SRC"#<litellm-src>#g; s#\$SRC#<litellm-src>#g; s#"\$V"#<scratch-relock>/u3/venv#g; s#\$V#<scratch-relock>/u3/venv#g; s#"\$R"#<scratch-relock>#g; s#\$R#<scratch-relock>#g; s#\$OH#blueprints/runtime-workers/openhands#g'; }
# run '<command text, variables left unexpanded>' [note]: prints "$ <command with placeholders>   # <UTC time>   [note]", the output as printed, then the exit code.
run() {
  printf '$ %s     # %s%s\n' "$(show "$1")" "$(date -u +%FT%TZ)" "${2:+     # $2}"
  bash -c "$1" 2>&1 | mask
  echo "exit: ${PIPESTATUS[0]}"
}
{
cat <<'HDR'
# OpenHands recipe runtime lock: litellm 1.93.2 relock (GHSA-3cv6-jpf6-8222), 2026-09-30
# Returned outputs of the commands blueprints/runtime-workers/openhands/research.md and evidence/receipts/osv-openhands-litellm-relock-20260930.json cite.
# A line that starts with '$ ' is a command as run (the time after '#' is when it ran; a script is run from its published copy in this directory); the lines under it are its output as printed,
# then its exit code. Exceptions: the relock script abbreviates hashes in the diff of section B; host paths are masked: <worktree> (a checkout of this repository on origin/main 29458b43 with this change
# applied), <main-checkout> (a read-only checkout of origin/main 29458b43), <scratch-relock>, <osv-bin>, <litellm-src>, <evidence-dir> (this directory) and the home directory as ~; the scan step masks its scan
# root as <scan-root> in the retained scanner outputs; a command's text shows the same placeholders in place of the variables it was run with. All sections were produced by one run of
# relock-2026-09-30-litellm.builder.sh.txt, in this order, from the published copies of the other scripts (a section's own time stamps are its commands' start times).
# Retained beside this file: the published scripts (relock-2026-09-30-litellm.*.sh.txt and *.py.txt) and the four scanner outputs of section F (relock-2026-09-30-litellm.osv-{before,after}.{ordinary,frozen}.txt).
HDR
echo
echo "## A. The advisory, the release dates and the OSV query for the locked and the relocked versions (OSV API and PyPI, read-only)"
run 'python3 "$E/relock-2026-09-30-litellm.osv-advisories.py.txt"'
echo
echo "## B. The relock (uv 0.12.17, the six upgrades of the 2026-09-30 urllib3 relock plus litellm==1.93.2 and the two-line workspace edit), returned output of relock-2026-09-30-litellm.relock.sh.txt"
run 'bash "$E/relock-2026-09-30-litellm.relock.sh.txt" "$R" "$W"'
echo
echo "## C. The upstream rules the relock edits, and what changed in upstream litellm between v1.93.0 and v1.93.2"
run 'sha256sum "$R/sdk-fcc102a.tar.gz"' 'the pinned SDK archive of pins.json source_archive'
run 'for f in $(tar -tzf "$R/sdk-fcc102a.tar.gz" | grep -E "/pyproject.toml$"); do echo "== $f"; tar -xzOf "$R/sdk-fcc102a.tar.gz" "$f" | grep -n -i litellm | cut -c1-200; done' 'every pyproject.toml of the archive, its lines that name litellm'
run 'tar -xzOf "$R/sdk-fcc102a.tar.gz" "$(tar -tzf "$R/sdk-fcc102a.tar.gz" | grep -E "^[^/]+/uv.lock$")" | grep -n -A1 "^name = \"litellm\""' "the archive's root uv.lock, upstream's own lock"
run 'bash "$E/relock-2026-09-30-litellm.upstream-diff.sh.txt" "$SRC"' 'BerriAI/litellm tags v1.93.0 and v1.93.2'
run 'bash "$E/relock-2026-09-30-litellm.sdk-main.sh.txt" "$R/sdk-main"' "upstream's default branch now: OpenHands/software-agent-sdk main"
run 'uv lock --help | grep -n -A2 -E "^[[:space:]]+(-P, --upgrade-package|--exclude-newer-package)"' "the pinned uv's own help for the two options the relock uses"
echo
echo "## D. The recipe's hashed install sequence on the new lock (scratch venv), returned output of relock-2026-09-30-litellm.install-check.sh.txt"
run 'bash "$E/relock-2026-09-30-litellm.install-check.sh.txt" "$R" "$W"'
echo
echo "## E. The lock differs in the litellm entry only"
run 'diff <(git -C "$W" show HEAD:$OH/requirements.lock | grep -E "^[A-Za-z0-9_.-]+(\[[^]]*\])? ?(==| @ )" | sed "s/ \\\\$//") <(grep -E "^[A-Za-z0-9_.-]+(\[[^]]*\])? ?(==| @ )" "$W/$OH/requirements.lock" | sed "s/ \\\\$//")' 'the requirement lines of origin/main 29458b43 and of this change; exit 1 = the inputs differ, here in one line'
run 'git -C "$W" diff --numstat HEAD -- $OH/requirements.lock' 'lines added, lines deleted'
run 'echo "hash lines on origin/main: $(git -C "$W" show HEAD:$OH/requirements.lock | grep -c -e "--hash=sha256:"); in this change: $(grep -c -e "--hash=sha256:" "$W/$OH/requirements.lock")"'
echo
echo "## F. OSV-Scanner 2.6.0 (the CI pin): the workflow's two scans over the inventory on origin/main 29458b43 (before) and on this change (after)"
run '"$OSV" --version'
run 'sha256sum "$OSV"' "the binary; the pin is the next command's OSV_SCANNER_SHA256"
run 'grep -n -E "OSV_SCANNER_(VERSION|SHA256)" "$W/.github/workflows/security-scan.yml" | cut -c1-200'
run 'bash "$E/relock-2026-09-30-litellm.osv-scan-step.sh.txt" "$OSV" "$MAIN" "$E/relock-2026-09-30-litellm.osv-before"' 'before: origin/main 29458b43'
run 'bash "$E/relock-2026-09-30-litellm.osv-scan-step.sh.txt" "$OSV" "$W" "$E/relock-2026-09-30-litellm.osv-after"' 'after: this change'
echo
echo "## G. The lock file's digest and the digest recorded for it"
run 'sha256sum "$W/$OH/requirements.lock"' 'the new lock; the previous one is 1d11bae34f09707d1ad353e24c33d25c7b004f25de9821d434e065b10969559c'
run 'jq -r .requirements_sha256 "$W/$OH/pins.json"' 'pins.json of this change'
run 'git -C "$W" show HEAD:$OH/pins.json | jq -r .requirements_sha256' 'pins.json of origin/main 29458b43'
echo
echo "## H. Where litellm is used in the installed recipe venv (static text search of the installed .py files; context for the advisory, not the basis of the decision)"
run 'grep -rl --include="*.py" -E "^[[:space:]]*(from|import)[[:space:]]+litellm\.proxy|^[[:space:]]*from[[:space:]]+litellm[[:space:]]+import[[:space:]]+proxy" "$V/lib/python3.13/site-packages" | grep -v "/site-packages/litellm/" | wc -l' 'files outside the litellm package that import litellm.proxy'
run 'grep -rn --include="*.py" -E "^[[:space:]]*(from|import)[[:space:]]+litellm" "$V/lib/python3.13/site-packages/openhands" | sed -E "s#^.*/site-packages/##" | cut -c1-160' 'the imports of litellm in the installed openhands packages'
run 'grep -n -i -E "litellm" "$W/$OH/recipe.py" "$W/$OH/worker.py" | cut -c1-200' 'the recipe code that names litellm'
echo
echo "## I. The litellm hash lines against PyPI, and the carried-forward oauthlib review's premise for litellm"
run 'python3 "$E/relock-2026-09-30-litellm.hash-check.py.txt" "$R/u3/base.lock"' 'the lock of origin/main 29458b43 (litellm 1.93.0)'
run 'python3 "$E/relock-2026-09-30-litellm.hash-check.py.txt" "$W/$OH/requirements.lock"' 'the lock of this change (litellm 1.93.2)'
run 'echo "files of the installed litellm 1.93.2 that mention oauthlib: $(grep -rli oauthlib "$V/lib/python3.13/site-packages/litellm" | wc -l)"'
run 'echo "lines of the v1.93.0 to v1.93.2 source diff that mention oauthlib: $(git -C "$SRC" diff v1.93.0 v1.93.2 | grep -c -i oauthlib)"'
echo
echo "## J. The unit tests of the coverage guard, of the workflow hardening and of the recipe, from the worktree root"
run 'cd "$W" && TMPDIR="$R/tmp" python3 -B -m unittest tests.test_osv_lockfile_coverage tests.test_workflow_hardening tests.test_runtime_worker_openhands 2>&1 | tail -4'
} > "$OUT"
echo "wrote $OUT ($(wc -l < "$OUT") lines); host paths left: $(grep -c -E '/var/tmp|/home/|/tmp/' "$OUT")"
