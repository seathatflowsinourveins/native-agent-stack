"""Negative controls for the OSV guard changes of 2026-09-30: each mutant breaks one input of the guard tests in memory (nothing on disk changes) and the
named test must then fail with the expected rule; the unmutated run must pass. R1-R3 are the mutants of the cross-family review of the first head (a quoted
key beside an unquoted fixed one, an affected next in a package-lock.json, a version below the advisory's introduced release); R4-R11 those of its review of the
first repair (a pnpm tarball entry with its own name and version, a yarn berry alias, package-lock aliases and path-named entries, backtracking on whitespace, a
canary above the fix, an integer lockfileVersion); R12-R21 those of its reviews of the second to fifth (an inline pnpm record, repeated or differently cased JSON keys, an
optional peer pin that is not a package, the yarn root workspace, deep nesting, a pin in a second pnpm document). M1-M12 mutate the guard's own inputs. Run from
the repository root."""
import sys
import time
import unittest
from pathlib import Path
from unittest import mock

sys.path.insert(0, ".")
from tests import test_osv_lockfile_coverage as T  # noqa: E402

MAC = "evidence/artifacts/macos-application-20260924/variant/pnpm-lock.yaml"
LIVE = "blueprints/convergence-practice/application-delivery/pnpm-lock.yaml"
WSL = "blueprints/convergence-practice/wsl-retrieval/package-lock.json"
OH = "blueprints/runtime-workers/openhands/requirements.lock"
NEXT_TEST = "test_only_the_allowed_lock_pins_an_affected_next"
REAL_READ = Path.read_text
HIDES = "pins next 16.3.5, which the GHSA-vcvr-r3jv-pc5j ignore hides"
UNREADABLE = "cannot be read strictly"
HEAD = "lockfileVersion: '9.0'\n\npackages:\n\n"
RES = "    resolution: {integrity: sha512-x}\n"
outcomes = []


def run(test_names):
    suite = unittest.TestSuite(T.AllowedLockTests(name) for name in test_names)
    result = unittest.TestResult()
    suite.run(result)
    return result


def failure_text(result):
    return "\n".join(text for _, text in result.failures + result.errors)


def check(label, ok, detail=""):
    print(f"{'PASS' if ok else 'FAIL'}  {label}{': ' + detail if detail else ''}")
    outcomes.append(ok)


def reading(replacements):
    """Path.read_text returns the given text for a path that ends with a key; every other file is read normally."""
    def read_text(self, *args, **kwargs):
        for suffix, text in replacements.items():
            if str(self).endswith(suffix):
                return text
        return REAL_READ(self, *args, **kwargs)
    return mock.patch.object(Path, "read_text", read_text)


def guarded(replacements, extra_entries=(), mutate_allowed=None):
    """Run the next test with the mutation applied and return its result."""
    entries = T.AllowedLockTests.inventory["lockfiles"]
    added = [dict(entry) for entry in extra_entries]
    entries.extend(added)
    kept = dict(T.IGNORE_ALLOWED_LOCKS)
    try:
        if mutate_allowed:
            mutate_allowed(T.IGNORE_ALLOWED_LOCKS)
        with reading(replacements or {}):
            return run([NEXT_TEST])
    finally:
        for entry in added:
            entries.remove(entry)
        T.IGNORE_ALLOWED_LOCKS.clear()
        T.IGNORE_ALLOWED_LOCKS.update(kept)


def caught(label, expect, replacements=None, extra_entries=(), mutate_allowed=None):
    """The next test must fail and its message must contain `expect`."""
    result = guarded(replacements, extra_entries, mutate_allowed)
    text = failure_text(result)
    check(label, bool(result.failures + result.errors) and expect in text, f"{len(result.failures + result.errors)} failing test(s); expected rule {expect!r} {'seen' if expect in text else 'NOT seen'}")


def clean(label, replacements):
    """The findings must stay empty: the mutated input is not a package OSV reports, so the guard must not flag it."""
    with reading(replacements):
        findings = T.npm_next_findings(T.AllowedLockTests.inventory["lockfiles"])
    check(label, findings == [], "the guard stays quiet" if findings == [] else repr(findings)[:200])


all_names = [n for n in dir(T.AllowedLockTests) if n.startswith("test_")]
base = run(all_names)
check("unmutated", base.wasSuccessful(), f"ran {base.testsRun}, failures {len(base.failures)}, errors {len(base.errors)}")
scope = T.IGNORE_SCOPES[T.NEXT_ADVISORY]
FIXED_AND_QUOTED = HEAD + "  next@16.3.6:\n" + RES + "\n  {q}next@16.3.5{q}:\n" + RES

caught("R1a a double-quoted affected key beside an unquoted fixed one (live pnpm lock)", UNREADABLE, {LIVE: FIXED_AND_QUOTED.replace("{q}", '"')})
caught("R1b a single-quoted affected key beside an unquoted fixed one (live pnpm lock)", HIDES, {LIVE: FIXED_AND_QUOTED.replace("{q}", "'")})
caught("R2 an affected next in the retrieval recipe's package-lock.json", HIDES, {WSL: '{"lockfileVersion":3,"packages":{"node_modules/next":{"version":"16.3.5"}}}'})
check("R3 a version below the introduced release is not affected", T.npm_affected("16.1.9", scope["fixed"], scope["introduced"]) is False and T.npm_affected("16.2.0", scope["fixed"], scope["introduced"]) is True)
caught("R4 a pnpm tarball entry with its own name and version (live pnpm lock)", HIDES,
       {LIVE: HEAD + "  next@https://registry.npmjs.org/next/-/next-16.3.5.tgz:\n    resolution: {tarball: https://registry.npmjs.org/next/-/next-16.3.5.tgz}\n    version: 16.3.5\n"})
caught("R5 a yarn berry alias to next in an extra inventory lock", HIDES,
       {"zz/yarn.lock": '__metadata:\n  version: 8\n\n"foo@npm:next@16.3.5":\n  version: 16.3.5\n  resolution: "next@npm:16.3.5"\n'}, extra_entries=[{"path": "zz/yarn.lock"}])
caught("R6 a package-lock lockfileVersion 1 alias to next", HIDES, {WSL: '{"lockfileVersion":1,"dependencies":{"foo":{"version":"npm:next@16.3.5"}}}'})
caught("R7 a package-lock entry node_modules/next with an empty name", HIDES, {WSL: '{"lockfileVersion":3,"packages":{"node_modules/next":{"name":"","version":"16.3.5"}}}'})
caught("R8 a package-lock entry vendor/next without a name", HIDES, {WSL: '{"lockfileVersion":3,"packages":{"vendor/next":{"version":"16.3.5"}}}'})
heavy = "  foo@1.0.0:\n" + RES + "    peerDependenciesMeta:\n      next:\n" + " " * 160 + "optional: true\n" + " " * 160 + "injected: true\n" + " " * 160 + "other: true\n"
started = time.perf_counter()
result_heavy = T.pnpm_next_pins("lockfileVersion: 9.0\npackages:\n" + heavy * 500)
check("R9 whitespace-heavy input returns in linear time", result_heavy == [] and time.perf_counter() - started < 2.0, f"{time.perf_counter() - started:.3f} s for {len(heavy * 500)} bytes")
check("R10 a canary above the fix, build metadata on the fix and a canary of the introduced release are not affected",
      T.npm_affected("16.4.0-canary.1", scope["fixed"], scope["introduced"]) is False and T.npm_affected("16.3.6+build.1", scope["fixed"], scope["introduced"]) is False
      and T.npm_affected("16.2.0-canary.1", scope["fixed"], scope["introduced"]) is False and T.npm_affected("16.3.6-canary.1", scope["fixed"], scope["introduced"]) is True)
check("R11 an integer lockfileVersion is read", T.pnpm_next_pins("lockfileVersion: 9\n\npackages:\n\n  next@16.3.6:\n" + RES) == ["16.3.6"])
caught("R12 an inline pnpm record with its own name and version (live pnpm lock)", UNREADABLE, {LIVE: HEAD + "  file:vendor/next: {name: next, version: 16.3.5}\n"})
caught("R13 a repeated packages key in a package-lock.json", UNREADABLE, {WSL: '{"lockfileVersion":3,"packages":{"node_modules/next":{"version":"16.3.5"}},"packages":{}}'})
caught("R14 a differently cased Packages key in a package-lock.json", UNREADABLE, {WSL: '{"lockfileVersion":3,"Packages":{"node_modules/next":{"version":"16.3.5"}}}'})
clean("R15 an optional peer declaration of next is not a package", {LIVE: HEAD + "  plugin@1.0.0:\n" + RES + "    peerDependencies:\n      next: 16.3.5\n    peerDependenciesMeta:\n      next:\n        optional: true\n"})
check("R16 the yarn root workspace is skipped", T.yarn_lock_next_pins('__metadata:\n  version: 8\n"next@workspace:.":\n  version: 16.3.5\n') == [])
deep = '{"lockfileVersion":1,"dependencies":' + '{"p":{"dependencies":' * 1200 + '{"next":{"version":"16.3.5"}}' + '}}' * 1200 + '}'
try:
    deep_result = T.package_lock_next_pins(deep)
    check("R17 deep nesting is read or refused, never an error or an empty read", deep_result in (None, ["16.3.5"]), repr(deep_result))
except RecursionError:
    check("R17 deep nesting is read or refused, never an error or an empty read", False, "RecursionError")
caught("R19 a package-lock lockfileVersion 1 dependency named next installed from a local path", UNREADABLE, {WSL: '{"lockfileVersion":1,"dependencies":{"next":{"version":"file:vendor/next"}}}'})
caught("R20 a package-lock lockfileVersion 1 dependency named next installed from a git commit", UNREADABLE,
       {WSL: '{"lockfileVersion":1,"dependencies":{"next":{"version":"git+https://github.com/vercel/next.js.git#0123456789abcdef0123456789abcdef01234567"}}}'})
caught("R21 a package-lock entry for next whose resolved URL is a git commit, with a fixed-looking version", UNREADABLE,
       {WSL: '{"lockfileVersion":3,"packages":{"node_modules/next":{"version":"16.3.6","resolved":"git+ssh://git@github.com/vercel/next.js.git#0123456789abcdef0123456789abcdef01234567"}}}'})
caught("R18 a pin in the second document of a two-document pnpm lock (live pnpm lock)", HIDES, {LIVE: "---\n" + HEAD + "  react@19.3.0:\n" + RES + "\n---\n" + HEAD + "  next@16.3.5:\n" + RES})

caught("M1 an entry key at the wrong indentation", UNREADABLE, {LIVE: HEAD + "    next@16.3.5:\n" + RES})
caught("M2 an old slash key in a lockfileVersion 9 file", UNREADABLE, {LIVE: HEAD + "  /next/16.3.5:\n" + RES})
clean("M3 an importer-only pin is not a package", {LIVE: "lockfileVersion: '9.0'\n\nimporters:\n  .:\n    dependencies:\n      next:\n        specifier: ^16.3.0\n        version: 16.3.5(react@19.3.0)\n\n"
                                                       "packages:\n\n  react@19.3.0:\n" + RES})
caught("M4 an unknown lockfileVersion", UNREADABLE, {LIVE: "lockfileVersion: '10.0'\n\npackages:\n\n  next@16.3.5:\n" + RES})
caught("M5 an extra yarn.lock in the inventory pins an affected next", HIDES, {"zz/yarn.lock": '# yarn lockfile v1\n\nnext@^16.3.0:\n  version "16.3.5"\n'}, extra_entries=[{"path": "zz/yarn.lock"}])
caught("M6 an extra lock in a format no reader knows (bun.lockb)", UNREADABLE, {}, extra_entries=[{"path": "zz/bun.lockb"}])
caught("M7 the macOS row removed", HIDES, mutate_allowed=lambda allowed: allowed.pop(MAC))
caught("M8 the allowed lock no longer pins next", "the allowed lock no longer pins next", {MAC: HEAD + "  react@19.3.0:\n" + RES})
caught("M9 the allowed lock cannot be read strictly", UNREADABLE, {MAC: "lockfileVersion: '10.0'\n"})

# a wrong sha256 in either exact-lock row must fail the digest test. The checks bind IGNORE_ALLOWED_LOCKS as a default argument at definition time, so
# the entry is mutated in place and restored.
for label, key in (("M10 OpenHands row sha256 wrong", OH), ("M11 macOS row sha256 wrong", MAC)):
    kept = T.IGNORE_ALLOWED_LOCKS[key]["sha256"]
    T.IGNORE_ALLOWED_LOCKS[key]["sha256"] = "0" * 64
    try:
        result = run([n for n in all_names if n != NEXT_TEST])
    finally:
        T.IGNORE_ALLOWED_LOCKS[key]["sha256"] = kept
    problems = result.failures + result.errors
    check(label, any("test_every_allowed_lock_matches_its_reviewed_digest" in str(t[0]) and key in t[1] for t in problems),
          f"{len(problems)} failing test(s): {[str(t[0]).split()[0] for t in problems]}")

# the next ignore entry without its scope must fail.
scopes = {k: v for k, v in T.IGNORE_SCOPES.items() if k != T.NEXT_ADVISORY}
with mock.patch.object(T, "IGNORE_SCOPES", scopes):
    result = run([n for n in all_names if n != NEXT_TEST])
    problems = result.failures + result.errors
    check("M12 the next scope entry removed", any("test_allowed_advisories_are_scoped_active_ignores" in str(t[0]) for t in problems),
          f"{len(problems)} failing test(s): {[str(t[0]).split()[0] for t in problems]}")

# the unmutated tree must still pass after all of the above (every mutation was restored)
again = run(all_names)
check("unmutated again", again.wasSuccessful(), f"ran {again.testsRun}, failures {len(again.failures)}, errors {len(again.errors)}")

print("ALL MUTANTS CAUGHT AND CONTROL CLEAN" if all(outcomes) else "SOMETHING DID NOT BEHAVE AS EXPECTED")
sys.exit(0 if all(outcomes) else 1)
